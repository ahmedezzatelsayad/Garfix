     1→/**
     2→ * accounting.ts — React Query hooks for Accounting CRUD and report operations.
     3→ *
     4→ * Provides typed query and mutation hooks for chart of accounts,
     5→ * journal entries (including reversal), and financial reports
     6→ * (profit & loss, balance sheet, cash flow, trial balance). All hooks
     7→ * use the centralized `queryKeys` factory for granular cache invalidation
     8→ * and the typed `apiGet`/`apiPost`/`apiDelete` helpers for consistent requests.
     9→ */
    10→"use client";
    11→
    12→import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
    13→import { apiGet, apiPost, apiPatch, apiDelete, apiUpload, apiDownloadBlob, ApiError } from "@/hooks/api-client";
    14→import { queryKeys } from "@/hooks/query-keys";
    15→
    16→// ─── Types ──────────────────────────────────────────────────────────────────

    22→  name: string;

    24→  nameEn?: string;

    27→  currency: string;

   136→/**
   137→ * Fetch the chart of accounts for a given company.
   138→ *
   139→ * The query is disabled when `companySlug` is empty, preventing
   140→ * unnecessary requests before the active company is known.
   141→ *
   142→ * @param companySlug - Slug of the company whose accounts to fetch.
   143→ */
   144→export function useAccounts(companySlug: string) {
   145→  return useQuery<AccountListResponse, ApiError>({
   146→    queryKey: queryKeys.accounting.accounts(companySlug),
   147→    queryFn: () =>
   148→      apiGet<AccountListResponse>(
   149→        `/api/accounting/accounts?companySlug=${encodeURIComponent(companySlug)}`,
   150→      ),
   151→    enabled: !!companySlug,
   152→  });
   153→}
   154→
   155→/**
   156→ * Create a new account in the chart of accounts.
   157→ *
   158→ * On success all account queries for the relevant company are
   159→ * invalidated so every mounted list view refetches with the new entry.
   160→ */
   161→export function useCreateAccount() {
   162→  const queryClient = useQueryClient();
   163→
   164→  return useMutation<AccountResponse, ApiError, CreateAccountPayload>({
   165→    mutationFn: (payload) =>
   166→      apiPost<CreateAccountPayload, AccountResponse>(
   167→        "/api/accounting/accounts",
   168→        payload,
   169→      ),
   170→    onSuccess: (_data, variables) => {
   171→      void queryClient.invalidateQueries({
   172→        queryKey: queryKeys.accounting.accounts(variables.companySlug),
   173→      });
   174→    },
   175→  });
   176→}
   177→
   178→/**
   179→ * Delete a single account from the chart of accounts.
   180→ *
   181→ * On success all account queries are invalidated to ensure the
   182→ * deleted entry no longer appears in any list.
   183→ *
   184→ * @param variables - Object containing the account `id` and `companySlug`.
   185→ */
   186→export function useDeleteAccount() {
   187→  const queryClient = useQueryClient();
   188→
   189→  return useMutation<
   190→    void,
   191→    ApiError,
   192→    { id: number; companySlug: string }
   193→  >({
   194→    mutationFn: ({ id }) => apiDelete<void>(`/api/accounting/accounts/${id}`),
   195→    onSuccess: (_data, variables) => {
   196→      void queryClient.invalidateQueries({
   197→        queryKey: queryKeys.accounting.accounts(variables.companySlug),
   198→      });
   199→    },
   200→  });
   201→}

   286→/**
   287→ * Reverse a journal entry by creating an offsetting entry.
   288→ *
   289→ * On success all journal entry queries for the relevant company are
   290→ * invalidated, since the reversal creates a new entry. Account queries
   291→ * and financial report queries are also invalidated as balances change.
   292→ *
   293→ * @param variables - Object containing the journal entry `id` and an optional `reason`.
   294→ */
   295→export function useReverseJournalEntry() {
   296→  const queryClient = useQueryClient();
   297→
   298→  return useMutation<
   299→    JournalEntryResponse,
   300→    ApiError,
   301→    ReverseJournalEntryPayload
   302→  >({
   303→    mutationFn: (variables) => {
   304→      const { id, ...body } = variables;
   305→      return apiPost<typeof body, JournalEntryResponse>(
   306→        `/api/accounting/journal-entries/${id}/reverse`,
   307→        body,
   308→      );
   309→    },
   310→    onSuccess: () => {
   311→      // Invalidate all journal entry queries across all companies
   312→      // since the payload may not contain companySlug
   313→      void queryClient.invalidateQueries({
   314→        predicate: (query) => {
   315→          const key = query.queryKey;
   316→          return (
   317→            Array.isArray(key) &&
   318→            key[0] === "accounting" &&
   319→            key[1] === "journal-entries"
   320→          );
   321→        },
   322→      });
   323→      // Invalidate all account queries since balances may change
   324→      void queryClient.invalidateQueries({
   325→        predicate: (query) => {
   326→          const key = query.queryKey;
   327→          return (
   328→            Array.isArray(key) &&
   329→            key[0] === "accounting" &&
   330→            key[1] === "accounts"
   331→          );
   332→        },
   333→      });
   334→      // Invalidate financial reports since they depend on journal entries
   335→      void queryClient.invalidateQueries({
   336→        predicate: (query) => {
   337→          const key = query.queryKey;
   338→          return (
   339→            Array.isArray(key) &&
   340→            key[0] === "accounting" &&
   341→            (key[1] === "profit-loss" ||
   342→              key[1] === "balance-sheet" ||
   343→              key[1] === "cash-flow" ||
   344→              key[1] === "trial-balance")
   345→          );
   346→        },
   347→      });
   348→    },
   349→  });
   350→}
