   285→function AnimatedKPICard({
   286→  title,
   287→  value,
   288→  subtitle,
   289→  trend,
   290→  icon,
   291→  color = 'emerald',
   292→  delay = 0,
   293→}: {
   294→  title: string;
   295→  value: number;
   296→  subtitle?: string;
   297→  trend?: { direction: 'up' | 'down'; value: number };
   298→  icon?: React.ReactNode;
   299→  color?: 'emerald' | 'blue' | 'gold' | 'red' | 'purple';

   311→
   312→  return (
   313→    <FadeUp delay={delay}>
   314→      <MotionCard className="overflow-hidden">

   318→              <p className="text-white/80 text-sm font-medium">{title}</p>

   330→                <p className="text-white/70 text-xs mt-1">{subtitle}</p>

   467→            dismissible
   468→            size="sm"
   469→            className="animate-in slide-in-from-right-2"
   470→          >
   471→            <div className="flex items-start gap-2">
   472→              <span className="text-lg">
   473→                {alert.level === 'critical' ? '🚨' :

   491→/**
   492→ * Main Enhanced Dashboard Component
   493→ */
   494→export function GarfixEnhancedAIDashboard() {
   495→  const [metrics, setMetrics] = useState<AIMetricsData | null>(null);
   496→  const [isLoading, setIsLoading] = useState(true);
   497→  const [activeTab, setActiveTab] = useState('overview');
   498→  const { resolvedTheme, toggleTheme } = useTheme();
   499→  const { shouldReduceMotion } = useReducedMotion();
   500→  const aiContext = useAIPersonalization();
   501→
   502→  // Fetch AI Metrics from Real API
   503→  const fetchMetrics = useCallback(async () => {
   504→    try {
   505→      setIsLoading(true);
   506→      
   507→      // 🔄 Call Real API: /api/ai/metrics
   508→      const response = await fetch('/api/ai/metrics');
   509→      
   510→      if (response.ok) {
   511→        const apiData = await response.json();
   512→        
   513→        if (apiData.success) {
   514→          // ✅ Use real data from API
   515→          setMetrics(apiData);
   516→          
   517→          // Track view event for personalization
   518→          aiContext?.trackEvent('dashboard_view', { 
   519→            tab: activeTab, 
   520→            dataSource: 'real_api' 
   521→          });
   522→          return;
   523→        }
   524→      }
   525→      
   526→      // ⚠️ Fallback to mock data if API fails
   527→      console.warn('AI Metrics API unavailable, using fallback data');
   528→      setMetrics(MOCK_AI_METRICS);
   529→      aiContext?.trackEvent('dashboard_view', { 
   530→        tab: activeTab, 
   531→        dataSource: 'mock_fallback' 
   532→      });
   533→      
   534→    } catch (error) {
   535→      console.error('Failed to fetch AI metrics:', error);
   536→      // ❌ Error fallback - use mock data
   537→      setMetrics(MOCK_AI_METRICS);
   538→    } finally {
   539→      setIsLoading(false);
   540→    }
   541→  }, [activeTab, aiContext]);
   542→
   543→  useEffect(() => {
   544→    fetchMetrics();
   545→    
   546→    // Auto-refresh every 15 seconds
   547→    const interval = setInterval(fetchMetrics, 15000);
   548→    return () => clearInterval(interval);
   549→  }, [fetchMetrics]);
   550→
   551→  // Get personalized recommendations
   552→  const personalizedRecs = useMemo(() => {
   553→    if (!aiContext?.recommendations.length) return MOCK_RECOMMENDATIONS;
   554→    
   555→    // Combine mock data with AI-personalized suggestions
   556→    const userPrefs = aiContext.userPreferences;
   557→    return MOCK_RECOMMENDATIONS.map(rec => ({
   558→      ...rec,
   559→      priority: userPrefs.preferredCategory === rec.category 
   560→        ? ('high' as const) 
   561→        : rec.priority,
   562→    }));
   563→  }, [aiContext?.recommendations, aiContext?.userPreferences]);
   564→
   565→  // Loading State
   566→  if (isLoading && !metrics) {
   567→    return (
   568→      <GarfixPageTransition showLoading isReady={false}>
   569→        <GarfixContainer padding="lg">
   570→          <DashboardSkeleton />
   571→        </GarfixContainer>
   572→      </GarfixPageTransition>
   573→    );
   574→  }
   575→
   576→  return (
   577→    <GarfixPageTransition enterAnimation="slideUp">
   578→      <div className="min-h-screen bg-gradient-to-br from-gray-50 to-gray-100 dark:from-gray-950 dark:to-gray-900" dir="rtl" lang="ar">
   579→        
   580→        {/* Header */}
   581→        <header className="bg-white dark:bg-gray-900 border-b border-gray-200 dark:border-gray-800 sticky top-0 z-40">
   582→          <GarfixContainer>
   583→            <div className="flex items-center justify-between h-16">
   584→              <div className="flex items-center gap-4">
   585→                <h1 className="text-xl font-bold text-gray-900 dark:text-white flex items-center gap-2">
   586→                  <span className="text-2xl">🤖</span>
   587→                  لوحة تحكم الذكاء الاصطناعي
   588→                </h1>
   589→                <GarfixBadge variant="primary" size="sm">
   590→                  v4.0
   591→                </GarfixBadge>
   592→                
   593→                {/* Data Source Indicator */}
   594→                {metrics && (
   595→                  <GarfixBadge 
   596→                    variant={metrics.timestamp?.includes('2026') ? "success" : "warning"} 
   597→                    size="sm"
   598→                    title={metrics.timestamp ? `آخر تحديث: ${new Date(metrics.timestamp).toLocaleTimeString('ar-EG')}` : 'بيانات تجريبية'}
   599→                  >
   600→                    {metrics.timestamp?.includes('2026') ? '🟢 API حقيقي' : '⚠️ تجريبي'}
   601→                  </GarfixBadge>
   602→                )}
   603→              </div>
   604→              
   605→              <div className="flex items-center gap-3">
   606→                {/* Theme Toggle */}

   981→function formatNumber(num: number): string {
   982→  if (num >= 1_000_000) return `${(num / 1_000_000).toFixed(1)}M`;
   983→  if (num >= 1_000) return `${(num / 1_000).toFixed(1)}K`;
   984→  return num.toString();
   985→}
   986→
   987→function formatTime(ms: number): string {
   988→  if (ms < 1000) return `${ms}ms`;
   989→  return `${(ms / 1000).toFixed(1)}s`;
   990→}
