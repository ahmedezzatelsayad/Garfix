   200→
   201→  // Canvas particle animation (same as original LandingPage)
   202→  useEffect(() => {
   203→    const c = cvsRef.current;
   204→    if (!c) return;
   205→    const ctx = c.getContext("2d");
   206→    if (!ctx) return;
   207→    let raf: number;
   208→
   209→    const dpr = typeof window !== "undefined" ? Math.min(window.devicePixelRatio || 1, 2) : 1;
   210→
   211→    const resize = () => {
   212→      const w = c.offsetWidth;
   213→      const h = c.offsetHeight;
   214→      c.width = w * dpr;
   215→      c.height = h * dpr;
   216→      ctx.scale(dpr, dpr);
   217→      c.dataset.logicalW = String(w);
   218→      c.dataset.logicalH = String(h);
   219→    };
   220→    resize();
   221→    window.addEventListener("resize", resize);
   222→
   223→    const isMobile = window.innerWidth < 768;
   224→    const particleCount = isMobile ? 20 : 50;
   225→    const connectionDist = isMobile ? 80 : 140;
   226→
   227→    const pts = Array.from({ length: particleCount }, () => ({
   228→      x: Math.random() * (parseInt(c.dataset.logicalW || "2000")),
   229→      y: Math.random() * (parseInt(c.dataset.logicalH || "1200")),
   230→      vx: (Math.random() - 0.5) * 0.1,
   231→      vy: (Math.random() - 0.5) * 0.1,
   232→      r: Math.random() * (isMobile ? 0.8 : 1.2) + 0.3,
   233→      o: Math.random() * 0.25 + 0.05,
   234→    }));
   235→
   236→    const draw = () => {
   237→      const w = parseInt(c.dataset.logicalW || String(c.offsetWidth));
   238→      const h = parseInt(c.dataset.logicalH || String(c.offsetHeight));
   239→      ctx.clearRect(0, 0, w, h);
   240→      for (let i = 0; i < pts.length; i++) {
   241→        for (let j = i + 1; j < pts.length; j++) {
   242→          const dx = pts[i].x - pts[j].x;
   243→          const dy = pts[i].y - pts[j].y;
   244→          const d = Math.hypot(dx, dy);
   245→          if (d < connectionDist) {
   246→            ctx.beginPath();
   247→            ctx.moveTo(pts[i].x, pts[i].y);
   248→            ctx.lineTo(pts[j].x, pts[j].y);
   249→            ctx.strokeStyle = `rgba(167,139,250,${0.04 * (1 - d / connectionDist)})`;
   250→            ctx.lineWidth = isMobile ? 0.4 : 0.6;
   251→            ctx.stroke();
   252→          }
   253→        }
   254→      }
   255→      pts.forEach((p) => {
   256→        ctx.beginPath();
   257→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   258→        ctx.fillStyle = `rgba(167,139,250,${p.o})`;
   259→        ctx.fill();
   260→        p.x += p.vx; p.y += p.vy;
   261→        if (p.x < 0 || p.x > w) p.vx *= -1;
   262→        if (p.y < 0 || p.y > h) p.vy *= -1;
   263→      });
   264→      raf = requestAnimationFrame(draw);
   265→    };
   266→    draw();
   267→    return () => { cancelAnimationFrame(raf); window.removeEventListener("resize", resize); };
   268→  }, []);

   280→      className="min-h-dvh bg-[linear-gradient(180deg,#0f0a1e_0%,#1a1035_35%,#12082e_70%,#0f0a1e_100%)] text-white overflow-x-hidden"
   281→    >
