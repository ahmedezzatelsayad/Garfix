    92→export function AccountingView() {
    93→  const { activeCompany } = useBrand();
    94→  const [moduleTab, setModuleTab] = useState<ModuleTab>("core");
    95→
    96→  /* ─── Core sub-module state ───────────────────────────────────────────── */
    97→  const [tab, setTab] = useState<Tab>("accounts");
    98→  const [showForm, setShowForm] = useState(false);
    99→  const [selectedIds, setSelectedIds] = useState<Set<number>>(new Set());
   100→  const [currentPage, setCurrentPage] = useState(1);
   101→  const [bulkDeleting, setBulkDeleting] = useState(false);
   102→  const [reversingId, setReversingId] = useState<number | null>(null);
   103→  const [reverseConfirm, setReverseConfirm] = useState<JournalEntry | null>(null);
   104→
   105→  const slug = activeCompany ? encodeURIComponent(activeCompany.slug) : "";
   106→
   107→  // TanStack Query hooks for data fetching
   108→  const accountsQuery = useAccounts(slug);
   109→  const journalEntriesQuery = useJournalEntries(slug);
   110→  const trialBalanceQuery = useTrialBalance(slug);
   111→  const fiscalPeriodsQuery = useFiscalPeriods(slug);
   112→  const costCentersQuery = useCostCenters(slug);
   113→  const agingQuery = useAging(slug);
   114→  const bankAccountsListQuery = useBankAccountsList(slug);
   115→
   116→  const accounts = accountsQuery.data?.accounts ?? [];
   117→  const entries = (journalEntriesQuery.data as { entries?: JournalEntry[] } | undefined)?.entries ?? (journalEntriesQuery.data?.journalEntries ?? []);
   118→  const trial = (trialBalanceQuery.data as { accounts?: TrialRow[]; grandDebit?: number; grandCredit?: number; isBalanced?: boolean } | null) ?? null;
   119→  const fiscalPeriods = (fiscalPeriodsQuery.data as { periods?: FiscalPeriod[] } | undefined)?.periods ?? [];
   120→  const costCenters = (costCentersQuery.data as { costCenters?: CostCenter[] } | undefined)?.costCenters ?? [];
   121→  const agingData = (agingQuery.data as { buckets?: AgingBucket[] } | undefined)?.buckets ?? [];
   122→  const bankAccounts = (bankAccountsListQuery.data as { accounts?: BankAccount[] } | undefined)?.accounts ?? [];
   123→
   124→  const loading = accountsQuery.isLoading || journalEntriesQuery.isLoading;
   125→
   126→  // Mutation hooks
   127→  const deleteAccountMutation = useDeleteAccount();
   128→  const deleteJournalEntryMutation = useDeleteJournalEntry();
   129→  const reverseJournalEntryMutation = useReverseJournalEntry();
   130→  const createAccountMutation = useCreateAccount();
   131→  const createJournalEntryMutation = useCreateJournalEntry();
   132→  const closeFiscalPeriodMutation = useCloseFiscalPeriod();

   135→  const createCostCenterMutation = useCreateCostCenter();

   138→  const load = () => { accountsQuery.refetch(); journalEntriesQuery.refetch(); };
   139→  const loadTrial = () => { trialBalanceQuery.refetch(); };
   140→  const loadFiscalPeriods = () => { fiscalPeriodsQuery.refetch(); };
   141→  const loadCostCenters = () => { costCentersQuery.refetch(); };
   142→  const loadAging = () => { agingQuery.refetch(); };
   143→  const loadBankAccounts = () => { bankAccountsListQuery.refetch(); };

   155→    setTab(t);
   156→    setShowForm(false);
   157→    setSelectedIds(new Set());
   158→    setCurrentPage(1);
   159→  };

   161→  const itemsForTab = (): Array<{ id: number }> => (tab === "accounts" ? accounts : entries);
   162→
   163→  const allItems = itemsForTab();

   169→    if (selectedIds.size === pageItems.length && pageItems.length > 0) setSelectedIds(new Set());
   170→    else setSelectedIds(new Set(pageItems.map((i) => i.id)));
   171→  };

   173→    setSelectedIds((prev) => {
   174→      const next = new Set(prev);
   175→      if (next.has(id)) next.delete(id);
   176→      else next.add(id);
   177→      return next;
   178→    });

   181→    if (selectedIds.size === 0) return;
   182→    if (!confirm(`حذف ${selectedIds.size} عنصر؟`)) return;

   185→    const mutation = tab === "accounts" ? deleteAccountMutation : deleteJournalEntryMutation;
   186→    for (const id of selectedIds) {
   187→      try {
   188→        await mutation.mutateAsync({ id, companySlug: activeCompany!.slug });
   189→        okCount++;
   190→      } catch { failCount++; }
   191→    }
   192→    setBulkDeleting(false);
   193→    setSelectedIds(new Set());
   194→    if (okCount > 0) toast.success(`تم حذف ${okCount} عنصر`);
   195→    if (failCount > 0) toast.error(`تعذّر حذف ${failCount} عنصر`);
   196→    load();
   197→  };
   198→  const handleDelete = (id: number) => {
   199→    if (!confirm("حذف هذا العنصر؟")) return;
   200→    const mutation = tab === "accounts" ? deleteAccountMutation : deleteJournalEntryMutation;
   201→    mutation.mutate(
   202→      { id, companySlug: activeCompany!.slug },
   203→      {
   204→        onSuccess: () => { toast.success("تم الحذف"); load(); },
   205→        onError: (err) => { toast.error(err.message || "تعذّر الحذف"); },
   206→      },
   207→    );
   208→  };
   209→
   210→  const handleReverse = async (entry: JournalEntry) => {
   211→    if (!activeCompany) return;
   212→    setReverseConfirm(entry);
   213→  };
   214→
   215→  const confirmReverse = () => {
   216→    if (!reverseConfirm || !activeCompany) return;
   217→    const entry = reverseConfirm;
   218→    setReversingId(entry.id);
   219→    reverseJournalEntryMutation.mutate(
   220→      { id: entry.id, reason: "عكس القيد", companySlug: activeCompany.slug },
   221→      {
   222→        onSuccess: () => {
   223→          toast.success("تم عكس القيد");
   224→          setReverseConfirm(null);
   225→          load();
   226→          setReversingId(null);
   227→        },
   228→        onError: (err) => {
   229→          toast.error(err.message || "خطأ أثناء العكس");
   230→          setReversingId(null);
   231→        },
   232→      },
   233→    );
   234→  };
   235→
   236→  const pageBtnStyle = (disabled: boolean): string =>
   237→    disabled
   238→      ? "py-1.5 px-3 rounded-md bg-transparent text-muted-foreground border border-border text-[12px] font-bold cursor-not-allowed opacity-50"
   239→      : "py-1.5 px-3 rounded-md bg-card text-foreground border border-border text-[12px] font-bold cursor-pointer";
   240→
   241→  if (!activeCompany) return <div className="p-8 md:p-12 text-center text-muted-foreground">اختر شركة</div>;
   242→
   243→  const ACCOUNT_TYPES: Record<string, { label: string; color: string; badge: string }> = {
   244→    asset: { label: "أصول", color: "#10b981", badge: "bg-emerald-500/15 text-emerald-500" },
   245→    liability: { label: "خصوم", color: "#ef4444", badge: "bg-red-500/15 text-red-500" },
   246→    equity: { label: "حقوق ملكية", color: "#047857", badge: "bg-emerald-600/15 text-emerald-600" },
   247→    revenue: { label: "إيرادات", color: "#3b82f6", badge: "bg-blue-500/15 text-blue-500" },
   248→    expense: { label: "مصروفات", color: "#f59e0b", badge: "bg-amber-500/15 text-amber-500" },
   249→    contra_revenue: { label: "مقابل إيرادات", color: "#9ca3af", badge: "bg-gray-400/15 text-gray-400" },
   250→    contra_asset: { label: "مقابل أصول", color: "#9ca3af", badge: "bg-gray-400/15 text-gray-400" },
   251→  };
   252→
   253→  const tabs: Array<{ key: Tab; label: string; icon?: React.ComponentType<{ size?: number }> }> = [
   254→    { key: "dashboard", label: "لوحة التحكم", icon: LayoutDashboard },
   255→    { key: "accounts", label: `الحسابات (${accounts.length})` },
   256→    { key: "journal", label: `القيود (${entries.length})` },
   257→    { key: "trial", label: "ميزان المراجعة", icon: Scale },
   258→    { key: "statements", label: "القوائم المالية", icon: FileBarChart },
   259→    { key: "fiscal-periods", label: "الفترات المالية", icon: Clock },
   260→    { key: "cost-centers", label: "مراكز التكلفة", icon: Building },
   261→    { key: "aging", label: "تقادم الذمم", icon: BarChart3 },
   262→    { key: "banking", label: "البنوك", icon: Landmark },
   263→  ];
   264→
   265→  // Dashboard metrics
   266→  const totalRevenue = accounts.filter(a => a.type === "revenue").reduce((s, a) => s + a.balance, 0);
   267→  const totalExpenses = accounts.filter(a => a.type === "expense").reduce((s, a) => s + a.balance, 0);
   268→  const totalAssets = accounts.filter(a => a.type === "asset").reduce((s, a) => s + a.balance, 0);
   269→  const totalLiabilities = accounts.filter(a => a.type === "liability").reduce((s, a) => s + a.balance, 0);

   271→  const totalAR = agingData.reduce((s, b) => s + b.receivable, 0);
   272→  const totalAP = agingData.reduce((s, b) => s + b.payable, 0);
   273→  const totalCash = bankAccounts.reduce((s, b) => s + b.balance, 0);
   274→  const currentPeriod = fiscalPeriods.find(p => p.status === "open");
   275→
   276→  /* ─── Render ──────────────────────────────────────────────────────────── */
   277→  return (
   278→    <div className="flex flex-col gap-4">
   279→      {/* Module-level header */}
   280→      <div className="flex flex-wrap justify-between items-center gap-3">
   281→        <div><h1 className="text-xl md:text-2xl font-extrabold flex items-center gap-2"><Calculator size={20} /> المحاسبة</h1><p className="text-[13px] text-muted-foreground">{activeCompany.nameAr || activeCompany.name}</p></div>
   282→      </div>
   283→
   284→      {/* Module-level navigation tabs (top row, visually distinct) */}

   296→                  ? "bg-primary text-primary-foreground shadow-sm"
   297→                  : "bg-card text-muted-foreground border border-border hover:bg-accent hover:text-accent-foreground",
   298→              )}
   299→            >
   300→              <Icon size={16} />

   306→
   307→      {/* ─── Module content ─────────────────────────────────────────────── */}

   309→        /* ─── Core sub-module: exact original inner content ───────────── */
   310→        <>
   311→          <div className="flex flex-wrap justify-between items-center gap-3">

   315→          </div>
   316→          <div className="flex gap-1.5 flex-wrap">

   323→                )}>
   324→                  {Icon && <Icon size={14} />}
   325→                  {t.label}
   326→                </button>
   327→              );
   328→            })}
   329→          </div>
   330→
   331→          {loading ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : tab === "dashboard" ? (
   332→            <FinancialDashboard
   333→              totalRevenue={totalRevenue}
   334→              totalExpenses={totalExpenses}
   335→              netProfit={netProfit}
   336→              totalAssets={totalAssets}
   337→              totalLiabilities={totalLiabilities}
   338→              totalAR={totalAR}
   339→              totalAP={totalAP}
   340→              totalCash={totalCash}
   341→              currentPeriod={currentPeriod}
   342→              entriesCount={entries.length}
   343→              accountsCount={accounts.length}
   344→            />
   345→          ) : tab === "trial" ? (
   346→            <TrialBalanceTable data={trial} loading={loading} />
   347→          ) : tab === "statements" ? (
   348→            <FinancialStatements company={activeCompany} />
   349→          ) : tab === "fiscal-periods" ? (
   350→            showForm ? <FiscalPeriodForm company={activeCompany} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadFiscalPeriods(); }} /> : (
   351→              <FiscalPeriodsTable periods={fiscalPeriods} company={activeCompany} onRefresh={loadFiscalPeriods} />
   352→            )
   353→          ) : tab === "cost-centers" ? (
   354→            showForm ? <CostCenterForm company={activeCompany} costCenters={costCenters} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadCostCenters(); }} /> : (
   355→              <CostCentersTable costCenters={costCenters} />
   356→            )
   357→          ) : tab === "aging" ? (
   358→            <AgingReport data={agingData} totalAR={totalAR} totalAP={totalAP} />

   361→              <BankAccountsList accounts={bankAccounts} company={activeCompany} onRefresh={loadBankAccounts} />
   362→            )
   363→          ) : showForm ? (
   364→            tab === "accounts" ? (
   365→              <AccountForm company={activeCompany} accounts={accounts} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); load(); }} />
   366→            ) : (
   367→              <JournalForm company={activeCompany} accounts={accounts} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); load(); }} />
   368→            )
   369→          ) : (
   370→            <>
   371→              {selectedIds.size > 0 && (
   372→                <div className="py-2.5 px-4 bg-destructive text-white rounded-[10px] flex flex-wrap justify-between items-center gap-2">
   373→                  <span className="font-bold text-[13px]">{selectedIds.size} عنصر محدد</span>
   374→                  <div className="flex gap-2">
   375→                    <button onClick={() => setSelectedIds(new Set())} disabled={bulkDeleting} className="bg-white/15 text-white border-none rounded-md py-1.5 px-3.5 text-[12px] font-bold cursor-pointer disabled:cursor-not-allowed">إلغاء التحديد</button>
   376→                    <button onClick={handleBulkDelete} disabled={bulkDeleting} className="bg-white/25 text-white border-none rounded-md py-1.5 px-3.5 text-[12px] font-bold cursor-pointer disabled:cursor-not-allowed disabled:opacity-70">{bulkDeleting ? "جارٍ الحذف…" : "حذف المحدد"}</button>
   377→                  </div>
   378→                </div>
   379→              )}
   380→
   381→              <div className="bg-card rounded-[14px] border border-border overflow-hidden">
   382→                {allItems.length === 0 ? <Empty label={tab === "accounts" ? "حسابات" : "قيود يومية"} /> : (
   383→                  <>
   384→                    <div className="overflow-x-auto garfix-scroll">

  1084→  const [asOf, setAsOf] = useState(today);

  1086→  const profitLossQuery = useProfitLoss(company.slug, from, to);

  1088→  const cashFlowQuery = useCashFlow(company.slug, from, to);

  1090→  const loading = statementType === "profit-loss" ? profitLossQuery.isLoading
  1091→    : statementType === "balance-sheet" ? balanceSheetQuery.isLoading
  1092→    : cashFlowQuery.isLoading;
  1093→  const queryError = statementType === "profit-loss" ? profitLossQuery.error
  1094→    : statementType === "balance-sheet" ? balanceSheetQuery.error
  1095→    : cashFlowQuery.error;
  1096→  const data: ProfitLossData | BalanceSheetData | CashFlowData | null =
  1097→    statementType === "profit-loss" ? (profitLossQuery.data as unknown as ProfitLossData | null ?? null)

  1104→    else cashFlowQuery.refetch();

  1107→  const statementTabs: Array<{ key: StatementType; label: string; icon: React.ComponentType<{ size?: number }> }> = [

  1109→    { key: "balance-sheet", label: "الميزانية العمومية", icon: Scale },

  1113→  const exportCsv = () => {

  1120→      rows = [["البند", "الكود", "النوع", "المبلغ"], ...pl.accounts.map((a) => [a.nameAr, a.code, a.type, String(a.amount)]), [], ["إجمالي الإيرادات", "", "", String(pl.revenue.total)], ["صافي الربح", "", "", String(pl.netProfit)], ["هامش الربح", "", "", pl.margin]];

  1124→      rows = [["القسم", "الكود", "الحساب", "الرصيد"]];

  1127→      bs.liabilities.accounts.forEach((a) => rows.push(["الخصوم", a.code, a.nameAr, String(a.balance)]));
  1128→      rows.push(["إجمالي الخصوم", "", "", String(bs.liabilities.total)]);
  1129→      bs.equity.accounts.forEach((a) => rows.push(["حقوق الملكية", a.code, a.nameAr, String(a.balance)]));
  1130→      rows.push(["إجمالي حقوق الملكية", "", "", String(bs.equity.total)]);

  1162→            <div className="flex items-center gap-1.5"><label className="text-[11px] font-bold text-muted-foreground">إلى</label><input type="date" value={to} onChange={(e) => setTo(e.target.value)} className={cn(inputStyle, "w-auto")} dir="ltr" /></div>

  1353→function JournalForm({ company, accounts, onClose, onSaved }: { company: { slug: string }; accounts: Account[]; onClose: () => void; onSaved: () => void }) {
  1354→  const [date, setDate] = useState(new Date().toISOString().slice(0, 10));
  1355→  const [description, setDescription] = useState("");
  1356→  const [reference, setReference] = useState("");
  1357→  const [status, setStatus] = useState("draft");
  1358→  const [lines, setLines] = useState<Array<{ accountId: number | null; debit: number; credit: number; description?: string }>>([{ accountId: null, debit: 0, credit: 0 }]);
  1359→  const [saving, setSaving] = useState(false);
  1360→  const createJournalEntryMutation = useCreateJournalEntry();
  1361→
  1362→  const totalDebit = lines.reduce((s, l) => s + Number(l.debit || 0), 0);
  1363→  const totalCredit = lines.reduce((s, l) => s + Number(l.credit || 0), 0);
  1364→  const isBalanced = Math.abs(totalDebit - totalCredit) < 0.001;
  1365→
  1366→  const updateLine = (i: number, field: string, value: number | string) => { setLines((arr) => arr.map((l, idx) => idx === i ? { ...l, [field]: value } : l)); };
  1367→  const addLine = () => setLines((arr) => [...arr, { accountId: null, debit: 0, credit: 0 }]);
  1368→  const removeLine = (i: number) => setLines((arr) => arr.filter((_, idx) => idx !== i));
  1369→
  1370→  const submit = async () => {
  1371→    if (lines.length === 0) { toast.error("أضف بنداً واحداً على الأقل"); return; }
  1372→    if (!isBalanced) { toast.error("القيد غير متوازن — المدين ≠ الدائن"); return; }
  1373→    if (lines.some((l) => !l.accountId)) { toast.error("كل بند يجب أن يحدد حساباً"); return; }
  1374→    setSaving(true);
  1375→    try {
  1376→      await createJournalEntryMutation.mutateAsync({
  1377→        date, description, reference, status, companySlug: company.slug,
  1378→        lines: lines.filter((l) => l.accountId !== null).map((l) => ({ accountId: l.accountId!, debit: Number(l.debit || 0), credit: Number(l.credit || 0), description: l.description })),
  1379→      });
  1380→      toast.success("تم إنشاء القيد"); onSaved();
  1381→    } catch (err) { toast.error(err instanceof Error ? err.message : "خطأ"); }
  1382→    finally { setSaving(false); }
  1383→  };
  1384→
  1385→  return (
  1386→    <div className="flex flex-col gap-4">
  1387→      <div className="bg-card rounded-[14px] border border-border p-5 flex flex-col gap-3.5">
  1388→        <h3 className="text-[15px] font-bold">قيد يومية جديد</h3>
  1389→        <div className="grid grid-cols-[repeat(auto-fit,minmax(160px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(200px,1fr))] gap-3">
  1390→          <div><label className={labelStyle}>التاريخ</label><input type="date" value={date} onChange={(e) => setDate(e.target.value)} className={inputStyle} dir="ltr" /></div>
  1391→          <div><label className={labelStyle}>المرجع</label><input value={reference} onChange={(e) => setReference(e.target.value)} className={inputStyle} dir="ltr" /></div>
  1392→          <div><label className={labelStyle}>الحالة</label>
  1393→            <select value={status} onChange={(e) => setStatus(e.target.value)} className={inputStyle}><option value="draft">مسودة</option><option value="posted">مُرحّل</option></select>
  1394→          </div>
  1395→        </div>
  1396→        <div><label className={labelStyle}>الوصف</label><input value={description} onChange={(e) => setDescription(e.target.value)} className={inputStyle} /></div>
  1397→        <div>
  1398→          <div className="flex justify-between items-center mb-2">
  1399→            <label className={cn(labelStyle, "mb-0")}>البنود</label>
  1400→            <button onClick={addLine} className="bg-accent text-accent-foreground border border-border rounded-[6px] py-1 px-2.5 text-[11px] font-bold cursor-pointer inline-flex items-center gap-1"><Plus size={12} /> إضافة</button>
  1401→          </div>
  1402→          <div className="flex flex-col gap-2">
  1403→            {lines.map((l, i) => (
  1404→              <div key={i} className="grid grid-cols-[1fr_80px_100px_28px] sm:grid-cols-[1fr_100px_100px_32px] gap-1 sm:gap-2 items-center">
  1405→                <select value={l.accountId ?? ""} onChange={(e) => updateLine(i, "accountId", Number(e.target.value))} className={inputStyle}><option value="">— اختر حساب —</option>{accounts.map((a) => <option key={a.id} value={a.id}>{a.code} — {a.nameAr}</option>)}</select>
  1406→                <input type="number" placeholder="مدين" value={l.debit} onChange={(e) => updateLine(i, "debit", Number(e.target.value))} className={inputStyle} dir="ltr" />
  1407→                <input type="number" placeholder="دائن" value={l.credit} onChange={(e) => updateLine(i, "credit", Number(e.target.value))} className={inputStyle} dir="ltr" />
  1408→                <button onClick={() => removeLine(i)} className="bg-transparent border border-border text-destructive rounded-[6px] cursor-pointer flex items-center justify-center h-8"><X size={12} /></button>
  1409→              </div>
  1410→            ))}
  1411→          </div>
  1412→          <div className={cn("flex justify-between mt-3 py-2 px-3 rounded-sm text-[12px] font-bold", isBalanced ? "bg-emerald-500/10" : "bg-red-500/10")}>
  1413→            <span>مدين: <span className="[direction:ltr]">{totalDebit.toLocaleString("ar-EG")}</span></span>
  1414→            <span>دائن: <span className="[direction:ltr]">{totalCredit.toLocaleString("ar-EG")}</span></span>
  1415→            <span className={cn(isBalanced ? "text-emerald-500" : "text-red-500")}>{isBalanced ? "متوازن ✓" : "غير متوازن ✗"}</span>
  1416→          </div>
  1417→        </div>
  1418→      </div>
  1419→      <div className="flex gap-2.5 justify-end">
  1420→        <button onClick={onClose} className="py-2.5 px-5 rounded-md bg-transparent text-muted-foreground border border-border text-[13px] font-bold cursor-pointer">إلغاء</button>
  1421→        <button onClick={submit} disabled={saving || !isBalanced} className={cn("py-2.5 px-6 rounded-md bg-primary text-primary-foreground border-none text-[13px] font-extrabold", saving ? "cursor-not-allowed" : "cursor-pointer", (saving || !isBalanced) && "opacity-70")}>{saving ? "جارٍ…" : "حفظ"}</button>
  1422→      </div>
  1423→    </div>
  1424→  );
  1425→}
