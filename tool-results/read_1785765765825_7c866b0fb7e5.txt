   144→// ── Main Component ──────────────────────────────────────────
   145→
   146→export default function AISettingsPage() {
   147→  // State
   148→  const [config, setConfig] = useState<AIConfig | null>(null);
   149→  const [usageData, setUsageData] = useState<UsageData | null>(null);
   150→  const [isLoading, setIsLoading] = useState(true);
   151→  const [isSaving, setIsSaving] = useState(false);
   152→  const [isTesting, setIsTesting] = useState(false);
   153→  
   154→  // Form state
   155→  const [apiKey, setApiKey] = useState('');
   156→  const [showApiKey, setShowApiKey] = useState(false);
   157→  const [selectedModel, setSelectedModel] = useState('gemini-2.0-flash');
   158→  const [systemPrompt, setSystemPrompt] = useState('');
   159→  const [temperature, setTemperature] = useState(0.7);
   160→  
   161→  // UI state
   162→  const [activeTab, setActiveTab] = useState('config');
   163→  const [testResult, setTestResult] = useState<TestResult | null>(null);
   164→  const [showTestModal, setShowTestModal] = useState(false);
   165→  const [alert, setAlert] = useState<{ type: 'success' | 'error'; message: string } | null>(null);
   166→  
   167→  // Features toggles
   168→  const [features, setFeatures] = useState({
   169→    chat: true,
   170→    smartParse: true,
   171→    invoiceExtraction: true,
   172→    memory: true,
   173→  });
   174→  
   175→  // Fetch config on mount
   176→  useEffect(() => {
   177→    fetchConfig();
   178→    fetchUsage();
   179→  }, []);
   180→  
   181→  // ── API Calls ─────────────────────────────────────────────
   182→  
   183→  const fetchConfig = async () => {
   184→    try {
   185→      setIsLoading(true);
   186→      const response = await fetch('/api/founder-panel/ai-config');
   187→      const data = await response.json();
   188→      
   189→      if (data.success) {
   190→        setConfig(data.data);
   191→        
   192→        // Update form state
   193→        if (data.data.primaryProvider) {
   194→          setSelectedModel(data.data.primaryProvider.model || 'gemini-2.0-flash');
   195→          setTemperature(data.data.primaryProvider.temperature || 0.7);
   196→        }
   197→        if (data.data.systemPrompt) {
   198→          setSystemPrompt(data.data.systemPrompt);
   199→        }
   200→        if (data.data.features) {
   201→          setFeatures(data.data.features);
   202→        }
   203→      }
   204→    } catch (error) {
   205→      console.error('Failed to fetch AI config:', error);
   206→      setAlert({ type: 'error', message: 'فشل في تحميل الإعدادات' });
   207→    } finally {
   208→      setIsLoading(false);
   209→    }
   210→  };
   211→  
   212→  const fetchUsage = async () => {
   213→    try {
   214→      const response = await fetch('/api/founder-panel/ai-config/usage?days=30');
   215→      const data = await response.json();
   216→      
   217→      if (data.success) {
   218→        setUsageData(data.data);
   219→      }
   220→    } catch (error) {
   221→      console.error('Failed to fetch usage:', error);
   222→    }
   223→  };
   224→  
   225→  const saveConfig = async () => {
   226→    try {
   227→      setIsSaving(true);
   228→      
   229→      const payload = {
   230→        primaryProvider: {
   231→          provider: 'google-gemini',
   232→          apiKey: apiKey || config?.primaryProvider?.apiKey?.replace(/•/g, '') || '',
   233→          model: selectedModel,
   234→          maxTokens: 4096,
   235→          temperature,
   236→          enabled: true,
   237→          rateLimitRpm: 60,
   238→          monthlyTokenQuota: 1000000,
   239→        },
   240→        systemPrompt,
   241→        ...features,
   242→        costOptimization: 'balanced',
   243→        notifyHighUsage: true,
   244→        usageNotificationThreshold: 80,
   245→      };
   246→      
   247→      const response = await fetch('/api/founder-panel/ai-config', {
   248→        method: 'PUT',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify(payload),
   251→      });
   252→      
   253→      const data = await response.json();
   254→      
   255→      if (data.success) {
   256→        setAlert({ type: 'success', message: 'تم حفظ الإعدادات بنجاح ✅' });
   257→        fetchConfig();
   258→      } else {
   259→        setAlert({ type: 'error', message: data.error || 'فشل في الحفظ' });
   260→      }
   261→    } catch (error) {
   262→      console.error('Failed to save config:', error);
   263→      setAlert({ type: 'error', message: 'خطأ في الاتصال بالخادم' });
   264→    } finally {
   265→      setIsSaving(false);
   266→      
   267→      // Clear alert after 3 seconds
   268→      setTimeout(() => setAlert(null), 3000);
   269→    }
   270→  };
   271→  
   272→  const testConnection = async () => {
   273→    try {
   274→      setIsTesting(true);
   275→      setTestResult(null);
   276→      
   277→      const keyToTest = apiKey || config?.primaryProvider?.apiKey?.replace(/•/g, '') || '';
   278→      
   279→      if (!keyToTest && !apiKey) {
   280→        setTestResult({
   281→          success: false,
   282→          latencyMs: 0,
   283→          model: selectedModel,
   284→          error: 'يرجى إدخال مفتاح API أولاً',
   285→        });
   286→        setShowTestModal(true);
   287→        return;
   288→      }
   289→      
   290→      const response = await fetch('/api/founder-panel/ai-config/test', {
   291→        method: 'POST',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({
   294→          provider: 'google-gemini',
   295→          apiKey: keyToTest,
   296→          model: selectedModel,
   297→        }),
   298→      });
   299→      
   300→      const data = await response.json();
   301→      
   302→      if (data.success) {
   303→        setTestResult(data.data);
   304→      } else {
   305→        setTestResult({
   306→          success: false,
   307→          latencyMs: 0,
   308→          model: selectedModel,
   309→          error: data.error || 'فشل الاختبار',
   310→        });
   311→      }
   312→      
   313→      setShowTestModal(true);
   314→    } catch (error) {
   315→      setTestResult({
   316→        success: false,
   317→        latencyMs: 0,
   318→        model: selectedModel,
   319→        error: 'خطأ في الاتصال',
   320→      });
   321→      setShowTestModal(true);
   322→    } finally {
   323→      setIsTesting(false);
   324→    }
   325→  };
   326→  
   327→  // ── Render Helpers ───────────────────────────────────────
   328→  
   329→  const renderLoading = () => (
   330→    <div className="space-y-6">
   331→      <GarfixSkeleton variant="card" className="h-48" />
   332→      <GarfixGrid cols={2}>
   333→        <GarfixSkeleton variant="card" className="h-64" />
   334→        <GarfixSkeleton variant="card" className="h-64" />
   335→      </GarfixGrid>
   336→    </div>
   337→  );
   338→  
   339→  const renderAlert = () => {
   340→    if (!alert) return null;
   341→    
   342→    return (
   343→      <GarfixAlert
   344→        variant={alert.type === 'success' ? 'success' : 'error'}
   345→        dismissible
   346→        onClose={() => setAlert(null)}
   347→        className="mb-4"
   348→      >
   349→        {alert.message}
   350→      </GarfixAlert>
   351→    );
   352→  };
   353→  
   354→  // ── Main Render ───────────────────────────────────────────
   355→  
   356→  if (isLoading) {
   357→    return (
   358→      <GarfixPageTransition>
   359→        <GarfixContainer padding="lg">
   360→          {renderLoading()}
   361→        </GarfixContainer>
   362→      </GarfixPageTransition>
   363→    );
   364→  }
   365→  
   366→  return (
   367→    <GarfixPageTransition enterAnimation="slideUp">
   368→      <div className="min-h-screen bg-gray-50 dark:bg-gray-950 py-8" dir="rtl" lang="ar">
   369→        <GarfixContainer>
   370→          {/* Header */}
   371→          <GarfixPageHeader
   372→            title="إعدادات الذكاء الاصطناعي"
   373→            breadcrumbs={[
   374→              { label: 'لوحة الفاوندر', href: '/founder-panel' },
   375→              { label: 'الإعدادات' },
   376→              { label: 'الذكاء الاصطناعي' },
   377→            ]}
   378→            actions={
   379→              <GarfixButton
   380→                onClick={saveConfig}
   381→                isLoading={isSaving}
   382→                icon={<span>💾</span>}
   383→              >
   384→                حفظ الإعدادات
   385→              </GarfixButton>
   386→            }
   387→          />
   388→          
   389→          {/* Alert */}
   390→          {renderAlert()}
   391→          
   392→          {/* Tabs */}
   393→          <GarfixTabs
   394→            defaultValue="config"
   395→            value={activeTab}
   396→            onValueChange={setActiveTab}
   397→            className="mt-6"
   398→          >
   399→            <div className="flex gap-1 p-1 bg-white dark:bg-gray-900 rounded-xl shadow-sm w-fit border border-gray-200 dark:border-gray-800">
   400→              {[
   401→                { value: 'config', label: '⚙️ الإعدادات الأساسية' },
   402→                { value: 'features', label: '🔧 الميزات' },
   403→                { value: 'usage', label: '📊 الاستخدام' },
   404→                { value: 'test', label: '🧪 الاختبار' },
   405→              ].map(tab => (
   406→                <button
   407→                  key={tab.value}
   408→                  data-value={tab.value}
   409→                  onClick={() => setActiveTab(tab.value)}
   410→                  className={cn(
   411→                    'px-4 py-2 rounded-lg text-sm font-medium transition-all',
   412→                    activeTab === tab.value
   413→                      ? 'bg-emerald-500 text-white shadow-sm'
   414→                      : 'text-gray-600 dark:text-gray-400 hover:text-gray-900 dark:hover:text-white'
   415→                  )}
   416→                >
   417→                  {tab.label}
   418→                </button>
   419→              ))}
   420→            </div>
   421→            
   422→            {/* ── Config Tab ──────────────────────────────── */}
   423→            <GarfixTabPanel value="config" className="mt-6">
   424→              <GarfixGrid cols={2} gap="lg">
   425→                {/* API Key Configuration */}
   426→                <FadeUp delay={0}>
   427→                  <MotionCard className="p-6">
   428→                    <h3 className="text-lg font-semibold text-gray-900 dark:text-white mb-4 flex items-center gap-2">
   429→                      <span>🔑</span>
   430→                      مفتاح Google Gemini API
   431→                    </h3>
   432→                    
   433→                    <div className="space-y-4">
   434→                      <div>
   435→                        <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
   436→                          مفتاح API
   437→                        </label>
   438→                        <div className="relative">
   439→                          <GarfixInput
   440→                            type={showApiKey ? 'text' : 'password'}
   441→                            value={apiKey}
   442→                            onChange={(e) => setApiKey(e.target.value)}
   443→                            placeholder={
   444→                              config?.primaryProvider?.hasApiKey
   445→                                ? '•••••••• (موجد بالفعل)'
   446→                                : 'أدخل مفتاح Google AI الجديد'
   447→                            }
   448→                            className="font-mono text-sm"
   449→                          />
   450→                          <button
   451→                            type="button"
   452→                            onClick={() => setShowApiKey(!showApiKey)}
   453→                            className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600"
   454→                          >
   455→                            {showApiKey ? '🙈' : '👁️'}
   456→                          </button>
   457→                        </div>
   458→                        <p className="text-xs text-gray-500 mt-1">
   459→                          المفتاح الحالي: {config?.primaryProvider?.hasApiKey ? '✅ مفعل' : '❌ غير موجود'}
   460→                        </p>
   461→                      </div>
   462→

   465→                          النموذج (Model)
   466→                        </label>
   467→                        <select

   469→                          onChange={(e) => setSelectedModel(e.target.value)}
   470→                          className="w-full px-3 py-2 rounded-lg border border-gray-300 dark:border-gray-600 bg-white dark:bg-gray-800 text-gray-900 dark:text-white"

   476→                        </select>
   477→                      </div>
   478→                      
   479→                      <div>
   480→                        <label className="block text-sm font-medium text-gray-700 dark:text-gray-300 mb-1">
   481→                          درجة الحرارة (Temperature): {temperature}

   483→                        <input

   485→                          min="0"
   486→                          max="2"

   488→                          value={temperature}
   489→                          onChange={(e) => setTemperature(parseFloat(e.target.value))}
   490→                          className="w-full accent-emerald-500"
   491→                        />
   492→                        <div className="flex justify-between text-xs text-gray-500 mt-1">
   493→                          <span>دقيق (0)</span>

   495→                        </div>

   499→                        <GarfixButton

   501→                          size="sm"
   502→                          onClick={testConnection}
   503→                          isLoading={isTesting}
   504→                          icon={<span>🧪</span>}

   512→                </FadeUp>
   513→                
   514→                {/* System Prompt */}
   515→                <FadeUp delay={80}>
   516→                  <MotionCard className="p-6">
   517→                    <h3 className="text-lg font-semibold text-gray-900 dark:text-white mb-4 flex items-center gap-2">

   543→                          {[

   547→                          ].map(template => (

   551→                              className="px-3 py-1.5 text-xs rounded-full bg-gray-100 dark:bg-gray-800 text-gray-700 dark:text-gray-300 hover:bg-emerald-100 dark:hover:bg-emerald-900 transition-colors"
