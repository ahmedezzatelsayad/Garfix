    88→export function AccountingView() {
    89→  const { activeCompany } = useBrand();
    90→  const [moduleTab, setModuleTab] = useState<ModuleTab>("core");
    91→
    92→  /* ─── Cursor pagination hooks (TanStack Query infinite scroll) ──────── */
    93→  const accountsCursor = useAccountsCursor(activeCompany?.slug || "", { limit: 50 });
    94→  const journalCursor = useJournalEntriesCursor(activeCompany?.slug || "", { limit: 20 });
    95→
    96→  /* ─── Core sub-module state (preserved exactly as-is) ────────────────── */
    97→  const [tab, setTab] = useState<Tab>("accounts");
    98→  const [trial, setTrial] = useState<{ accounts: TrialRow[]; grandDebit: number; grandCredit: number; isBalanced: boolean } | null>(null);
    99→  const [fiscalPeriods, setFiscalPeriods] = useState<FiscalPeriod[]>([]);
   100→  const [costCenters, setCostCenters] = useState<CostCenter[]>([]);
   101→  const [agingData, setAgingData] = useState<AgingBucket[]>([]);
   102→  const [bankAccounts, setBankAccounts] = useState<BankAccount[]>([]);
   103→  const [loading, setLoading] = useState(true);
   104→  const [showForm, setShowForm] = useState(false);
   105→  const [selectedIds, setSelectedIds] = useState<Set<number>>(new Set());
   106→  const [currentPage, setCurrentPage] = useState(1);
   107→  const [bulkDeleting, setBulkDeleting] = useState(false);
   108→  const [reversingId, setReversingId] = useState<number | null>(null);
   109→  const [reverseConfirm, setReverseConfirm] = useState<JournalEntry | null>(null);
   110→
   111→  const load = useCallback(async () => {
   112→    if (!activeCompany) { setLoading(false); return; }
   113→    setLoading(true);
   114→    // Accounts and journal entries are now fetched via cursor pagination hooks
   115→    // (useAccountsCursor / useJournalEntriesCursor) — no manual fetch needed.
   116→    // We still load trial balance, fiscal periods, etc. manually.
   117→    try {
   118→      await Promise.all([
   119→        accountsCursor.refetch(),
   120→        journalCursor.refetch(),
   121→      ]);
   122→    } catch (err) {
   123→      toast.error(err instanceof Error ? err.message : "تعذّر تحميل بيانات المحاسبة");
   124→    } finally { setLoading(false); }
   125→  }, [activeCompany]);
   126→
   127→  const loadTrial = useCallback(async () => {
   128→    if (!activeCompany) return;
   129→    try {
   130→      const res = await authedFetch(`/api/accounting/trial-balance?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   131→      if (!res.ok) {
   132→        const e = await res.json().catch(() => ({}));
   133→        toast.error((e as Record<string, unknown>)?.error as string || "تعذّر تحميل ميزان المراجعة");
   134→        setTrial(null);
   135→        return;
   136→      }
   137→      setTrial(await res.json());
   138→    } catch (err) {
   139→      toast.error(err instanceof Error ? err.message : "تعذّر تحميل ميزان المراجعة");
   140→      setTrial(null);
   141→    }
   142→  }, [activeCompany]);
   143→
   144→  const loadFiscalPeriods = useCallback(async () => {
   145→    if (!activeCompany) return;
   146→    try {
   147→      const res = await authedFetch(`/api/accounting/fiscal-periods?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   148→      if (res.ok) { const d = await res.json(); setFiscalPeriods(d.periods || []); }
   149→      else { setFiscalPeriods([]); }
   150→    } catch { setFiscalPeriods([]); }
   151→  }, [activeCompany]);
   152→
   153→  const loadCostCenters = useCallback(async () => {
   154→    if (!activeCompany) return;
   155→    try {
   156→      const res = await authedFetch(`/api/accounting/cost-centers?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   157→      if (res.ok) { const d = await res.json(); setCostCenters(d.costCenters || []); }
   158→      else { setCostCenters([]); }
   159→    } catch { setCostCenters([]); }
   160→  }, [activeCompany]);
   161→
   162→  const loadAging = useCallback(async () => {
   163→    if (!activeCompany) return;
   164→    try {
   165→      const res = await authedFetch(`/api/accounting/aging?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   166→      if (res.ok) { const d = await res.json(); setAgingData(d.buckets || []); }
   167→      else { setAgingData([]); }
   168→    } catch { setAgingData([]); }
   169→  }, [activeCompany]);
   170→
   171→  const loadBankAccounts = useCallback(async () => {
   172→    if (!activeCompany) return;
   173→    try {
   174→      const res = await authedFetch(`/api/accounting/bank-accounts?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   175→      if (res.ok) { const d = await res.json(); setBankAccounts(d.accounts || []); }
   176→      else { setBankAccounts([]); }
   177→    } catch { setBankAccounts([]); }
   178→  }, [activeCompany]);
   179→
   180→  useEffect(() => { load(); }, [load]);
   181→  useEffect(() => {
   182→    if (tab === "trial" && activeCompany) loadTrial();
   183→    if (tab === "fiscal-periods" && activeCompany) loadFiscalPeriods();
   184→    if (tab === "cost-centers" && activeCompany) loadCostCenters();
   185→    if (tab === "aging" && activeCompany) loadAging();
   186→    if (tab === "banking" && activeCompany) loadBankAccounts();
   187→    if (tab === "dashboard" && activeCompany) { load(); loadAging(); loadBankAccounts(); }
   188→  }, [tab, activeCompany, loadTrial, loadFiscalPeriods, loadCostCenters, loadAging, loadBankAccounts, load]);
   189→
   190→  const switchTab = (t: Tab) => {
   191→    setTab(t);
   192→    setShowForm(false);
   193→    setSelectedIds(new Set());
   194→    setCurrentPage(1);
   195→  };
   196→
   197→  // Derived data from cursor pagination hooks (replaces manual state)
   198→  const accounts = accountsCursor.items as Account[];
   199→  const entries = journalCursor.items as JournalEntry[];
   200→  const accountsHasNextPage = accountsCursor.hasNextPage;
   201→  const accountsFetchingNext = accountsCursor.isFetchingNextPage;
   202→  const entriesHasNextPage = journalCursor.hasNextPage;
   203→  const entriesFetchingNext = journalCursor.isFetchingNextPage;
   204→
   205→  const itemsForTab = (): Array<{ id: number }> => (tab === "accounts" ? accounts : entries);
   206→
   207→  const allItems = itemsForTab();
   208→  const totalPages = Math.max(1, Math.ceil(allItems.length / PAGE_SIZE));
   209→  const pageItems = allItems.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE);
   210→  const safePage = Math.min(currentPage, totalPages);
   211→
   212→  const toggleSelectAll = () => {
   213→    if (selectedIds.size === pageItems.length && pageItems.length > 0) setSelectedIds(new Set());
   214→    else setSelectedIds(new Set(pageItems.map((i) => i.id)));
   215→  };
   216→  const toggleRow = (id: number) => {
   217→    setSelectedIds((prev) => {
   218→      const next = new Set(prev);
   219→      if (next.has(id)) next.delete(id);
   220→      else next.add(id);
   221→      return next;
   222→    });
   223→  };
   224→  const handleBulkDelete = async () => {
   225→    if (selectedIds.size === 0) return;
   226→    if (!confirm(`حذف ${selectedIds.size} عنصر؟`)) return;
   227→    const endpoint = tab === "accounts" ? "/api/accounting/accounts" : "/api/accounting/journal-entries";
   228→    setBulkDeleting(true);
   229→    let okCount = 0, failCount = 0;
   230→    for (const id of selectedIds) {
   231→      try {
   232→        const res = await authedFetch(`${endpoint}/${id}`, { method: "DELETE" });
   233→        if (res.ok) okCount++;
   234→        else { failCount++; }
   235→      } catch { failCount++; }
   236→    }
   237→    setBulkDeleting(false);
   238→    setSelectedIds(new Set());
   239→    if (okCount > 0) toast.success(`تم حذف ${okCount} عنصر`);
   240→    if (failCount > 0) toast.error(`تعذّر حذف ${failCount} عنصر`);
   241→    load();
   242→  };
   243→  const handleDelete = async (id: number) => {
   244→    if (!confirm("حذف هذا العنصر؟")) return;
   245→    const endpoint = tab === "accounts" ? "/api/accounting/accounts" : "/api/accounting/journal-entries";
   246→    const res = await authedFetch(`${endpoint}/${id}`, { method: "DELETE" });
   247→    if (res.ok) { toast.success("تم الحذف"); load(); }
   248→    else { const e = await res.json().catch(() => ({})); toast.error(e.error || "تعذّر الحذف"); }
   249→  };
   250→
   251→  const handleReverse = async (entry: JournalEntry) => {
   252→    if (!activeCompany) return;
   253→    setReverseConfirm(entry);
   254→  };
   255→
   256→  const confirmReverse = async () => {
   257→    if (!reverseConfirm || !activeCompany) return;
   258→    const entry = reverseConfirm;
   259→    setReversingId(entry.id);
   260→    try {
   261→      const url = `/api/accounting/journal-entries/${entry.id}/reverse?companySlug=${encodeURIComponent(activeCompany.slug)}`;
   262→      const res = await authedFetch(url, { method: "POST" });
   263→      if (!res.ok) {
   264→        const e = await res.json().catch(() => ({}));
   265→        throw new Error(e.error || "تعذّر العكس");
   266→      }
   267→      const data = await res.json().catch(() => ({}));
   268→      const reversalId = data?.reversal?.id;
   269→      toast.success(reversalId ? `تم إنشاء قيد عكسي #${reversalId}` : "تم عكس القيد");
   270→      setReverseConfirm(null);
   271→      await load();
   272→    } catch (err) {
   273→      toast.error(err instanceof Error ? err.message : "خطأ أثناء العكس");
   274→    } finally { setReversingId(null); }
   275→  };
   276→
   277→  const pageBtnStyle = (disabled: boolean): string =>
   278→    disabled
   279→      ? "py-1.5 px-3 rounded-md bg-transparent text-muted-foreground border border-border text-[12px] font-bold cursor-not-allowed opacity-50"
   280→      : "py-1.5 px-3 rounded-md bg-card text-foreground border border-border text-[12px] font-bold cursor-pointer";
   281→
   282→  if (!activeCompany) return <div className="p-8 md:p-12 text-center text-muted-foreground">اختر شركة</div>;
   283→
   284→  const ACCOUNT_TYPES: Record<string, { label: string; color: string; badge: string }> = {
   285→    asset: { label: "أصول", color: "#10b981", badge: "bg-emerald-500/15 text-emerald-500" },
   286→    liability: { label: "خصوم", color: "#ef4444", badge: "bg-red-500/15 text-red-500" },
   287→    equity: { label: "حقوق ملكية", color: "#7c3aed", badge: "bg-violet-500/15 text-violet-500" },
   288→    revenue: { label: "إيرادات", color: "#3b82f6", badge: "bg-blue-500/15 text-blue-500" },
   289→    expense: { label: "مصروفات", color: "#f59e0b", badge: "bg-amber-500/15 text-amber-500" },
   290→    contra_revenue: { label: "مقابل إيرادات", color: "#9ca3af", badge: "bg-gray-400/15 text-gray-400" },
   291→    contra_asset: { label: "مقابل أصول", color: "#9ca3af", badge: "bg-gray-400/15 text-gray-400" },
   292→  };
   293→
   294→  const tabs: Array<{ key: Tab; label: string; icon?: React.ComponentType<{ size?: number }> }> = [
   295→    { key: "dashboard", label: "لوحة التحكم", icon: LayoutDashboard },
   296→    { key: "accounts", label: `الحسابات (${accounts.length})` },
   297→    { key: "journal", label: `القيود (${entries.length})` },
   298→    { key: "trial", label: "ميزان المراجعة", icon: Scale },
   299→    { key: "statements", label: "القوائم المالية", icon: FileBarChart },
   300→    { key: "fiscal-periods", label: "الفترات المالية", icon: Clock },
   301→    { key: "cost-centers", label: "مراكز التكلفة", icon: Building },
   302→    { key: "aging", label: "تقادم الذمم", icon: BarChart3 },
   303→    { key: "banking", label: "البنوك", icon: Landmark },
   304→  ];
   305→
   306→  // Dashboard metrics
   307→  const totalRevenue = accounts.filter(a => a.type === "revenue").reduce((s, a) => s + a.balance, 0);
   308→  const totalExpenses = accounts.filter(a => a.type === "expense").reduce((s, a) => s + a.balance, 0);
   309→  const totalAssets = accounts.filter(a => a.type === "asset").reduce((s, a) => s + a.balance, 0);
   310→  const totalLiabilities = accounts.filter(a => a.type === "liability").reduce((s, a) => s + a.balance, 0);
   311→  const netProfit = totalRevenue - totalExpenses;
   312→  const totalAR = agingData.reduce((s, b) => s + b.receivable, 0);
   313→  const totalAP = agingData.reduce((s, b) => s + b.payable, 0);
   314→  const totalCash = bankAccounts.reduce((s, b) => s + b.balance, 0);
   315→  const currentPeriod = fiscalPeriods.find(p => p.status === "open");
   316→
   317→  /* ─── Render ──────────────────────────────────────────────────────────── */
   318→  return (
   319→    <div className="flex flex-col gap-4">
   320→      {/* Module-level header */}
   321→      <div className="flex flex-wrap justify-between items-center gap-3">
   322→        <div><h1 className="text-xl md:text-2xl font-extrabold flex items-center gap-2"><Calculator size={20} /> المحاسبة</h1><p className="text-[13px] text-muted-foreground">{activeCompany.nameAr || activeCompany.name}</p></div>
   323→      </div>
   324→
   325→      {/* Module-level navigation tabs (top row, visually distinct) */}
   326→      <div className="flex gap-1.5 flex-wrap bg-muted/50 rounded-[12px] p-2">
   327→        {MODULE_TABS.map((mt) => {
   328→          const Icon = mt.icon;
   329→          const active = moduleTab === mt.key;
   330→          return (
   331→            <button
   332→              key={mt.key}
   333→              onClick={() => setModuleTab(mt.key)}
   334→              className={cn(
   335→                "py-2.5 px-4 rounded-[10px] text-[13px] font-bold cursor-pointer inline-flex items-center gap-2 transition-colors",
   336→                active
   337→                  ? "bg-primary text-primary-foreground shadow-sm"
   338→                  : "bg-card text-muted-foreground border border-border hover:bg-accent hover:text-accent-foreground",
   339→              )}
   340→            >
   341→              <Icon size={16} />
   342→              {mt.label}
   343→            </button>
   344→          );
   345→        })}
   346→      </div>
   347→
   348→      {/* ─── Module content ─────────────────────────────────────────────── */}
   349→      {moduleTab === "core" ? (
   350→        /* ─── Core sub-module: exact original inner content ───────────── */
   351→        <>
   352→          <div className="flex flex-wrap justify-between items-center gap-3">
   353→            {(tab === "accounts" || tab === "journal" || tab === "fiscal-periods" || tab === "cost-centers" || tab === "banking") && (
   354→              <button onClick={() => setShowForm(true)} className="inline-flex items-center gap-1.5 py-2.5 px-[18px] rounded-[10px] bg-primary text-primary-foreground border-none text-[13px] font-bold cursor-pointer"><Plus size={16} /> إضافة</button>
   355→            )}
   356→          </div>
   357→          <div className="flex gap-1.5 flex-wrap">
   358→            {tabs.map((t) => {
   359→              const Icon = t.icon;
   360→              return (
   361→                <button key={t.key} onClick={() => switchTab(t.key)} className={cn(
   362→                  "py-2 px-4 rounded-[10px] border border-border text-[12px] font-bold cursor-pointer inline-flex items-center gap-1.5",
   363→                  tab === t.key ? "bg-primary text-primary-foreground" : "bg-card text-muted-foreground",
   364→                )}>
   365→                  {Icon && <Icon size={14} />}
   366→                  {t.label}
   367→                </button>
   368→              );
   369→            })}
   370→          </div>
   371→
   372→          {loading ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : tab === "dashboard" ? (
   373→            <FinancialDashboard
   374→              totalRevenue={totalRevenue}
   375→              totalExpenses={totalExpenses}
   376→              netProfit={netProfit}
   377→              totalAssets={totalAssets}
   378→              totalLiabilities={totalLiabilities}
   379→              totalAR={totalAR}
   380→              totalAP={totalAP}
   381→              totalCash={totalCash}
   382→              currentPeriod={currentPeriod}
   383→              entriesCount={entries.length}
   384→              accountsCount={accounts.length}
   385→            />
   386→          ) : tab === "trial" ? (
   387→            <TrialBalanceTable data={trial} loading={loading} />
   388→          ) : tab === "statements" ? (
   389→            <FinancialStatements company={activeCompany} />
   390→          ) : tab === "fiscal-periods" ? (
   391→            showForm ? <FiscalPeriodForm company={activeCompany} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadFiscalPeriods(); }} /> : (
   392→              <FiscalPeriodsTable periods={fiscalPeriods} company={activeCompany} onRefresh={loadFiscalPeriods} />

  1431→function JournalForm({ company, accounts, onClose, onSaved }: { company: { slug: string }; accounts: Account[]; onClose: () => void; onSaved: () => void }) {
  1432→  const [date, setDate] = useState(new Date().toISOString().slice(0, 10));
  1433→  const [description, setDescription] = useState("");
  1434→  const [reference, setReference] = useState("");
  1435→  const [status, setStatus] = useState("draft");
  1436→  const [lines, setLines] = useState<Array<{ accountId: number | null; debit: number; credit: number; description?: string }>>([{ accountId: null, debit: 0, credit: 0 }]);
  1437→  const [saving, setSaving] = useState(false);
  1438→
  1439→  const totalDebit = lines.reduce((s, l) => s + Number(l.debit || 0), 0);
  1440→  const totalCredit = lines.reduce((s, l) => s + Number(l.credit || 0), 0);
  1441→  const isBalanced = Math.abs(totalDebit - totalCredit) < 0.001;
  1442→
  1443→  const updateLine = (i: number, field: string, value: number | string) => { setLines((arr) => arr.map((l, idx) => idx === i ? { ...l, [field]: value } : l)); };
  1444→  const addLine = () => setLines((arr) => [...arr, { accountId: null, debit: 0, credit: 0 }]);
  1445→  const removeLine = (i: number) => setLines((arr) => arr.filter((_, idx) => idx !== i));
  1446→
  1447→  const submit = async () => {
  1448→    if (lines.length === 0) { toast.error("أضف بنداً واحداً على الأقل"); return; }
  1449→    if (!isBalanced) { toast.error("القيد غير متوازن — المدين ≠ الدائن"); return; }
  1450→    if (lines.some((l) => !l.accountId)) { toast.error("كل بند يجب أن يحدد حساباً"); return; }
  1451→    setSaving(true);
  1452→    try {
  1453→      const res = await authedFetch("/api/accounting/journal-entries", {
  1454→        method: "POST", headers: { "Content-Type": "application/json" },
  1455→        body: JSON.stringify({ date, description, reference, status, lines, companySlug: company.slug }),
  1456→      });
  1457→      if (!res.ok) { const e = await res.json().catch(() => ({})); throw new Error(e.error || "Failed"); }
  1458→      toast.success("تم إنشاء القيد"); onSaved();
  1459→    } catch (err) { toast.error(err instanceof Error ? err.message : "خطأ"); }
  1460→    finally { setSaving(false); }
  1461→  };
  1462→
  1463→  return (
  1464→    <div className="flex flex-col gap-4">
  1465→      <div className="bg-card rounded-[14px] border border-border p-5 flex flex-col gap-3.5">
  1466→        <h3 className="text-[15px] font-bold">قيد يومية جديد</h3>
  1467→        <div className="grid grid-cols-[repeat(auto-fit,minmax(160px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(200px,1fr))] gap-3">
  1468→          <div><label className={labelStyle}>التاريخ</label><input type="date" value={date} onChange={(e) => setDate(e.target.value)} className={inputStyle} dir="ltr" /></div>
  1469→          <div><label className={labelStyle}>المرجع</label><input value={reference} onChange={(e) => setReference(e.target.value)} className={inputStyle} dir="ltr" /></div>
  1470→          <div><label className={labelStyle}>الحالة</label>
  1471→            <select value={status} onChange={(e) => setStatus(e.target.value)} className={inputStyle}><option value="draft">مسودة</option><option value="posted">مُرحّل</option></select>
  1472→          </div>
