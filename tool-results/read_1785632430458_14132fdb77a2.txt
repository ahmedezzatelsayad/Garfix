    71→  generateSuppliers() {

    75→      'شركة البناء الحديث', 'المجموعة الذهبية', 'شركة التقنية المتقدمة',
    76→      'مؤسسة النجاح التجارية'
    77→    ];
    78→    
    79→    const englishNames = [
    80→      'Gulf Trading LLC', 'Arab Commerce Ltd', 'Future Tech Solutions',

   117→  generateInvoicesWithSplit() {

   130→        } else if (rand < this.trainRatio + this.validationRatio) {

   139→          tenantId: supplier.tenantId,

   160→  generateAmount(formatType) {
   161→    const ranges = {
   162→      arabic_tax: [500, 50000], english_commercial: [1000, 100000],
   163→      mixed_gulf: [200, 25000], ecommerce: [50, 5000],
   164→      service: [1000, 20000], restaurant: [20, 500],
   165→      medical: [200, 10000], legal: [5000, 50000],
   166→      tabular: [100, 30000], simplified_arabic: [100, 10000]
   167→    };
   168→    const [min, max] = ranges[formatType] || [100, 10000];
   169→    return Math.round((min + Math.random() * (max - min)) * 100) / 100;
   170→  }
   171→  
   172→  computeLayoutHash(formatType) {
   173→    let hash = 0;
   174→    for (let i = 0; i < formatType.length; i++) {
   175→      const char = formatType.charCodeAt(i);
   176→      hash = ((hash << 5) - hash) + char;
   177→      hash = hash & hash;
   178→    }
   179→    return `layout_${Math.abs(hash).toString(16).padStart(8, '0')}`;
   180→  }

   182→  applySimilarLayouts() {
   183→    const groups = Math.min(this.similarLayoutGroups, Math.floor(this.supplierCount / 10));
   184→    
   185→    for (let g = 0; g < groups; g++) {
   186→      const groupSize = 2 + Math.floor(Math.random() * 3);
   187→      const startIndex = g * 10;
   188→      const sharedHash = `shared_layout_${g.toString(16).padStart(4, '0')}`;
   189→      
   190→      for (let i = 0; i < groupSize && startIndex + i < this.suppliers.length; i++) {
   191→        const supplier = this.suppliers[startIndex + i];
   192→        
   193→        // Apply to ALL splits
   194→        for (const split of ['train', 'validation', 'holdout']) {
   195→          this.invoices[split]
   196→            .filter(inv => inv.supplierId === supplier.id)
   197→            .forEach(inv => {
   198→              inv.layoutHash = sharedHash;
   199→              inv.hasSharedLayout = true;
   200→              inv.layoutGroupId = g;
   201→            });
   202→        }
   203→      }
   204→    }
   205→    
   206→    let totalShared = 0;
   207→    for (const split of ['train', 'validation', 'holdout']) {
   208→      totalShared += this.invoices[split].filter(i => i.hasSharedLayout).length;
   209→    }
   210→    console.log(`   📋 Applied similar layouts: ${totalShared} invoices in ${groups} groups`);
   211→  }
   212→  
   213→  applyOcrErrors() {
   214→    const errorTypes = [
   215→      { type: 'char_substitution', field: 'supplierName' },
   216→      { type: 'char_omission', field: 'trn' },
   217→      { type: 'position_drift', field: 'fieldPositions' }
   218→    ];
   219→    
   220→    let errorCount = 0;
   221→    
   222→    for (const split of ['train', 'validation', 'holdout']) {
   223→      for (const invoice of this.invoices[split]) {
   224→        if (Math.random() > this.ocrErrorRate) continue;
   225→        
   226→        invoice.hasOcrError = true;
   227→        const errorType = errorTypes[Math.floor(Math.random() * errorTypes.length)];
   228→        
   229→        invoice.ocrErrors.push({
   230→          type: errorType.type,
   231→          field: errorType.field,
   232→          severity: Math.random() > 0.5 ? 'minor' : 'major'
   233→        });
   234→        errorCount++;
   235→      }
   236→    }
   237→    
   238→    console.log(`   🔤 Applied OCR errors: ${errorCount} invoices affected`);
   239→  }
   240→  
   241→  buildGroundTruth() {
   242→    // Build ground truth from ALL splits
   243→    for (const split of ['train', 'validation', 'holdout']) {
   244→      for (const invoice of this.invoices[split]) {
   245→        this.groundTruth.set(invoice.id, {
   246→          supplierId: invoice.supplierId,
   247→          tenantId: invoice.tenantId,
   248→          supplierName: invoice.expectedSupplierName,
   249→          trn: invoice.expectedTrn,
   250→          formatType: invoice.formatType,
   251→          isCorrectMatch: null // Will be filled after processing
   252→        });
   253→      }
   254→    }
   255→  }
   256→  
   257→  getStats() {
   258→    const totalInvoices = this.invoices.train.length + this.invoices.validation.length + this.invoices.holdout.length;
   259→    return {
   260→      totalSuppliers: this.suppliers.length,
   261→      totalInvoices: totalInvoices,
   262→      splits: {
   263→        train: this.invoices.train.length,
   264→        validation: this.invoices.validation.length,
   265→        holdout: this.invoices.holdout.length
   266→      },
   267→      ocrErrorRate: ((Array.from(this.groundTruth.values()).filter(gt => /* would need OCR flag */ false).length / totalInvoices) * 100).toFixed(1) + '%'
   268→    };
   269→  }

   273→// SECTION 2: CONFUSION MATRIX CALCULATOR
   274→// ============================================================
   275→
   276→class ConfusionMatrixCalculator {
   277→  constructor() {
   278→    this.reset();
   279→  }
   280→  
   281→  reset() {
   282→    this.tp = 0; // True Positive: Pattern matched correctly
   283→    this.fp = 0; // False Positive: Pattern matched WRONG supplier (FALSE MATCH!)
   284→    this.fn = 0; // False Negative: Should have matched but went to AI
   285→    this.tn = 0; // True Negative: Correctly sent to AI
   286→    
   287→    // Detailed tracking
   288→    this.predictions = [];
   289→  }
   290→  
   291→  /**
   292→   * Record a single prediction outcome
   293→   * @param {Object} prediction - { invoiceId, predictedSupplierId, actualSupplierId, confidence, outcome }
   294→   */
   295→  recordPrediction(prediction) {
   296→    const { predictedSupplierId, actualSupplierId, outcome, confidence } = prediction;
   297→    
   298→    this.predictions.push(prediction);
   299→    
   300→    // Classify into confusion matrix
   301→    if (outcome === 'PATTERN_MATCH') {
   302→      if (predictedSupplierId === actualSupplierId) {
   303→        this.tp++; // Correct pattern match
   304→      } else {
   305→        this.fp++; // FALSE MATCH - matched wrong pattern!
   306→      }
   307→    } else if (outcome === 'AI_FALLBACK') {
   308→      // Was this the right decision?
   309→      // We consider it FN if there WAS a correct pattern available but system chose AI
   310→      // We consider it TN if there was NO correct pattern or AI was genuinely better
   311→      this.fn++; // Conservative: count all fallbacks as missed opportunities
   312→    } else if (outcome === 'FALSE_MATCH') {
   313→      this.fp++; // Explicitly recorded as false match
   314→    }
   315→  }
   316→  
   317→  /**
   318→   * Calculate derived metrics
   319→   */
   320→  getMetrics() {
   321→    const total = this.tp + this.fp + this.fn + this.tn;
   322→    const actualPositives = this.tp + this.fn; // All that should have been matched
   323→    const predictedPositives = this.tp + this.fp; // All that were matched by pattern
   324→    
   325→    // Core metrics
   326→    const precision = predictedPositives > 0 ? this.tp / predictedPositives : 0;
   327→    const recall = actualPositives > 0 ? this.tp / actualPositives : 0;
   328→    const specificity = (this.tn + this.fp) > 0 ? this.tn / (this.tn + this.fp) : 0;
   329→    const f1Score = (precision + recall) > 0 ? (2 * precision * recall) / (precision + recall) : 0;
   330→    
   331→    // False Match Rate (the critical metric!)
   332→    const falseMatchRate = predictedPositives > 0 ? this.fp / predictedPositives : 0;
   333→    
   334→    // Accuracy
   335→    const accuracy = total > 0 ? (this.tp + this.tn) / total : 0;
   336→    
   337→    // Pattern Opportunity Rate (NEW METRIC)
   338→    const opportunityRate = actualPositives > 0 ? this.tp / actualPositives : 0;
   339→    
   340→    return {
   341→      // Raw counts
   342→      matrix: {
   343→        tp: this.tp,
   344→        fp: this.fp,   // FALSE MATCHES - should be minimized!
   345→        fn: this.fn,   // Missed opportunities
   346→        tn: this.tn    // Correct rejections
   347→      },
   348→      totalPredictions: total,
   349→      
   350→      // Primary metrics
   351→      precision: precision,           // Of all pattern matches, how many were correct?
   352→      recall: recall,                 // Of all possible matches, how many did we catch?
   353→      specificity: specificity,       // Of all negatives, how many did we reject?
   354→      f1Score: f1Score,               // Harmonic mean of precision and recall
   355→      accuracy: accuracy,             // Overall correctness
   356→      
   357→      // Critical business metric
   358→      falseMatchRate: falseMatchRate, // Of all pattern matches, how many were WRONG?
   359→      
   360→      // Opportunity metric
   361→      opportunityRate: opportunityRate, // Of all matchable invoices, what % did we match?
   362→      
   363→      // Summary string
   364→      summary: this.generateSummary(precision, recall, f1Score, falseMatchRate)
   365→    };
   366→  }
   367→  
   368→  generateSummary(precision, recall, f1Score, fmr) {
   369→    return {
   370→      headline: `FMR: ${(fmr * 100).toFixed(2)}% | F1: ${(f1Score * 100).toFixed(1)} | P: ${(precision * 100).toFixed(1)}% R: ${(recall * 100).toFixed(1)}%`,
   371→      assessment: this.getAssessment(fmr, f1Score),
   372→      recommendation: this.getRecommendation(fmr, precision, recall)
   373→    };
   374→  }
   375→  
   376→  getAssessment(fmr, f1Score) {
   377→    if (fmr < 0.005 && f1Score > 0.7) {
   378→      return 'EXCELLENT - Production Ready';
   379→    } else if (fmr < 0.01 && f1Score > 0.5) {
   380→      return 'GOOD - Near Production';
   381→    } else if (fmr < 0.05) {
   382→      return 'ACCEPTABLE - Needs Monitoring';
   383→    } else if (fmr < 0.10) {
   384→      return 'CONCERNING - High False Match Risk';
   385→    } else {
   386→      return 'CRITICAL - Not Production Safe';
   387→    }
   388→  }
   389→  
   390→  getRecommendation(fmr, precision, recall) {
   391→    if (fmr >= 0.05) {
   392→      return 'URGENT: Reduce False Matches before any production consideration';
   393→    } else if (precision < 0.7) {
   394→      return 'Improve precision - too many incorrect pattern matches';
   395→    } else if (recall < 0.5) {
   396→      return 'Consider improving recall - missing valid pattern opportunities';
   397→    } else {
   398→      return 'Metrics within acceptable range';
   399→    }
   400→  }
   401→  
   402→  /**
   403→   * Generate a visual confusion matrix table
   404→   */
   405→  getTable() {
   406→    return [
   407→      ['', 'Predicted: Pattern', 'Predicted: AI', 'Total'],
   408→      [`Actual: Matchable`, String(this.tp), String(this.fn), String(this.tp + this.fn)],
   409→      [`Actual: Not Matchable`, String(this.fp), String(this.tn), String(this.fp + this.tn)],
   410→      ['Total', String(this.tp + this.fp), String(this.fn + this.tn), String(this.tp + this.fp + this.fn + this.tn)]
   411→    ];
   412→  }
   413→}

   416→// SECTION 3: PATTERN OPPORTUNITY ANALYZER
   417→// ============================================================
   418→
   419→class PatternOpportunityAnalyzer {
   420→  constructor(dataset) {
   421→    this.dataset = dataset;
   422→    this.opportunities = {
   423→      totalInvoices: 0,
   424→      couldMatchByPattern: 0,  // Had a matching layout in pattern store
   425→      actuallyMatchedByPattern: 0, // System successfully matched
   426→      correctMatches: 0,         // Matched correct supplier
   427→      falseMatches: 0,           // Matched wrong supplier (FMR source)
   428→      correctAiFallbacks: 0,     // Rightfully sent to AI
   429→      missedOpportunities: 0,    // Could have matched but didn't
   430→      forcedToAi: 0              // No pattern available at all
   431→    };
   432→  }
   433→  
   434→  analyze(results) {
   435→    const allInvoices = [
   436→      ...this.dataset.invoices.train,
   437→      ...this.dataset.invoices.validation,
   438→      ...this.dataset.invoices.holdout
   439→    ];
   440→    
   441→    this.opportunities.totalInvoices = allInvoices.length;
   442→    
   443→    for (const invoice of allInvoices) {
   444→      const result = results.get(invoice.id);
   445→      if (!result) continue;
   446→      
   447→      // Analyze each outcome
   448→      switch (result.outcome) {
   449→        case 'PATTERN_MATCH':
   450→          this.opportunities.actuallyMatchedByPattern++;
   451→          if (result.isCorrectMatch) {
   452→            this.opportunities.correctMatches++;
   453→          } else {
   454→            this.opportunities.falseMatches++;
   455→          }
   456→          break;
   457→          
   458→        case 'AI_FALLBACK':
   459→          if (result.hadPatternAvailable) {
   460→            this.opportunities.missedOpportunities++;
   461→          } else {
   462→            this.opportunities.forcedToAi++;
   463→          }
   464→          this.opportunities.correctAiFallbacks++;
   465→          break;
   466→          
   467→        case 'FALSE_MATCH':
   468→          this.opportunities.actuallyMatchedByPattern++;
   469→          this.opportunities.falseMatches++;
   470→          break;
   471→      }
   472→    }
   473→    
   474→    // Calculate rates
   475→    this.opportunities.couldMatchByPattern = 
   476→      this.opportunities.correctMatches + 
   477→      this.opportunities.falseMatches + 
   478→      this.opportunities.missedOpportunities;
   479→    
   480→    return this.getReport();
   481→  }
   482→  
   483→  getReport() {
   484→    const opp = this.opportunities;
   485→    const total = opp.totalInvoices || 1;
   486→    
   487→    return {
   488→      counts: { ...opp },
   489→      rates: {
   490→        patternUtilizationRate: (opp.actuallyMatchedByPattern / opp.couldMatchByPattern * 100).toFixed(1) + '%',
   491→        falseMatchAmongPattern: (opp.falseMatches / opp.actuallyMatchedByPattern * 100).toFixed(2) + '%',
   492→        aiFallbackRate: ((opp.correctAiFallbacks) / total * 100).toFixed(1) + '%',
   493→        opportunityWasteRate: (opp.missedOppportunities / opp.couldMatchByPattern * 100).toFixed(1) + '%'
   494→      },
   495→      keyMetric: {
   496→        // The REAL question: of all invoices that COULD be pattern-matched,
   497→        // what percentage were matched CORRECTLY without false matches?
   498→        safePatternMatchRate: (
   499→          opp.correctMatches / opp.couldMatchByPattern * 100
   500→        ).toFixed(1) + '%'
   501→      },
   502→      interpretation: this.interpret()
   503→    };
   504→  }
   505→  
   506→  interpret() {
   507→    const opp = this.opportunities;
   508→    
   509→    if (opp.falseMatches === 0 && opp.correctMatches > 0) {
   510→      if (opp.missedOpportunities > opp.correctMatches) {
   511→        return 'System is CONSERVATIVE - Zero false matches but missing valid opportunities';
   512→      } else {
   513→        return 'System is WELL-BALANCED - Good matches with minimal risk';
   514→      }
   515→    } else if (opp.falseMatches > opp.correctMatches * 0.01) {
   516→      return 'System is AGGRESSIVE - Too many false matches relative to correct ones';
   517→    } else {
   518→      return 'MODERATE RISK - Some false matches exist but within acceptable bounds';
   519→    }
   520→  }
   521→}

   526→
   527→class OverfittingDetector {
   528→  constructor() {
   529→    this.splitResults = {};
   530→  }
   531→  
   532→  recordSplit(splitName, metrics) {
   533→    this.splitResults[splitName] = metrics;
   534→  }
   535→  
   536→  detect() {
   537→    const { train, validation, holdout } = this.splitResults;
   538→    
   539→    if (!train || !validation || !holdout) {
   540→      return {
   541→        overfittingDetected: null,
   542→        reason: 'Insufficient data - need all three splits',
   543→        recommendation: 'Run benchmark on complete dataset with splits enabled'
   544→      };
   545→    }
   546→    
   547→    const issues = [];
   548→    
   549→    // Check 1: Training significantly better than holdout (>10% gap)
   550→    const f1Gap = train.f1Score - holdout.f1Score;
   551→    if (f1Gap > 0.1) {
   552→      issues.push({
   553→        type: 'F1_SCORE_GAP',
   554→        severity: f1Gap > 0.2 ? 'HIGH' : 'MEDIUM',
   555→        message: `Training F1 ${(train.f1Score * 100).toFixed(1)}% vs Holdout F1 ${(holdout.f1Score * 100).toFixed(1)}% (gap: ${(f1Gap * 100).toFixed(1)}%)`
   556→      });
   557→    }
   558→    
   559→    // Check 2: False Match Rate much worse on holdout
   560→    const fmrGap = holdout.falseMatchRate - train.falseMatchRate;
   561→    if (fmrGap > 0.02) {
   562→      issues.push({
   563→        type: 'FMR_DEGRADATION',
   564→        severity: fmrGap > 0.05 ? 'CRITICAL' : 'MEDIUM',
   565→        message: `Holdout FMR ${(holdout.falseMatchRate * 100).toFixed(2)}% vs Training FMR ${(train.falseMatchRate * 100).toFixed(2)}%`
   566→      });
   567→    }
   568→    
   569→    // Check 3: Precision drops significantly on unseen data
   570→    const precisionGap = train.precision - holdout.precision;
   571→    if (precisionGap > 0.15) {
   572→      issues.push({
   573→        type: 'PRECISION_DROP',
   574→        severity: 'HIGH',
   575→        message: `Precision drops from ${(train.precision * 100).toFixed(1)}% (train) to ${(holdout.precision * 100).toFixed(1)}% (holdout)`
   576→      });
   577→    }
   578→    
   579→    return {
   580→      overfittingDetected: issues.length > 0,
   581→      issueCount: issues.length,
   582→      issues: issues,
   583→      overallAssessment: issues.length === 0 ? 'NO OVERFITTING DETECTED' : 'POTENTIAL OVERFITTING',
   584→      recommendation: this.generateRecommendation(issues)
   585→    };
   586→  }
   587→  
   588→  generateRecommendation(issues) {
   589→    if (issues.some(i => i.severity === 'CRITICAL')) {
   590→      return 'STOP - Significant overfitting detected. Review model complexity and regularization.';
   591→    } else if (issues.length > 0) {
   592→      return 'CAUTION - Minor performance gaps detected. Monitor closely in production.';
   593→    } else {
   594→      return 'GOOD - Model generalizes well across all data splits.';
   595→    }
   596→  }

   774→  processInvoiceRigorously(invoice, patternStore) {
   775→    const startTime = performance.now();
   776→    
   777→    // Find candidates by layout
   778→    const candidates = Array.from(patternStore.values())
   779→      .filter(p => p.layoutHash === invoice.layoutHash);
   780→    
   781→    // Simulate enhanced engine logic
   782→    let outcome;
   783→    let matchedSupplierId = null;
   784→    let isCorrectMatch = false;
   785→    let hadPatternAvailable = candidates.length > 0;
   786→    let confidence = 0;
   787→    
   788→    if (candidates.length === 0) {
   789→      // No pattern available - must use AI
   790→      outcome = 'AI_FALLBACK';
   791→      confidence = 0;
   792→    } else if (candidates.length === 1) {
   793→      // Single candidate - check if correct
   794→      const candidate = candidates[0];
   795→      matchedSupplierId = candidate.supplierId;
   796→      isCorrectMatch = candidate.supplierId === invoice.supplierId;
   797→      
   798→      // Apply semantic checks (simplified)
   799→      const semanticScore = this.calculateSemanticScore(invoice, candidate);
   800→      confidence = semanticScore;
   801→      
   802→      if (semanticScore > 0.7 && isCorrectMatch) {
   803→        outcome = 'PATTERN_MATCH';
   804→      } else if (semanticScore > 0.7 && !isCorrectMatch) {
   805→        // This is a FALSE MATCH - high confidence but wrong supplier!
   806→        outcome = 'FALSE_MATCH';
   807→      } else {
   808→        // Low semantic score - send to AI even if layout matches
   809→        outcome = 'AI_FALLBACK';
   810→      }
   811→    } else {
   812→      // Multiple candidates (shared layout scenario) - DANGEROUS
   813→      const bestCandidate = this.selectBestCandidate(invoice, candidates);
   814→      matchedSupplierId = bestCandidate.supplierId;
   815→      isCorrectMatch = bestCandidate.supplierId === invoice.supplierId;
   816→      confidence = this.calculateSemanticScore(invoice, bestCandidate);
   817→      
   818→      // With multiple candidates, we're more conservative
   819→      if (confidence > 0.85 && isCorrectMatch) {
   820→        outcome = 'PATTERN_MATCH';
   821→      } else if (!isCorrectMatch && confidence > 0.7) {
   822→        outcome = 'FALSE_MATCH'; // Wrong match!
   823→      } else {
   824→        outcome = 'AI_FALLBACK'; // Too risky
   825→      }
   826→    }
   827→    
   828→    const processingTime = performance.now() - startTime;
   829→    
   830→    return {
   831→      invoiceId: invoice.id,
   832→      outcome,
   833→      matchedSupplierId,
   834→      actualSupplierId: invoice.supplierId,
   835→      isCorrectMatch,
   836→      hadPatternAvailable,
   837→      confidence,
   838→      processingTimeMs: processingTime,
   839→      candidateCount: candidates.length,
   840→      hadSharedLayout: invoice.hasSharedLayout,
   841→      hadOcrError: invoice.hasOcrError
   842→    };
   843→  }

   845→  calculateSemanticScore(invoice, pattern) {
   846→    let score = 0.5; // Base score
   847→    
   848→    // Supplier name similarity (if available)
   849→    if (invoice.expectedSupplierName && pattern.supplierName) {
   850→      const nameOverlap = this.stringSimilarity(
   851→        invoice.expectedSupplierName.toLowerCase(),
   852→        pattern.supplierName.toLowerCase()
   853→      );
   854→      score += nameOverlap * 0.3;
   855→    }
   856→    
   857→    // Currency check
   858→    if (invoice.currency === pattern.currency) {
   859→      score += 0.15;
   860→    } else {
   861→      score -= 0.2; // Penalty for mismatch
   862→    }
   863→    
   864→    // Tenant check
   865→    if (invoice.tenantId === pattern.tenantId) {
   866→      score += 0.15;
   867→    } else {
   868→      score -= 0.3; // Big penalty for wrong tenant
   869→    }
   870→    
   871→    // Format type check
   872→    if (invoice.formatType === pattern.formatType) {
   873→      score += 0.1;
   874→    }
   875→    
   876→    // OCR error penalty
   877→    if (invoice.hasOcrError) {
   878→      score -= 0.1;
   879→    }
   880→    
   881→    // Shared layout penalty (more conservative when layouts are shared)
   882→    if (invoice.hasSharedLayout) {
   883→      score -= 0.05;
   884→    }
   885→    
   886→    return Math.max(0, Math.min(1, score));
   887→  }

   889→  selectBestCandidate(invoice, candidates) {
   890→    let bestScore = -1;
   891→    let bestCandidate = null;
   892→    
   893→    for (const candidate of candidates) {
   894→      const score = this.calculateSemanticScore(invoice, candidate);
   895→      if (score > bestScore) {
   896→        bestScore = score;
   897→        bestCandidate = candidate;
   898→      }
   899→    }
   900→    
   901→    return bestCandidate || candidates[0];
   902→  }

   904→  stringSimilarity(str1, str2) {
   905→    if (str1 === str2) return 1.0;
   906→    if (!str1 || !str2) return 0.0;
   907→    
   908→    const longer = str1.length > str2.length ? str1 : str2;
   909→    const shorter = str1.length > str2.length ? str2 : str1;
   910→    
   911→    if (longer.length === 0) return 1.0;
   912→    
   913→    const costs = [];
   914→    for (let i = 0; i <= longer.length; i++) {
   915→      let lastValue = i;
   916→      for (let j = 0; j <= shorter.length; j++) {
   917→        if (i === 0) {
   918→          costs[j] = j;
   919→        } else if (j > 0) {
   920→          let newValue = costs[j - 1];
   921→          if (longer.charAt(i - 1) !== shorter.charAt(j - 1)) {
   922→            newValue = Math.min(Math.min(newValue, lastValue), costs[j]) + 1;
   923→          }
   924→          costs[j - 1] = lastValue;
   925→          lastValue = newValue;
   926→        }
   927→      }
   928→      if (i > 0) costs[shorter.length] = lastValue;
   929→    }
   930→    
   931→    return (longer.length - costs[shorter.length]) / longer.length;
   932→  }

  1128→  variance(values) {
  1129→    if (values.length === 0) return 0;
  1130→    const mean = values.reduce((a, b) => a + b, 0) / values.length;
  1131→    return values.reduce((sum, val) => sum + Math.pow(val - mean, 2), 0) / values.length;
  1132→  }
