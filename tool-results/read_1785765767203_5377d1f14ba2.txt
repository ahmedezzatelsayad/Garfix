   186→function StatementView({ type, company, data, setData }: { type: "client" | "supplier"; company: { slug: string }; data: ClientStatement | null; setData: (d: ClientStatement | null) => void }) {
   187→  const [selectedId, setSelectedId] = useState<number | null>(null);
   188→  const [loadingStatement, setLoadingStatement] = useState(false);
   189→
   190→  const clientsQuery = useClients(company.slug);
   191→  const suppliersQuery = useSuppliers(company.slug);
   192→  const clientStatementQuery = useClientStatement(company.slug, type === "client" ? selectedId : null);
   193→  const supplierStatementQuery = useSupplierStatement(company.slug, type === "supplier" ? selectedId : null);
   194→
   195→  const contacts = type === "client"
   196→    ? (clientsQuery.data?.clients || [])
   197→    : (suppliersQuery.data?.suppliers || []);
   198→
   199→  const loadStatement = () => {
   200→    if (!selectedId) { toast.error("اختر " + (type === "client" ? "عميل" : "مورد")); return; }
   201→    setLoadingStatement(true);
   202→    const query = type === "client" ? clientStatementQuery : supplierStatementQuery;
   203→    if (query.data) {
   204→      if (type === "client") {
   205→        const cs = (query.data as { clientStatement: { clientName: string; openingBalance: number; lines: ClientStatementLine[]; closingBalance: number } }).clientStatement;
   206→        setData(cs);
   207→      } else {
   208→        const ss = (query.data as { supplierStatement: { supplierName: string; openingBalance: number; lines: ClientStatementLine[]; closingBalance: number } }).supplierStatement;
   209→        setData({ clientName: ss.supplierName, openingBalance: ss.openingBalance, lines: ss.lines, closingBalance: ss.closingBalance });
   210→      }
   211→      setLoadingStatement(false);
   212→    } else if (query.isError) {
   213→      toast.error("تعذّر تحميل كشف الحساب");
   214→      setData(null);
   215→      setLoadingStatement(false);
   216→    }
   217→  };
   218→
   219→  const totalDebit = data?.lines.reduce((s, l) => s + l.debit, 0) || 0;
   220→  const totalCredit = data?.lines.reduce((s, l) => s + l.credit, 0) || 0;
   221→
   222→  return (
   223→    <div className="flex flex-col gap-4">
   224→      <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex gap-3 items-center flex-wrap">
   225→        <div className="flex items-center gap-1.5">
   226→          <label className="text-[11px] font-bold text-muted-foreground">{type === "client" ? "العميل" : "المورد"}</label>
   227→          <select value={selectedId ?? ""} onChange={(e) => setSelectedId(e.target.value ? Number(e.target.value) : null)} className={cn(inputStyle, "w-auto min-w-[200px]")} disabled={loadingStatement}>
   228→            <option value="">— اختر —</option>
   229→            {contacts.map((c) => <option key={c.id} value={c.id}>{c.name}</option>)}
   230→          </select>
   231→        </div>
   232→        <button onClick={loadStatement} disabled={!selectedId || loadingStatement} className="py-2 px-4 rounded-sm bg-accent text-accent-foreground border border-border text-[12px] font-bold cursor-pointer disabled:opacity-70 inline-flex items-center gap-1.5">
   233→          <RefreshCw size={12} className={loadingStatement ? "animate-spin" : ""} /> {loadingStatement ? "جارٍ…" : "عرض"}
   234→        </button>
   235→      </div>
   236→
   237→      {loadingStatement ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : !data ? (
   238→        <div className="bg-card rounded-[14px] border border-border p-12 text-center text-muted-foreground">
   239→          <FileText size={36} className="opacity-30 mb-2" />اختر {type === "client" ? "عميل" : "مورد"} ثم اضغط عرض
   240→        </div>
   241→      ) : (
   242→        <div className="flex flex-col gap-4">
   243→          <div className="grid grid-cols-[repeat(auto-fit,minmax(140px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(180px,1fr))] gap-3">
   244→            <div className="kpi-card bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3 hover-lift">
   245→              <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-emerald-600/20 text-emerald-600"><FileText size={18} /></div>
   246→              <div><div className="text-[11px] text-muted-foreground">{data.clientName}</div><div className="text-[13px] font-bold">رصيد افتتاحي: {fmt(data.openingBalance)}</div></div>
   247→            </div>
   248→            <div className="kpi-card bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3 hover-lift">
   249→              <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-emerald-500/20 text-emerald-500"><TrendingUp size={18} /></div>
   250→              <div><div className="text-[11px] text-muted-foreground">إجمالي مدين</div><div className="text-lg font-extrabold [direction:ltr] text-end">{fmt(totalDebit)}</div></div>
   251→            </div>
   252→            <div className="kpi-card bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3 hover-lift">
   253→              <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-red-500/20 text-red-500"><TrendingDown size={18} /></div>
   254→              <div><div className="text-[11px] text-muted-foreground">إجمالي دائن</div><div className="text-lg font-extrabold [direction:ltr] text-end">{fmt(totalCredit)}</div></div>
   255→            </div>
   256→            <div className="kpi-card bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3 hover-lift">
   257→              <div className={cn("w-10 h-10 rounded-sm flex items-center justify-center", data.closingBalance >= 0 ? "bg-emerald-600/20 text-emerald-600" : "bg-red-500/20 text-red-500")}><CheckCircle2 size={18} /></div>
   258→              <div><div className="text-[11px] text-muted-foreground">رصيد إقفالي</div><div className={cn("text-lg font-extrabold [direction:ltr] text-end", data.closingBalance >= 0 ? "text-emerald-600" : "text-red-500")}>{fmt(data.closingBalance)}</div></div>
   259→            </div>
   260→          </div>
   261→
   262→          <div className="bg-card rounded-[14px] border border-border overflow-hidden">
   263→            {data.lines.length === 0 ? <Empty label="حركات" /> : (
   264→              <div className="overflow-x-auto garfix-scroll">
   265→                <table className="table-enterprise w-full border-collapse">
   266→                  <thead><tr className="border-b border-border bg-muted">
   267→                    <th className={thStyle}>التاريخ</th><th className={thStyle}>النوع</th><th className={thStyle}>المرجع</th>
   268→                    <th className={cn(thStyle, "text-end")}>مدين</th><th className={cn(thStyle, "text-end")}>دائن</th><th className={cn(thStyle, "text-end")}>الرصيد</th>
   269→                  </tr></thead>
   270→                  <tbody>
   271→                    {data.lines.map((l, i) => (
   272→                      <tr key={i} className="border-b border-border">
   273→                        <td className={tdStyle} dir="ltr">{l.date}</td>

   275→                        <td className={cn(tdStyle, "font-mono")}>{l.reference}</td>
   276→                        <td className={cn(tdStyle, "[direction:ltr] text-end", l.debit > 0 ? "text-emerald-600" : "")}>{l.debit ? fmt(l.debit) : "—"}</td>

   279→                      </tr>
   280→                    ))}
   281→                  </tbody>
   282→                  <tfoot>
   283→                    <tr className="border-t-2 border-border bg-muted font-extrabold">

   300→/* ─── PDC List ──────────────────────────────────────────────────────────────── */
   301→function PDCList({ pdcs, company, onRefresh }: { pdcs: PDC[]; company: { slug: string }; onRefresh: () => void }) {
   302→  const [actionId, setActionId] = useState<number | null>(null);
   303→  const pdcActionMutation = usePDCAction();
   304→
   305→  const handleAction = (pdcId: number, action: string) => {
   306→    setActionId(pdcId);
   307→    pdcActionMutation.mutate(
   308→      { id: pdcId, action, companySlug: company.slug },
   309→      {
   310→        onSuccess: () => {
   311→          const actionLabel = action === "deposit" ? "تسليم الشيك" : action === "clear" ? "تحصيل الشيك" : "إرجاع الشيك";
   312→          toast.success(`تم ${actionLabel}`);
   313→          onRefresh();
   314→          setActionId(null);
   315→        },
   316→        onError: (err) => { toast.error(err.message || "تعذّر تنفيذ الإجراء"); setActionId(null); },
   317→      },
   318→    );
   319→  };
   320→

   327→    };

   329→  };
   330→
   331→  const totalAmount = pdcs.reduce((s, p) => s + p.amount, 0);
   332→  const pendingTotal = pdcs.filter(p => p.status === "pending").reduce((s, p) => s + p.amount, 0);
   333→

   340→        </div>
   341→        <div className="kpi-card bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3 hover-lift">
   342→          <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-amber-500/20 text-amber-500"><Clock size={18} /></div>
   343→          <div><div className="text-[11px] text-muted-foreground">المعلّقة</div><div className="text-lg font-extrabold [direction:ltr] text-end text-amber-500">{fmt(pendingTotal)}</div></div>
   344→        </div>
   345→      </div>
   346→      <div className="bg-card rounded-[14px] border border-border overflow-hidden">
   347→        {pdcs.length === 0 ? <Empty label="شيكات مؤجلة" /> : (

   349→            <table className="table-enterprise w-full border-collapse">

   352→                <th className={thStyle}>الاسم</th><th className={cn(thStyle, "text-end")}>المبلغ</th><th className={thStyle}>تاريخ الاستحقاق</th>
   353→                <th className={thStyle}>الحالة</th><th className={thStyle}>إجراء</th>
   354→              </tr></thead>

   362→                      <td className={tdStyle}>{p.bankName}</td>
   363→                      <td className={tdStyle}><span className={cn("py-0.5 px-2.5 rounded-[12px] text-[11px] font-bold", p.direction === "receivable" ? "bg-emerald-500/15 text-emerald-500" : "bg-red-500/15 text-red-500")}>{p.direction === "receivable" ? "مقبوض" : "مدفوع"}</span></td>
   364→                      <td className={tdStyle}>{name || "—"}</td>
   365→                      <td className={cn(tdStyle, "[direction:ltr] text-end font-bold")}>{fmt(p.amount)}</td>
   366→                      <td className={tdStyle} dir="ltr">{p.dueDate}</td>
   367→                      <td className={tdStyle}><span className={cn("py-0.5 px-2.5 rounded-[12px] text-[11px] font-bold", sc.badge)}>{sc.label}</span></td>
   368→                      <td className={tdStyle}>
   369→                        <div className="flex items-center gap-1">
   370→                          {p.status === "pending" && <button onClick={() => handleAction(p.id, "deposit")} disabled={actionId === p.id} className="py-1 px-2 rounded-md bg-blue-500/10 border border-blue-500/30 text-blue-600 text-[10px] font-bold cursor-pointer disabled:opacity-50">تسليم</button>}
