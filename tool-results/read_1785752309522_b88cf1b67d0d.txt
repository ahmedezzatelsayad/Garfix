   111→// Generate animated counter value
   112→function useAnimatedValue(target: number, duration = 800): number {
   113→  const [current, setCurrent] = useState(0);
   114→  const startTime = useRef<number | null>(null);
   115→  const startValue = useRef(0);
   116→
   117→  useEffect(() => {
   118→    if (target === 0) {
   119→      setCurrent(0);
   120→      return;
   121→    }
   122→
   123→    const animate = (timestamp: number) => {
   124→      if (!startTime.current) {
   125→        startTime.current = timestamp;
   126→        startValue.current = current;
   127→      }
   128→
   129→      const elapsed = timestamp - startTime.current;
   130→      const progress = Math.min(elapsed / duration, 1);
   131→      
   132→      // Easing function (ease-out cubic)
   133→      const eased = 1 - Math.pow(1 - progress, 3);
   134→      
   135→      setCurrent(Math.floor(startValue.current + (target - startValue.current) * eased));
   136→
   137→      if (progress < 1) {
   138→        requestAnimationFrame(animate);
   139→      }
   140→    };
   141→
   142→    requestAnimationFrame(animate);
   143→  }, [target, duration]);
   144→
   145→  return current;
   146→}

   165→/**
   166→ * ProgressRing - Circular progress indicator
   167→ */
   168→function ProgressRing({ 
   169→  percentage, 
   170→  size = 60, 
   171→  strokeWidth = 6, 
   172→  color = "#047857" 
   173→}: { 
   174→  percentage: number; 
   175→  size?: number; 
   176→  strokeWidth?: number; 
   177→  color?: string;
   178→}) {
   179→  const radius = (size - strokeWidth) / 2;
   180→  const circumference = radius * 2 * Math.PI;
   181→  const offset = circumference - (percentage / 100) * circumference;
   182→
   183→  return (
   184→    <svg width={size} height={size} className="transform -rotate-90">
   185→      {/* Background circle */}
   186→      <circle
   187→        cx={size / 2}
   188→        cy={size / 2}
   189→        r={radius}
   190→        fill="none"
   191→        stroke="rgba(255,255,255,0.05)"
   192→        strokeWidth={strokeWidth}
   193→      />
   194→      {/* Progress circle */}
   195→      <circle
   196→        cx={size / 2}
   197→        cy={size / 2}
   198→        r={radius}
   199→        fill="none"
   200→        stroke={color}
   201→        strokeWidth={strokeWidth}
   202→        strokeDasharray={circumference}
   203→        strokeDashoffset={offset}
   204→        strokeLinecap="round"
   205→        style={{ transition: 'stroke-dashoffset 1s ease-out' }}
   206→      />
   207→      {/* Percentage text */}
   208→      <text
   209→        x={size / 2}
   210→        y={size / 2}
   211→        textAnchor="middle"
   212→        dominantBaseline="central"
   213→        fill="#f3f4f6"
   214→        fontSize="14"
   215→        fontWeight="bold"
   216→        className="tabular-nums"
   217→      >
   218→        {Math.round(percentage)}%
   219→      </text>
   220→    </svg>
   221→  );
   222→}
