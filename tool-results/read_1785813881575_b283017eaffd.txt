    47→function roleColor(role: string): string {
    48→  const preset = ROLE_PRESETS.find((r) => r.value === role);
    49→  return preset?.color || "#6b7280";

    52→/** Build a short text summary of a member's permissions for the table. */
    53→function permSummary(perms: Record<string, number>, role: string, isFounder?: boolean): string {
    54→  if (isFounder || role === "admin") return "كل الصلاحيات";
    55→  const enabled = PERMISSION_CATALOG.filter((p) => !p.locked && perms[p.key]);
    56→  if (enabled.length === 0) return "بدون صلاحيات";
    57→  if (enabled.length === PERMISSION_CATALOG.filter((p) => !p.locked).length) return "كل الصلاحيات (غير الإدارية)";
    58→  return `${enabled.length} صلاحية`;
    59→}

   296→                  </div>
   297→                </div>

   306→            </div>
   307→          </div>
   308→          </>
   309→        )}
   310→      </div>
   311→
   312→      {showInvite && (
   313→        <InviteDialog
   314→          companySlug={activeCompany.slug}
   315→          onClose={() => setShowInvite(false)}
   316→          onSaved={() => { setShowInvite(false); refetch(); }}

   319→
   320→      {editing && (

   334→function InviteDialog({
   335→  companySlug, onClose, onSaved,
   336→}: { companySlug: string; onClose: () => void; onSaved: () => void }) {
   337→  const [email, setEmail] = useState("");
   338→  const [displayName, setDisplayName] = useState("");
   339→  const [role, setRole] = useState<string>("employee");
   340→  const [perms, setPerms] = useState<Record<string, number>>({});
   341→  const addMemberMutation = useAddCompanyMember();
   342→  const [result, setResult] = useState<{ temporaryPassword?: string | null; created?: boolean; email?: string } | null>(null);
   343→
   344→  const togglePerm = (key: string) => {
   345→    setPerms((p) => ({ ...p, [key]: p[key] ? 0 : 1 }));
   346→  };
   347→
   348→  const submit = async () => {
   349→    if (!email) { toast.error("البريد الإلكتروني مطلوب"); return; }
   350→    try {
   351→      const raw = await addMemberMutation.mutateAsync({
   352→        slug: companySlug,
   353→        uid: email, // API accepts email as uid for invite
   354→        role,
   355→        permissions: perms,
   356→        email,
   357→        displayName: displayName || undefined,
   358→      });
   359→      const data = raw as unknown as Record<string, unknown>;
   360→      if (data.created) {
   361→        toast.success("تم إنشاء الحساب وإضافته للشركة");
   362→      } else {
   363→        toast.success("تمت إضافة العضو للشركة");
   364→      }
   365→      setResult({
   366→        temporaryPassword: data.temporaryPassword as string | null | undefined,
   367→        created: data.created as boolean | undefined,
   368→        email: (data.member as Record<string, unknown> | undefined)?.email as string | undefined,
   369→      });
   370→      // Don't close immediately if a temp password was returned — let admin copy it
   371→      if (!data.temporaryPassword) {
   372→        onSaved();
   373→      }
   374→    } catch (err) {
   375→      toast.error(err instanceof Error ? err.message : "خطأ");
   376→    }
   377→  };
   378→
   379→  const saving = addMemberMutation.isPending;
   380→
   381→  return (
   382→    <DialogShell title="دعوة عضو جديد" icon={<UserPlus size={18} />} onClose={onClose}>
   383→      {result ? (
   384→        <div className="flex flex-col gap-3.5">
   385→          <div className="p-3.5 rounded-md bg-muted border border-border">
   386→            <div className="text-[13px] font-bold mb-1.5">
   387→              {result.created ? "✅ تم إنشاء حساب جديد" : "✅ تمت إضافة العضو"}
   388→            </div>
   389→            {result.temporaryPassword ? (
   390→              <div className="text-xs text-muted-foreground flex flex-col gap-1.5">
   391→                <div>البريد: <span dir="ltr">{result.email}</span></div>
   392→                <div>كلمة المرور المؤقتة:</div>
   393→                <div className="[direction:ltr] text-center p-2.5 rounded-sm bg-background border border-dashed border-border font-mono text-sm font-bold tracking-wider">
   394→                  {result.temporaryPassword}
   395→                </div>
   396→                <div className="text-[11px] text-muted-foreground">
   397→                  شاركها مع المستخدم. يمكنه تغييرها لاحقاً عبر "نسيت كلمة المرور".
   398→                </div>
   399→              </div>
   400→            ) : (
   401→              <div className="text-xs text-muted-foreground">
   402→                المستخدم موجود مسبقاً وتم إضافته للشركة بالصلاحيات المحددة.
   403→              </div>
   404→            )}
   405→          </div>
   406→          <div className="flex justify-end">
   407→            <button onClick={onSaved} className={primaryBtn}>تم</button>
   408→          </div>
   409→        </div>
   410→      ) : (
   411→        <div className="flex flex-col gap-3.5">
   412→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   413→            <div>

   458→function EditDialog({
   459→  member, companySlug, onClose, onSaved,
   460→}: { member: Member; companySlug: string; onClose: () => void; onSaved: () => void }) {
   461→  const updateMemberMutation = useUpdateCompanyMember();
   462→  const [role, setRole] = useState<string>(member.role);
   463→  const [perms, setPerms] = useState<Record<string, number>>({ ...member.permissions });
   464→
   465→  const togglePerm = (key: string) => {
   466→    setPerms((p) => ({ ...p, [key]: p[key] ? 0 : 1 }));
   467→  };
   468→
   469→  const save = async () => {
   470→    try {
   471→      await updateMemberMutation.mutateAsync({
   472→        slug: companySlug, uid: member.uid, role, permissions: perms,
   473→      });
   474→      toast.success("تم تحديث العضو");
   475→      onSaved();
   476→    } catch (err) {
   477→      toast.error(err instanceof Error ? err.message : "خطأ");
   478→    }
   479→  };
   480→
   481→  const saving = updateMemberMutation.isPending;
   482→
   483→  const isFounder = member.isFounder;
   484→
   485→  return (
   486→    <DialogShell title={`تعديل: ${member.displayName}`} icon={<Key size={18} />} onClose={onClose}>
   487→      <div className="flex flex-col gap-3.5">
   488→        <div className="p-2.5 px-3 rounded-md bg-muted border border-border text-xs text-muted-foreground flex items-center gap-2">
   489→          <Mail size={14} />
   490→          <span dir="ltr">{member.email}</span>
   491→        </div>
   492→
   493→        <div>
   494→          <label className={labelStyle}>الدور</label>
   495→          {isFounder ? (
   496→            <div className="p-2.5 px-3 rounded-md bg-[#f59e0b22] border border-[#f59e0b] text-[#f59e0b] text-xs font-bold flex items-center gap-1.5">
   497→              <Crown size={14} /> مؤسس — لا يمكن تغيير الدور
   498→            </div>
   499→          ) : (
   500→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   501→              {ROLE_PRESETS.map((r) => (
   502→                <button
   503→                  key={r.value}
   504→                  onClick={() => setRole(r.value)}
   505→                  className={`max-md:min-h-[44px] py-2 px-2.5 rounded-lg font-inherit text-[11px] font-bold cursor-pointer text-right ${role === r.value ? `[background:${r.color}22] text-[${r.color}]` : "bg-[var(--background)] text-[var(--foreground)]"}`}
   506→                  style={{ border: `1.5px solid ${role === r.value ? r.color : "var(--border)"}` }}
   507→                >
   508→                  {r.label}
   509→                </button>
   510→              ))}
   511→            </div>
   512→          )}
   513→        </div>
   514→
   515→        {!isFounder && role !== "admin" && (
   516→          <PermEditor perms={perms} onToggle={togglePerm} />
   517→        )}
   518→
   519→        {(isFounder || role === "admin") && (
   520→          <div className="p-2.5 px-3 rounded-md bg-muted border border-border text-[11px] text-muted-foreground">
   521→            {isFounder
   522→              ? "👑 المؤسس يملك كل الصلاحيات تلقائياً ولا يمكن تقييدها."
   523→              : "👑 المدير يملك كل الصلاحيات تلقائياً ولا يمكن تقييدها."}
   524→          </div>
   525→        )}
   526→
   527→        <div className="flex gap-2.5 justify-end">
   528→          <button onClick={onClose} className={ghostBtn}>إلغاء</button>
   529→          <button onClick={save} disabled={saving || isFounder} className={primaryBtn}>
   530→            {saving ? "جارٍ…" : "حفظ"}
   531→          </button>
   532→        </div>
   533→      </div>
   534→    </DialogShell>
   535→  );
   536→}

   538→// ─── Permission toggles ───────────────────────────────────────────────────
   539→
   540→function PermEditor({
   541→  perms, onToggle, compact = false,
   542→}: { perms: Record<string, number>; onToggle: (key: string) => void; compact?: boolean }) {
   543→  // Group permissions by their `group` field
   544→  const groups = useMemo(() => {
   545→    const m: Record<string, typeof PERMISSION_CATALOG> = {};
   546→    for (const p of PERMISSION_CATALOG) {
   547→      (m[p.group] = m[p.group] || []).push(p);
   548→    }
   549→    return m;
   550→  }, []);
   551→
   552→  return (
   553→    <div>
   554→      <label className={labelStyle}>الصلاحيات التفصيلية</label>
   555→      <div className="bg-background border border-border rounded-md p-2.5 flex flex-col gap-2.5">
   556→        {Object.entries(groups).map(([group, items]) => (
   557→          <div key={group}>
   558→            <div className="text-[10px] font-bold text-muted-foreground uppercase tracking-wide mb-1.5">{group}</div>
   559→            <div className={cn("grid gap-1.5", compact ? "grid-cols-1" : "grid-cols-1 sm:grid-cols-[repeat(auto-fill,minmax(180px,1fr))]")}>
   560→              {items.map((p) => {
   561→                const on = !!perms[p.key];
   562→                const locked = LOCKED_PERMS.includes(p.key);
   563→                return (
   564→                  <button
   565→                    key={p.key}
   566→                    onClick={() => onToggle(p.key)}
   567→                    className={`max-md:min-h-[44px] flex items-center gap-2 py-2 px-2.5 rounded-lg font-inherit text-xs cursor-pointer text-right text-[var(--foreground)] ${on ? "bg-[var(--primary)]/15" : "bg-[var(--card)]"} ${locked ? "opacity-65" : "opacity-100"}`}
   568→                    style={{ border: `1px solid ${on ? "var(--primary)" : "var(--border)"}` }}
   569→                    title={locked ? "صلاحية إدارية — تمنح تلقائياً للمدير فقط" : p.label}
   570→                  >
   571→                    <span className="text-sm">{p.icon}</span>
   572→                    <span className="flex-1">{p.label}</span>
   573→                    <span
   574→                      className={`w-8 h-[18px] rounded-full relative shrink-0 transition-[background] duration-[150ms] ${on ? "bg-[var(--primary)]" : "bg-[var(--muted)]"}`}
   575→                    >
   576→                      <span
   577→                        className={`absolute top-0.5 w-[14px] h-[14px] rounded-full bg-white transition-[left] duration-[150ms] ${on ? "left-0.5" : "left-4"}`}
   578→                      />
   579→                    </span>
   580→                  </button>
   581→                );
   582→              })}
   583→            </div>
   584→          </div>
   585→        ))}
   586→      </div>
   587→      <div className="text-[10px] text-muted-foreground mt-1.5">
   588→        🔒 الصلاحيات الإدارية تُمنح تلقائياً للمدير والمؤسس فقط.
   589→      </div>
   590→    </div>
   591→  );
   592→}
