     1→/**
     2→ * platform-admin.ts — React Query hooks for the platform-admin domain.
     3→ *
     4→ * Provides typed query and mutation hooks for tenant management, audit logs,
     5→ * platform stats, AI provider configuration, AI usage & orchestration,
     6→ * feature flags, announcements, integrations, review queue, queue failures,
     7→ * retention cleanup, support tickets, SaaS user management, SaaS payments,
     8→ * and landing content. All hooks use the centralized `queryKeys` factory
     9→ * for granular cache invalidation and the typed `apiGet`/`apiPost`/
    10→ * `apiPatch`/`apiDelete` helpers for consistent request handling.
    11→ */

   236→/**
   237→ * Fetch the list of all platform tenants.
   238→ *
   239→ * Returns every tenant registered on the platform including their
   240→ * slug, name, and current status.
   241→ */
   242→export function usePlatformTenants() {
   243→  return useQuery<{ tenants: PlatformTenant[] }, ApiError>({
   244→    queryKey: queryKeys.platformAdmin.tenants(),
   245→    queryFn: () =>
   246→      apiGet<{ tenants: PlatformTenant[] }>("/api/platform-admin/tenants"),
   247→  });
   248→}
   249→
   250→/**
   251→ * Fetch a single platform tenant by its slug.
   252→ *
   253→ * The query is disabled when `slug` is empty, preventing
   254→ * unnecessary requests before a tenant is selected.
   255→ *
   256→ * @param slug - Slug of the tenant to fetch.
   257→ */
   258→export function usePlatformTenant(slug: string) {
   259→  return useQuery<PlatformTenant, ApiError>({
   260→    queryKey: queryKeys.platformAdmin.tenantDetail(slug),
   261→    queryFn: () =>
   262→      apiGet<PlatformTenant>(`/api/platform-admin/tenants/${encodeURIComponent(slug)}`),
   263→    enabled: !!slug,
   264→  });
   265→}

   422→    onSuccess: () => {
   423→      void queryClient.invalidateQueries({

   446→/**
   447→ * Update an existing platform tenant.
   448→ *
   449→ * Accepts the tenant slug plus any updatable fields. On success
   450→ * both the tenants list and the specific tenant detail caches
   451→ * are invalidated so all views reflect the latest data.
   452→ *
   453→ * Variables shape: `{ slug: string, ...data }`
   454→ */
   455→export function useUpdatePlatformTenant() {
   456→  const queryClient = useQueryClient();
   457→
   458→  return useMutation<PlatformTenant, ApiError, UpdateTenantPayload>({
   459→    mutationFn: ({ slug, ...data }) =>
   460→      apiPatch<Record<string, unknown>, PlatformTenant>(
   461→        `/api/platform-admin/tenants/${encodeURIComponent(slug)}`,
   462→        data,
   463→      ),
   464→    onSuccess: (_data, variables) => {
   465→      void queryClient.invalidateQueries({
   466→        queryKey: queryKeys.platformAdmin.tenants(),
   467→      });
   468→      void queryClient.invalidateQueries({
   469→        queryKey: queryKeys.platformAdmin.tenantDetail(variables.slug),
   470→      });
   471→    },
   472→  });
   473→}
   474→
   475→/**
   476→ * Delete a platform tenant by slug.
   477→ *
   478→ * On success the tenants list is invalidated and the specific
   479→ * tenant detail cache entry is removed so stale data does not
   480→ * persist for a deleted tenant.
   481→ *
   482→ * Variables shape: `{ slug: string }`
   483→ */
   484→export function useDeletePlatformTenant() {
   485→  const queryClient = useQueryClient();
   486→
   487→  return useMutation<void, ApiError, { slug: string }>({
   488→    mutationFn: ({ slug }) =>
   489→      apiDelete<void>(`/api/platform-admin/tenants/${encodeURIComponent(slug)}`),
   490→    onSuccess: (_data, variables) => {
   491→      void queryClient.invalidateQueries({
   492→        queryKey: queryKeys.platformAdmin.tenants(),
   493→      });
   494→      void queryClient.removeQueries({
   495→        queryKey: queryKeys.platformAdmin.tenantDetail(variables.slug),
   496→      });
   497→    },
   498→  });
   499→}
