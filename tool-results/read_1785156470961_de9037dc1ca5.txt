    94→function pct(arr: number[], p: number): number {
    95→  if (arr.length === 0) return 0;
    96→  const sorted = [...arr].sort((a, b) => a - b);
    97→  const idx = Math.min(sorted.length - 1, Math.floor((p / 100) * sorted.length));
    98→  return sorted[idx];
    99→}
   100→
   101→function mean(arr: number[]): number {
   102→  if (arr.length === 0) return 0;
   103→  return arr.reduce((s, v) => s + v, 0) / arr.length;
   104→}

   106→function findServerPid(): number | null {
   107→  try {
   108→    // Find the PID listening on port 3000 (IPv4 or IPv6)
   109→    const out = execSync("ss -ltnp 2>/dev/null || netstat -ltnp 2>/dev/null", {
   110→      encoding: "utf8",
   111→      stdio: ["pipe", "pipe", "ignore"],
   112→    });
   113→    for (const line of out.split("\n")) {
   114→      if (line.includes(":3000")) {
   115→        const m = line.match(/pid=(\d+)/);
   116→        if (m) return parseInt(m[1], 10);
   117→      }
   118→    }
   119→  } catch {}
   120→  // Fallback: pgrep for next dev / next-server
   121→  try {
   122→    const out = execSync("pgrep -f 'next-server|next dev|next start' 2>/dev/null", {
   123→      encoding: "utf8",
   124→      stdio: ["pipe", "pipe", "ignore"],
   125→    });
   126→    const pid = parseInt(out.split("\n")[0], 10);
   127→    if (!Number.isNaN(pid)) return pid;
   128→  } catch {}
   129→  return null;
   130→}

   134→function readProcStat(pid: number): { cpuTicks: number; rssKB: number } | null {
   135→  try {
   136→    const stat = readFileSync(`/proc/${pid}/stat`, "utf8");
   137→    // /proc/[pid]/stat has the comm field in parens which may contain spaces —
   138→    // find the closing paren and parse the rest from there.
   139→    const closeParen = stat.lastIndexOf(")");
   140→    const rest = stat.slice(closeParen + 2).split(" ");
   141→    // After state (rest[0]), utime=rest[11], stime=rest[12], rss=rest[21]
   142→    // (relative to fields after the closing paren).
   143→    const utime = parseInt(rest[11], 10);
   144→    const stime = parseInt(rest[12], 10);
   145→    const rssPages = parseInt(rest[21], 10);
   146→    return { cpuTicks: utime + stime, rssKB: rssPages * 4 };
   147→  } catch {
   148→    return null;
   149→  }
   150→}

   152→function sampleServer(): { cpuTicks: number; rssKB: number; ts: number } {
   153→  if (!SERVER_PID) return { cpuTicks: 0, rssKB: 0, ts: Date.now() };
   154→  const s = readProcStat(SERVER_PID);
   155→  return { cpuTicks: s?.cpuTicks ?? 0, rssKB: s?.rssKB ?? 0, ts: Date.now() };
   156→}

   182→// ─── HTTP ────────────────────────────────────────────────────────────────────
   183→async function timedFetch(
   184→  method: string,
   185→  path: string,
   186→  body?: unknown,
   187→  timeoutMs = REQUEST_TIMEOUT_MS,
   188→): Promise<{ status: number; ok: boolean; latencyMs: number; errorType: string | null }> {
   189→  const t0 = performance.now();
   190→  const controller = new AbortController();
   191→  const timer = setTimeout(() => controller.abort(), timeoutMs);
   192→  try {
   193→    const init: RequestInit = {
   194→      method,
   195→      headers: {
   196→        "Content-Type": "application/json",
   197→        Cookie: `inv_token=${jwt}`,
   198→      },
   199→      signal: controller.signal,
   200→    };
   201→    if (body !== undefined) init.body = JSON.stringify(body);
   202→    const res = await fetch(`${BASE}${path}`, init);
   203→    const latencyMs = performance.now() - t0;
   204→    // Drain the body so the connection can be reused
   205→    try {
   206→      await res.text();
   207→    } catch {}
   208→    const ok = res.status >= 200 && res.status < 300;
   209→    let errorType: string | null = null;
   210→    if (!ok) errorType = `HTTP_${res.status}`;
   211→    return { status: res.status, ok, latencyMs, errorType };
   212→  } catch (err) {
   213→    const latencyMs = performance.now() - t0;
   214→    let errorType = "NETWORK_ERROR";
   215→    if (err instanceof Error) {
   216→      if (err.name === "AbortError") errorType = "TIMEOUT";
   217→      else errorType = err.message.slice(0, 80);
   218→    }
   219→    return { status: 0, ok: false, latencyMs, errorType };
   220→  } finally {
   221→    clearTimeout(timer);
   222→  }
   223→}

   293→// ─── Concurrency test runner ─────────────────────────────────────────────────
   294→async function runConcurrencyTest(
   295→  endpoint: string,
   296→  method: string,
   297→  pathBuilder: (n: number) => string,
   298→  bodyBuilder: ((n: number) => unknown) | null,
   299→  concurrency: number,
   300→  totalRequests: number,
   301→  notes: string[] = [],
   302→  timeoutMs: number = HTTP_REQUEST_TIMEOUT_MS,
   303→): Promise<TestResult> {
   304→  console.log(
   305→    `  ▶ ${endpoint} @ c=${concurrency} (n=${totalRequests}) ...`,
   306→  );
   307→
   308→  // Pre-sample
   309→  const serverBefore = sampleServer();
   310→  const runnerBefore = sampleRunner();
   311→  const eld = monitorEventLoopDelay();
   312→  eld.reset();
   313→  eld.enable();
   314→
   315→  const samples: Sample[] = [];
   316→  const t0 = performance.now();
   317→  let inflight = 0;
   318→  let nextIdx = 0;
   319→  const queue: Promise<void>[] = [];
   320→
   321→  async function worker() {
   322→    while (true) {
   323→      const i = nextIdx++;
   324→      if (i >= totalRequests) return;
   325→      inflight++;
   326→      const path = pathBuilder(i);
   327→      const body = bodyBuilder ? bodyBuilder(i) : undefined;
   328→      const r = await timedFetch(method, path, body, timeoutMs);
   329→      samples.push({
   330→        status: r.status,
   331→        ok: r.ok,
   332→        latencyMs: r.latencyMs,
   333→        errorType: r.errorType,
   334→      });
   335→      inflight--;
   336→    }
   337→  }
   338→
   339→  for (let w = 0; w < concurrency; w++) queue.push(worker());
   340→  await Promise.all(queue);
   341→
   342→  const wallMs = performance.now() - t0;
   343→  eld.disable();
   344→
   345→  const serverAfter = sampleServer();
   346→  const runnerAfter = sampleRunner();
   347→
   348→  const latencies = samples.map((s) => s.latencyMs);
   349→  const successCount = samples.filter((s) => s.ok).length;
   350→  const errorCount = samples.length - successCount;
   351→  const errorTypes: Record<string, number> = {};
   352→  for (const s of samples) {
   353→    if (s.errorType) errorTypes[s.errorType] = (errorTypes[s.errorType] || 0) + 1;
   354→  }
   355→
   356→  // Server CPU% over the test window (cpuTicks are in 1/100s of a second on Linux)
   357→  const wallSec = wallMs / 1000;
   358→  const cpuDeltaTicks = serverAfter.cpuTicks - serverBefore.cpuTicks;
   359→  const cpuDeltaSec = cpuDeltaTicks / 100; // 100 Hz clock
   360→  const serverCpuPct = wallSec > 0 ? (cpuDeltaSec / wallSec) * 100 : 0;
   361→  const serverRssDeltaMB = (serverAfter.rssKB - serverBefore.rssKB) / 1024;
   362→  const runnerRssDeltaMB = runnerAfter.rssMB - runnerBefore.rssMB;
   363→
   364→  const result: TestResult = {
   365→    endpoint,
   366→    method,
   367→    concurrency,
   368→    totalRequests,
   369→    successCount,
   370→    errorCount,
   371→    errorTypes,
   372→    latenciesMs: latencies,
   373→    p50: pct(latencies, 50),
   374→    p90: pct(latencies, 90),
   375→    p95: pct(latencies, 95),
   376→    p99: pct(latencies, 99),
   377→    mean: mean(latencies),
   378→    min: latencies.length ? Math.min(...latencies) : 0,
   379→    max: latencies.length ? Math.max(...latencies) : 0,
   380→    rps: wallMs > 0 ? (samples.length / wallMs) * 1000 : 0,
   381→    wallMs,
   382→    serverCpuPct,
   383→    serverRssDeltaMB,
   384→    runnerRssDeltaMB,
   385→    eventLoopLagP99Ms: eld.percentile(99) / 1e6, // ns → ms
   386→    eventLoopLagMaxMs: eld.max / 1e6,
   387→    notes,
   388→  };
   389→
   390→  console.log(
   391→    `    ✓ ok=${successCount}/${totalRequests} p50=${result.p50.toFixed(0)}ms p95=${result.p95.toFixed(0)}ms p99=${result.p99.toFixed(0)}ms rps=${result.rps.toFixed(1)} cpu=${serverCpuPct.toFixed(0)}% err=${errorCount}`,
   392→  );
   393→  return result;
   394→}

   396→// ─── Batch invoice creation ──────────────────────────────────────────────────
   397→async function runBatchTest(size: number): Promise<BatchResult> {
   398→  console.log(`\n  ▶ BATCH create ${size} invoices (c=${BATCH_CONCURRENCY}) ...`);
   399→
   400→  // C5 FIX: Cannot measure PostgreSQL DB size from file path like SQLite
   401→  const dbSizeBefore = 0;
   402→
   403→  // Measure list latency before (single sample, fresh)
   404→  const listBefore = await timedFetch(
   405→    "GET",
   406→    `/api/invoices?companySlug=${COMPANY_SLUG}&limit=500`,
   407→    undefined,
   408→    HTTP_REQUEST_TIMEOUT_MS,
   409→  );
   410→  // Take 5 samples for percentile
   411→  const listSamplesBefore: number[] = [listBefore.latencyMs];
   412→  for (let i = 0; i < 4; i++) {
   413→    const r = await timedFetch(
   414→      "GET",
   415→      `/api/invoices?companySlug=${COMPANY_SLUG}&limit=500`,
   416→      undefined,
   417→      HTTP_REQUEST_TIMEOUT_MS,
   418→    );
   419→    listSamplesBefore.push(r.latencyMs);
   420→  }
   421→
   422→  // Create `size` invoices with concurrency BATCH_CONCURRENCY
   423→  const t0 = performance.now();
   424→  let successCount = 0;
   425→  let errorCount = 0;
   426→  let nextIdx = 0;
   427→  const ts = Date.now();
   428→  async function worker() {
   429→    while (true) {
   430→      const i = nextIdx++;
   431→      if (i >= size) return;
   432→      const body = {
   433→        invoiceNumber: `BATCH-${ts}-${size}-${i}`,
   434→        clientName: `Batch Client ${i}`,
   435→        issueDate: "2026-01-15",
   436→        dueDate: "2026-02-15",
   437→        lineItems: [{ description: `Item ${i}`, qty: 2, price: "50" }],
   438→        subtotal: "100",
   439→        taxRate: "0",
   440→        taxAmount: "0",
   441→        total: "100",
   442→        status: "draft",
   443→        companySlug: COMPANY_SLUG,
   444→      };
   445→      const r = await timedFetch("POST", "/api/invoices", body, HTTP_REQUEST_TIMEOUT_MS);
   446→      if (r.ok) successCount++;
   447→      else errorCount++;
   448→    }
   449→  }
   450→  const workers: Promise<void>[] = [];
   451→  for (let w = 0; w < BATCH_CONCURRENCY; w++) workers.push(worker());
   452→  await Promise.all(workers);
   453→  const wallMs = performance.now() - t0;
   454→
   455→  // Wait a brief moment for DB to flush (C5 FIX: updated for PostgreSQL)
   456→  await new Promise((r) => setTimeout(r, 500));
   457→
   458→  // Measure list latency after (5 samples)
   459→  const listSamplesAfter: number[] = [];
   460→  for (let i = 0; i < 5; i++) {
   461→    const r = await timedFetch(
   462→      "GET",
   463→      `/api/invoices?companySlug=${COMPANY_SLUG}&limit=500`,
   464→      undefined,
   465→      HTTP_REQUEST_TIMEOUT_MS,
   466→    );
   467→    listSamplesAfter.push(r.latencyMs);
   468→  }
   469→
   470→  // C5 FIX: Cannot measure PostgreSQL DB size from file path like SQLite
   471→  const dbSizeAfter = 0;
   472→
   473→  const result: BatchResult = {
   474→    size,
   475→    wallMs,
   476→    throughputInvoicesPerSec: wallMs > 0 ? (successCount / wallMs) * 1000 : 0,
   477→    successCount,
   478→    errorCount,
   479→    listLatencyMsBefore: listBefore.latencyMs,
   480→    listLatencyMsAfter: listSamplesAfter[0],
   481→    dbSizeBytesBefore: dbSizeBefore,
   482→    dbSizeBytesAfter: dbSizeAfter,
   483→    listP50: pct(listSamplesAfter, 50),
   484→    listP95: pct(listSamplesAfter, 95),
   485→  };
   486→
   487→  console.log(
   488→    `    ✓ ok=${successCount}/${size} wall=${(wallMs / 1000).toFixed(1)}s throughput=${result.throughputInvoicesPerSec.toFixed(1)} inv/s listAfterP50=${result.listP50.toFixed(0)}ms dbGrowth=${((dbSizeAfter - dbSizeBefore) / 1024).toFixed(0)}KB`,
   489→  );
   490→  return result;
   491→}
