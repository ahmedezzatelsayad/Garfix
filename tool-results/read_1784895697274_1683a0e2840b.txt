    37→/* ─── Main Component ───────────────────────────────────────────────────────── */
    38→export function ArApView() {
    39→  const { activeCompany } = useBrand();
    40→  const [tab, setTab] = useState<Tab>("aging");
    41→  const [agingData, setAgingData] = useState<AgingSummary | null>(null);
    42→  const [agingDirection, setAgingDirection] = useState<Direction>("receivable");
    43→  const [clientStatement, setClientStatement] = useState<ClientStatement | null>(null);
    44→  const [supplierStatement, setSupplierStatement] = useState<ClientStatement | null>(null);
    45→  const [pdcs, setPdcs] = useState<PDC[]>([]);
    46→  const [installments, setInstallments] = useState<Installment[]>([]);
    47→  const [loading, setLoading] = useState(false);
    48→  const [showForm, setShowForm] = useState(false);
    49→
    50→  const slug = activeCompany ? encodeURIComponent(activeCompany.slug) : "";
    51→
    52→  const loadAging = useCallback(async () => {
    53→    if (!activeCompany) return;
    54→    setLoading(true);
    55→    try {
    56→      const res = await authedFetch(`/api/accounting/aging?companySlug=${slug}&direction=${agingDirection}`);
    57→      if (res.ok) { const d = await res.json(); setAgingData(d.summary || null); }
    58→      else setAgingData(null);
    59→    } catch { setAgingData(null); }
    60→    finally { setLoading(false); }
    61→  }, [activeCompany, slug, agingDirection]);
    62→
    63→  const loadPDCs = useCallback(async () => {
    64→    if (!activeCompany) return;
    65→    setLoading(true);
    66→    try {
    67→      const res = await authedFetch(`/api/accounting/post-dated-checks?companySlug=${slug}`);
    68→      if (res.ok) { const d = await res.json(); setPdcs(d.checks || []); }
    69→      else setPdcs([]);
    70→    } catch { setPdcs([]); }
    71→    finally { setLoading(false); }
    72→  }, [activeCompany, slug]);
    73→
    74→  const loadInstallments = useCallback(async () => {
    75→    if (!activeCompany) return;
    76→    setLoading(true);
    77→    try {
    78→      const res = await authedFetch(`/api/accounting/installments?companySlug=${slug}`);
    79→      if (res.ok) { const d = await res.json(); setInstallments(d.installments || []); }
    80→      else setInstallments([]);
    81→    } catch { setInstallments([]); }
    82→    finally { setLoading(false); }
    83→  }, [activeCompany, slug]);
    84→
    85→  useEffect(() => {
    86→    // eslint-disable-next-line react-hooks/set-state-in-effect
    87→    if (tab === "aging") loadAging();
    88→    if (tab === "pdc") loadPDCs();
    89→    if (tab === "installments") loadInstallments();
    90→  }, [tab, loadAging, loadPDCs, loadInstallments]);
    91→
    92→  if (!activeCompany) return <div className="p-12 text-center text-muted-foreground">اختر شركة</div>;
    93→
    94→  const tabs: Array<{ key: Tab; label: string; icon: React.ComponentType<{ size?: number }> }> = [
    95→    { key: "aging", label: "تقادم الذمم", icon: ArrowUpDown },
    96→    { key: "client-statement", label: "كشف حساب عميل", icon: FileText },
    97→    { key: "supplier-statement", label: "كشف حساب مورد", icon: FileText },
    98→    { key: "pdc", label: "شيكات مؤجلة", icon: Banknote },
    99→    { key: "installments", label: "التقسيط", icon: CalendarDays },
   100→  ];
   101→
   102→  return (
   103→    <div className="flex flex-col gap-4">
   104→      <div className="flex flex-wrap justify-between items-center gap-3">
   105→        <div>
   106→          <h1 className="text-2xl font-extrabold flex items-center gap-2"><ArrowUpDown size={20} /> الذمم</h1>
   107→          <p className="text-[13px] text-muted-foreground">{activeCompany.nameAr || activeCompany.name}</p>
   108→        </div>
   109→        {(tab === "pdc" || tab === "installments") && !showForm && (
   110→          <button onClick={() => setShowForm(true)} className="inline-flex items-center gap-1.5 py-2.5 px-[18px] rounded-[10px] bg-primary text-primary-foreground border-none text-[13px] font-bold cursor-pointer"><Plus size={16} /> إضافة</button>
   111→        )}
   112→      </div>
   113→      <div className="flex gap-1.5 flex-wrap">
   114→        {tabs.map((t) => { const Icon = t.icon; return (
   115→          <button key={t.key} onClick={() => { setTab(t.key); setShowForm(false); }} className={cn("py-2 px-4 rounded-[10px] border border-border text-[12px] font-bold cursor-pointer inline-flex items-center gap-1.5", tab === t.key ? "bg-primary text-primary-foreground" : "bg-card text-muted-foreground")}>
   116→            <Icon size={14} /> {t.label}
   117→          </button>
   118→        ); })}
   119→      </div>
   120→
   121→      {loading ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : tab === "aging" ? (
   122→        <AgingReportView data={agingData} direction={agingDirection} onDirectionChange={setAgingDirection} />
   123→      ) : tab === "client-statement" ? (
   124→        <StatementView type="client" company={activeCompany} data={clientStatement} setData={setClientStatement} />
   125→      ) : tab === "supplier-statement" ? (
   126→        <StatementView type="supplier" company={activeCompany} data={supplierStatement} setData={setSupplierStatement} />
   127→      ) : tab === "pdc" ? (
   128→        showForm ? <PDCForm company={activeCompany} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadPDCs(); }} /> : <PDCList pdcs={pdcs} company={activeCompany} onRefresh={loadPDCs} />
   129→      ) : tab === "installments" ? (
   130→        showForm ? <InstallmentForm company={activeCompany} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadInstallments(); }} /> : <InstallmentList installments={installments} />
   131→      ) : null}
   132→    </div>
   133→  );
   134→}

   208→/* ─── Client/Supplier Statement ────────────────────────────────────────────── */
   209→function StatementView({ type, company, data, setData }: { type: "client" | "supplier"; company: { slug: string }; data: ClientStatement | null; setData: (d: ClientStatement | null) => void }) {
   210→  const [selectedId, setSelectedId] = useState<number | null>(null);
   211→  const [contacts, setContacts] = useState<Contact[]>([]);
   212→  const [loadingStatement, setLoadingStatement] = useState(false);
   213→
   214→  useEffect(() => {
   215→    const endpoint = type === "client" ? "/api/clients" : "/api/suppliers";
   216→    authedFetch(`${endpoint}?companySlug=${encodeURIComponent(company.slug)}`)
   217→      .then(r => r.ok ? r.json() : { clients: [], suppliers: [] })
   218→      .then(d => setContacts(type === "client" ? (d.clients || []) : (d.suppliers || [])))
   219→      .catch(() => setContacts([]));
   220→  }, [type, company.slug]);
   221→
   222→  const loadStatement = async () => {
   223→    if (!selectedId) { toast.error("اختر " + (type === "client" ? "عميل" : "مورد")); return; }
   224→    setLoadingStatement(true);
   225→    try {
   226→      const endpoint = type === "client" ? "/api/accounting/client-statement" : "/api/accounting/supplier-statement";
   227→      const paramName = type === "client" ? "clientId" : "supplierId";
   228→      const res = await authedFetch(`${endpoint}?companySlug=${encodeURIComponent(company.slug)}&${paramName}=${selectedId}`);
   229→      if (res.ok) { setData(await res.json()); }
   230→      else { const e = await res.json().catch(() => ({})); toast.error(e.error || "تعذّر تحميل كشف الحساب"); setData(null); }
   231→    } catch { toast.error("خطأ في الاتصال"); setData(null); }
   232→    finally { setLoadingStatement(false); }
   233→  };
   234→
   235→  const totalDebit = data?.lines.reduce((s, l) => s + l.debit, 0) || 0;
   236→  const totalCredit = data?.lines.reduce((s, l) => s + l.credit, 0) || 0;
   237→
   238→  return (
   239→    <div className="flex flex-col gap-4">
   240→      <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex gap-3 items-center flex-wrap">
   241→        <div className="flex items-center gap-1.5">
   242→          <label className="text-[11px] font-bold text-muted-foreground">{type === "client" ? "العميل" : "المورد"}</label>
   243→          <select value={selectedId ?? ""} onChange={(e) => setSelectedId(e.target.value ? Number(e.target.value) : null)} className={cn(inputStyle, "w-auto min-w-[200px]")} disabled={loadingStatement}>
   244→            <option value="">— اختر —</option>
   245→            {contacts.map((c) => <option key={c.id} value={c.id}>{c.name}</option>)}
   246→          </select>
   247→        </div>
   248→        <button onClick={loadStatement} disabled={!selectedId || loadingStatement} className="py-2 px-4 rounded-sm bg-accent text-accent-foreground border border-border text-[12px] font-bold cursor-pointer disabled:opacity-70 inline-flex items-center gap-1.5">
   249→          <RefreshCw size={12} className={loadingStatement ? "animate-spin" : ""} /> {loadingStatement ? "جارٍ…" : "عرض"}
   250→        </button>
   251→      </div>
   252→
   253→      {loadingStatement ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : !data ? (
   254→        <div className="bg-card rounded-[14px] border border-border p-12 text-center text-muted-foreground">
   255→          <FileText size={36} className="opacity-30 mb-2" />اختر {type === "client" ? "عميل" : "مورد"} ثم اضغط عرض
   256→        </div>
   257→      ) : (
   258→        <div className="flex flex-col gap-4">
   259→          <div className="grid grid-cols-[repeat(auto-fit,minmax(140px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(180px,1fr))] gap-3">
   260→            <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   261→              <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-violet-500/20 text-violet-500"><FileText size={18} /></div>
   262→              <div><div className="text-[11px] text-muted-foreground">{data.clientName}</div><div className="text-[13px] font-bold">رصيد افتتاحي: {fmt(data.openingBalance)}</div></div>
   263→            </div>
   264→            <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   265→              <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-emerald-500/20 text-emerald-500"><TrendingUp size={18} /></div>
   266→              <div><div className="text-[11px] text-muted-foreground">إجمالي مدين</div><div className="text-lg font-extrabold [direction:ltr] text-end">{fmt(totalDebit)}</div></div>
   267→            </div>
   268→            <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   269→              <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-red-500/20 text-red-500"><TrendingDown size={18} /></div>
   270→              <div><div className="text-[11px] text-muted-foreground">إجمالي دائن</div><div className="text-lg font-extrabold [direction:ltr] text-end">{fmt(totalCredit)}</div></div>
   271→            </div>
   272→            <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   273→              <div className={cn("w-10 h-10 rounded-sm flex items-center justify-center", data.closingBalance >= 0 ? "bg-violet-500/20 text-violet-500" : "bg-red-500/20 text-red-500")}><CheckCircle2 size={18} /></div>
   274→              <div><div className="text-[11px] text-muted-foreground">رصيد إقفالي</div><div className={cn("text-lg font-extrabold [direction:ltr] text-end", data.closingBalance >= 0 ? "text-violet-500" : "text-red-500")}>{fmt(data.closingBalance)}</div></div>
   275→            </div>
   276→          </div>
   277→
   278→          <div className="bg-card rounded-[14px] border border-border overflow-hidden">
   279→            {data.lines.length === 0 ? <Empty label="حركات" /> : (
   280→              <div className="overflow-x-auto garfix-scroll">
   281→                <table className="w-full border-collapse">
   282→                  <thead><tr className="border-b border-border bg-muted">
   283→                    <th scope="col" className={thStyle}>التاريخ</th><th scope="col" className={thStyle}>النوع</th><th scope="col" className={thStyle}>المرجع</th>
   284→                    <th scope="col" className={cn(thStyle, "text-end")}>مدين</th><th scope="col" className={cn(thStyle, "text-end")}>دائن</th><th scope="col" className={cn(thStyle, "text-end")}>الرصيد</th>
   285→                  </tr></thead>
   286→                  <tbody>
   287→                    {data.lines.map((l, i) => (
   288→                      <tr key={i} className="border-b border-border">
   289→                        <td className={tdStyle} dir="ltr">{l.date}</td>
   290→                        <td className={tdStyle}>{l.type}</td>
   291→                        <td className={cn(tdStyle, "font-mono")}>{l.reference}</td>
   292→                        <td className={cn(tdStyle, "[direction:ltr] text-end", l.debit > 0 ? "text-violet-500" : "")}>{l.debit ? fmt(l.debit) : "—"}</td>
   293→                        <td className={cn(tdStyle, "[direction:ltr] text-end", l.credit > 0 ? "text-red-500" : "")}>{l.credit ? fmt(l.credit) : "—"}</td>
   294→                        <td className={cn(tdStyle, "[direction:ltr] text-end font-bold", l.balance >= 0 ? "text-violet-500" : "text-red-500")}>{fmt(l.balance)}</td>
   295→                      </tr>
   296→                    ))}
   297→                  </tbody>
   298→                  <tfoot>
   299→                    <tr className="border-t-2 border-border bg-muted font-extrabold">
   300→                      <td className={cn(tdStyle, "font-extrabold")} colSpan={3}>رصيد إقفالي</td>
   301→                      <td className={cn(cn(tdStyle, "[direction:ltr] text-end font-extrabold"), "text-violet-500")}>{fmt(totalDebit)}</td>
   302→                      <td className={cn(cn(tdStyle, "[direction:ltr] text-end font-extrabold"), "text-red-500")}>{fmt(totalCredit)}</td>
   303→                      <td className={cn(tdStyle, "[direction:ltr] text-end font-extrabold", data.closingBalance >= 0 ? "text-violet-500" : "text-red-500")}>{fmt(data.closingBalance)}</td>
   304→                    </tr>
   305→                  </tfoot>
   306→                </table>
   307→              </div>
   308→            )}
   309→          </div>
   310→        </div>
   311→      )}
   312→    </div>
   313→  );
   314→}

   316→/* ─── PDC List ──────────────────────────────────────────────────────────────── */
   317→function PDCList({ pdcs, company, onRefresh }: { pdcs: PDC[]; company: { slug: string }; onRefresh: () => void }) {
   318→  const [actionId, setActionId] = useState<number | null>(null);

   343→  const totalAmount = pdcs.reduce((s, p) => s + p.amount, 0);

   352→        </div>

   355→          <div><div className="text-[11px] text-muted-foreground">المعلّقة</div><div className="text-lg font-extrabold [direction:ltr] text-end text-amber-500">{fmt(pendingTotal)}</div></div>
   356→        </div>
   357→      </div>
   358→      <div className="bg-card rounded-[14px] border border-border overflow-hidden">
   359→        {pdcs.length === 0 ? <Empty label="شيكات مؤجلة" /> : (
   360→          <div className="overflow-x-auto garfix-scroll">
   361→            <table className="w-full border-collapse">
   362→              <thead><tr className="border-b border-border bg-muted">
   363→                <th scope="col" className={thStyle}>رقم الشيك</th><th scope="col" className={thStyle}>البنك</th><th scope="col" className={thStyle}>النوع</th>
   364→                <th scope="col" className={thStyle}>الاسم</th><th scope="col" className={cn(thStyle, "text-end")}>المبلغ</th><th scope="col" className={thStyle}>تاريخ الاستحقاق</th>
   365→                <th scope="col" className={thStyle}>الحالة</th><th scope="col" className={thStyle}>إجراء</th>
   366→              </tr></thead>
   367→              <tbody>
   368→                {pdcs.map((p) => {
   369→                  const sc = statusBadge(p.status);
   370→                  const name = p.direction === "receivable" ? p.clientName : p.supplierName;
   371→                  return (
   372→                    <tr key={p.id} className="border-b border-border">
   373→                      <td className={cn(tdStyle, "font-mono")} dir="ltr">{p.checkNumber}</td>
   374→                      <td className={tdStyle}>{p.bankName}</td>
   375→                      <td className={tdStyle}><span className={cn("py-0.5 px-2.5 rounded-[12px] text-[11px] font-bold", p.direction === "receivable" ? "bg-emerald-500/15 text-emerald-500" : "bg-red-500/15 text-red-500")}>{p.direction === "receivable" ? "مقبوض" : "مدفوع"}</span></td>

   399→/* ─── PDC Form ──────────────────────────────────────────────────────────────── */
   400→function PDCForm({ company, onClose, onSaved }: { company: { slug: string }; onClose: () => void; onSaved: () => void }) {
   401→  const [checkNumber, setCheckNumber] = useState("");
   402→  const [bankName, setBankName] = useState("");
   403→  const [amount, setAmount] = useState(0);
   404→  const [dueDate, setDueDate] = useState("");
   405→  const [direction, setDirection] = useState<"receivable" | "payable">("receivable");
   406→  const [clientName, setClientName] = useState("");
   407→  const [supplierName, setSupplierName] = useState("");
   408→  const [saving, setSaving] = useState(false);
   409→
   410→  const submit = async () => {
   411→    if (!checkNumber || !bankName || !dueDate || amount <= 0) { toast.error("جميع الحقول مطلوبة"); return; }
   412→    setSaving(true);
   413→    try {
   414→      const res = await authedFetch("/api/accounting/post-dated-checks", {
   415→        method: "POST", headers: { "Content-Type": "application/json" },
   416→        body: JSON.stringify({ checkNumber, bankName, amount, dueDate, direction, clientName: direction === "receivable" ? clientName : undefined, supplierName: direction === "payable" ? supplierName : undefined, companySlug: company.slug }),
   417→      });
   418→      if (!res.ok) { const e = await res.json().catch(() => ({})); throw new Error(e.error || "Failed"); }
   419→      toast.success("تم إنشاء الشيك المؤجل");
   420→      onSaved();
   421→    } catch (err) { toast.error(err instanceof Error ? err.message : "خطأ"); }
   422→    finally { setSaving(false); }
   423→  };
   424→
   425→  return (
   426→    <div className="bg-card rounded-[14px] border border-border p-5 flex flex-col gap-3.5">
   427→      <h3 className="text-[15px] font-bold flex items-center gap-2"><Banknote size={16} /> شيك مؤجل جديد</h3>

   429→        <div><label className={labelStyle}>رقم الشيك *</label><input value={checkNumber} onChange={(e) => setCheckNumber(e.target.value)} className={inputStyle} dir="ltr" /></div>
   430→        <div><label className={labelStyle}>البنك *</label><input value={bankName} onChange={(e) => setBankName(e.target.value)} className={inputStyle} /></div>
   431→        <div><label className={labelStyle}>المبلغ *</label><input type="number" value={amount} onChange={(e) => setAmount(Number(e.target.value))} className={inputStyle} dir="ltr" /></div>
   432→        <div><label className={labelStyle}>تاريخ الاستحقاق *</label><input type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} className={inputStyle} dir="ltr" /></div>
   433→        <div><label className={labelStyle}>النوع *</label>
   434→          <select value={direction} onChange={(e) => setDirection(e.target.value as "receivable" | "payable")} className={inputStyle}>
   435→            <option value="receivable">شيك مقبوض (مدين)</option><option value="payable">شيك مدفوع (دائن)</option>
   436→          </select>
   437→        </div>

   439→        {direction === "payable" && <div><label className={labelStyle}>اسم المورد</label><input value={supplierName} onChange={(e) => setSupplierName(e.target.value)} className={inputStyle} /></div>}
