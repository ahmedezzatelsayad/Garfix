   329→      apiGet<AccountingDashboardData>(

   332→  });

   415→/**
   416→ * Fetch audit log entries with advanced filtering.
   417→ *
   418→ * Supports filtering by action, company slug, user email, date range, and limit.
   419→ * Used by AuditView and EnhancedAuditView.
   420→ *
   421→ * @param params - Filter parameters for the audit log query.
   422→ */
   423→export interface AuditLogFilterParams {
   424→  action?: string;
   425→  companySlug?: string;
   426→  userEmail?: string;
   427→  startDate?: string;
   428→  endDate?: string;
   429→  limit?: number;
   430→}
   431→
   432→export function useAuditLogFiltered(params: AuditLogFilterParams) {
   433→  return useQuery<AuditLogListResponse, ApiError>({
   434→    queryKey: [...queryKeys.audit.all, "filtered", params] as const,
   435→    queryFn: () => {
   436→      const searchParams = new URLSearchParams();
   437→      if (params.action) searchParams.set("action", params.action);
   438→      if (params.companySlug) searchParams.set("companySlug", params.companySlug);
   439→      if (params.userEmail) searchParams.set("userEmail", params.userEmail);
   440→      if (params.startDate) searchParams.set("startDate", params.startDate);
   441→      if (params.endDate) searchParams.set("endDate", params.endDate);
   442→      if (params.limit) searchParams.set("limit", String(params.limit));
   443→      return apiGet<AuditLogListResponse>(`/api/audit?${searchParams.toString()}`);
   444→    },
   445→  });
   446→}

   488→/**
   489→ * Fetch the list of all companies the user has access to.
   490→ *
   491→ * `enabled: !!user` would be ideal but this hook is called from
   492→ * BrandProvider which lives above AuthProvider's user state. Instead
   493→ * we rely on the consumer (BrandProvider) to ignore 401s when user is
   494→ * null. The hook itself is unconditional so TanStack can cache the
   495→ * result for the next mount.
   496→ */
   497→export function useCompanies() {
   498→  return useQuery<CompanyListResponse, ApiError>({
   499→    queryKey: queryKeys.companies.lists(),
   500→    queryFn: () => apiGet<CompanyListResponse>("/api/companies"),
   501→    retry: (failureCount, error) => {
   502→      // Don't retry 401s — they mean the user isn't signed in yet.
   503→      // AuthContext will trigger a refetch indirectly via its own
   504→      // refresh() flow when the user logs in.
   505→      if (error?.status === 401) return false;
   506→      return failureCount < 1;
   507→    },
   508→  });
   509→}

   584→      void queryClient.invalidateQueries({
   585→        queryKey: queryKeys.companies.lists(),
   586→      });
   587→    },
   588→  });

   758→/**
   759→ * Fetch a filtered/generated report with type, date range, and optional format.
   760→ *
   761→ * Used by ReportsView for generating specific report types (sales, profit, etc.)
   762→ * within a date range. The `format` parameter defaults to JSON; when "csv" is
   763→ * specified the API returns a Blob instead — callers should use `authedFetch`
   764→ * directly for CSV downloads since TanStack Query expects JSON responses.
   765→ *
   766→ * @param params - Report filter parameters.
   767→ */
   768→export interface ReportFilterParams {
   769→  companySlug: string;
   770→  type: string;
   771→  from: string;
   772→  to: string;
   773→}
   774→
   775→export interface ReportResponse {
   776→  type: string;
   777→  companySlug: string;
   778→  dateRange: { from: string; to: string };
   779→  summary: Record<string, unknown>;
   780→  rows: Array<Record<string, unknown>>;
   781→  count: number;
   782→}
   783→
   784→export function useReportsFiltered(params: ReportFilterParams) {
   785→  return useQuery<ReportResponse, ApiError>({
   786→    queryKey: [...queryKeys.reports.all, "filtered", params] as const,
   787→    queryFn: () => {
   788→      const searchParams = new URLSearchParams({
   789→        companySlug: params.companySlug,
   790→        type: params.type,
   791→        from: params.from,
   792→        to: params.to,
   793→      });
   794→      return apiGet<ReportResponse>(`/api/reports?${searchParams.toString()}`);
   795→    },
   796→    enabled: !!params.companySlug && !!params.type,
   797→  });
   798→}

   879→// ─── Report CSV Download Hook ──────────────────────────────────────────────
   880→
   881→/**
   882→ * Download a report as a CSV file.
   883→ *
   884→ * Returns a Blob that the caller can convert to a download link.
   885→ * Since TanStack Query expects JSON responses by default, this
   886→ * mutation uses `apiDownloadBlob` to handle the binary response.
   887→ *
   888→ * This is a fire-and-forget style mutation — it does not invalidate
   889→ * any query caches automatically since it produces a file download,
   890→ * not a data change.
   891→ *
   892→ * @param variables - Object containing companySlug, report type, and date range.
   893→ */
   894→export function useDownloadReportCsv() {
   895→  return useMutation<
   896→    Blob,
   897→    ApiError,
   898→    { companySlug: string; type: string; from: string; to: string }
   899→  >({
   900→    mutationFn: ({ companySlug, type, from, to }) =>
   901→      apiDownloadBlob(
   902→        `/api/reports?companySlug=${encodeURIComponent(companySlug)}` +
   903→        `&type=${encodeURIComponent(type)}` +
   904→        `&from=${encodeURIComponent(from)}` +
   905→        `&to=${encodeURIComponent(to)}` +
   906→        `&format=csv`,
   907→      ),
   908→  });
