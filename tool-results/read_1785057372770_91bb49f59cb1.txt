   331→   331→export function createCustomRole(input: CustomRoleInput): RoleDefinition {
   332→   332→  // Validate parent role exists
   333→   333→  const parent = ROLE_DEFINITIONS[input.inheritsFrom] || customRoles.get(input.inheritsFrom);
   334→   334→  if (!parent) {
   335→   335→    throw new Error(`Parent role "${input.inheritsFrom}" does not exist`);
   336→   336→  }
   337→   337→  // Validate no circular inheritance
   338→   338→  if (wouldCreateCircularInheritance(input.id, input.inheritsFrom)) {
   339→   339→    throw new Error("Circular inheritance detected");
   340→   340→  }
   341→   341→  // Validate resource names
   342→   342→  for (const rp of input.resourcePermissions) {
   343→   343→    if (!AVAILABLE_RESOURCES.includes(rp.resource)) {
   344→   344→      throw new Error(`Unknown resource: "${rp.resource}"`);
   345→   345→    }
   346→   346→  }
   347→   347→
   348→   348→  const role: RoleDefinition = {
   349→   349→    id: input.id,
   350→   350→    label: input.label,
   351→   351→    labelAr: input.labelAr,
   352→   352→    inheritsFrom: input.inheritsFrom,
   353→   353→    resourcePermissions: input.resourcePermissions,
   354→   354→    flatOverrides: input.flatOverrides,
   355→   355→    timeRestrictions: input.timeRestrictions,
   356→   356→    isBuiltIn: false,
   357→   357→    color: "#6366f1",
   358→   358→  };
   359→   359→  customRoles.set(input.id, role);
   360→   360→  return role;
   361→   361→}

   363→   363→export function deleteCustomRole(id: string): boolean {
   364→   364→  const role = customRoles.get(id);
   365→   365→  if (!role) return false;
   366→   366→  if (role.isBuiltIn) throw new Error("Cannot delete built-in roles");
   367→   367→  // Check no other custom role inherits from this one
   368→   368→  const allCustom = Array.from(customRoles.entries());
   369→   369→  for (const [cid, crole] of allCustom) {
   370→   370→    if (crole.inheritsFrom === id) {
   371→   371→      throw new Error(`Role "${cid}" inherits from "${id}" — delete or reassign first`);
   372→   372→    }
   373→   373→  }
   374→   374→  customRoles.delete(id);
   375→   375→  return true;
   376→   376→}

   378→   378→export function updateCustomRole(id: string, updates: Partial<CustomRoleInput>): RoleDefinition | undefined {
   379→   379→  const existing = customRoles.get(id);
   380→   380→  if (!existing) return undefined;
   381→   381→  if (existing.isBuiltIn) throw new Error("Cannot modify built-in roles");
   382→   382→
   383→   383→  const updated: RoleDefinition = {
   384→   384→    ...existing,
   385→   385→    ...updates,
   386→   386→    isBuiltIn: false,
   387→   387→  };
   388→   388→
   389→   389→  // Validate inheritance if changed
   390→   390→  if (updates.inheritsFrom) {
   391→   391→    const parent = ROLE_DEFINITIONS[updates.inheritsFrom] || customRoles.get(updates.inheritsFrom);
   392→   392→    if (!parent) throw new Error(`Parent role "${updates.inheritsFrom}" does not exist`);
   393→   393→    if (wouldCreateCircularInheritance(id, updates.inheritsFrom)) {
   394→   394→      throw new Error("Circular inheritance detected");
   395→   395→    }
   396→   396→  }
   397→   397→
   398→   398→  // Validate resources if changed
   399→   399→  if (updates.resourcePermissions) {
   400→   400→    for (const rp of updates.resourcePermissions) {
   401→   401→      if (!AVAILABLE_RESOURCES.includes(rp.resource)) {
   402→   402→        throw new Error(`Unknown resource: "${rp.resource}"`);
   403→   403→      }
   404→   404→    }
   405→   405→  }
   406→   406→
   407→   407→  customRoles.set(id, updated);
   408→   408→  return updated;
   409→   409→}

   411→   411→// ── Permission Hierarchy ─────────────────────────────────────────────────────
   412→   412→
   413→   413→/**
   414→   414→ * Walk the inheritance chain and collect all permissions from ancestor roles.
   415→   415→ * Permissions are merged: higher-level permissions override lower-level ones,
   416→   416→ * wider scopes override narrower ones for the same level.
   417→   417→ */
   418→   418→export function getInheritedPermissions(roleId: string): ResourcePermission[] {
   419→   419→  const chain = getInheritanceChain(roleId);
   420→   420→  const collected: ResourcePermission[] = [];
   421→   421→
   422→   422→  // Process from root to leaf — all permissions are collected.
   423→   423→  // The checkPermission function handles deduplication by finding
   424→   424→  // the best matching permission (preferring wider scope & higher level).
   425→   425→  for (const rid of chain) {
   426→   426→    const def = ROLE_DEFINITIONS[rid] || customRoles.get(rid);
   427→   427→    if (!def) continue;
   428→   428→    collected.push(...def.resourcePermissions);
   429→   429→  }
   430→   430→
   431→   431→  return collected;
   432→   432→}
   433→   433→
   434→   434→/**
   435→   435→ * Get the full inheritance chain from root to the given role.
   436→   436→ */
   437→   437→export function getInheritanceChain(roleId: string): string[] {
   438→   438→  const chain: string[] = [];
   439→   439→  let current: string | null = roleId;
   440→   440→
   441→   441→  while (current) {
   442→   442→    chain.unshift(current); // add at beginning (root first)
   443→   443→    const def = ROLE_DEFINITIONS[current] || customRoles.get(current);
   444→   444→    current = def?.inheritsFrom ?? null;
   445→   445→  }
   446→   446→
   447→   447→  return chain;
   448→   448→}

   450→   450→// ── Permission Checking ──────────────────────────────────────────────────────
   451→   451→
   452→   452→/**
   453→   453→ * Check if a user has a specific permission on a resource at a given scope.
   454→   454→ *
   455→   455→ * This is the core authorization check. It considers:
   456→   456→ *   1. The user's role and inherited permissions
   457→   457→ *   2. Custom per-user overrides (from flat permission system)
   458→   458→ *   3. Time-based restrictions
   459→   459→ *   4. Scope requirements
   460→   460→ *   5. Founder override (full access)
   461→   461→ */
   462→   462→export function checkPermission(
   463→   463→  userPerms: Record<string, number>,
   464→   464→  role: string,
   465→   465→  resource: string,
   466→   466→  action: string,
   467→   467→  scope: PermissionScope,
   468→   468→  isFounder = false,
   469→   469→  customRolePerms?: ResourcePermission[],
   470→   470→  timeRestrictions?: Record<string, TimeRestriction>,
   471→   471→): boolean {
   472→   472→  // Founder always has full access
   473→   473→  if (isFounder) return true;
   474→   474→
   475→   475→  // Admin role has company-wide access to everything
   476→   476→  if (role === "admin") {
   477→   477→    if (scope === PermissionScope.platform) return false; // only founder
   478→   478→    return true;
   479→   479→  }
   480→   480→
   481→   481→  // Collect all resource permissions for the role (inherited + custom)
   482→   482→  const roleDef = ROLE_DEFINITIONS[role] || customRoles.get(role);
   483→   483→  const inheritedPerms = roleDef ? getInheritedPermissions(role) : [];
   484→   484→  const allPerms = [...inheritedPerms, ...(customRolePerms || [])];
   485→   485→
   486→   486→  // Check time restrictions
   487→   487→  if (timeRestrictions?.[resource] || roleDef?.timeRestrictions?.[resource]) {
   488→   488→    const restriction = timeRestrictions?.[resource] || roleDef?.timeRestrictions?.[resource];
   489→   489→    if (restriction && !isWithinTimeRestriction(restriction)) {
   490→   490→      return false;
   491→   491→    }
   492→   492→  }
   493→   493→
   494→   494→  // Find matching permission
   495→   495→  const matchingPerm = allPerms.find((rp) => {
   496→   496→    if (rp.resource !== resource) return false;
   497→   497→    const permAction = levelToAction(rp.level);
   498→   498→    // Check if the granted action covers the requested action
   499→   499→    if (!actionCovers(permAction, action)) return false;
   500→   500→    // Check if the granted scope covers the requested scope
   501→   501→    if (!scopeCovers(rp.scope, scope)) return false;
   502→   502→    return true;
   503→   503→  });
   504→   504→
   505→   505→  if (matchingPerm) return true;
   506→   506→
   507→   507→  // Fall back to legacy flat permission system
   508→   508→  return checkLegacyPermission(userPerms, resource, action);
   509→   509→}
   510→   510→
   511→   511→/**
   512→   512→ * Map PermissionLevel to action string.
   513→   513→ */
   514→   514→function levelToAction(level: PermissionLevel): string {
   515→   515→  switch (level) {
   516→   516→    case PermissionLevel.read: return "read";
   517→   517→    case PermissionLevel.write: return "write";
   518→   518→    case PermissionLevel.approve: return "approve";
   519→   519→    case PermissionLevel.admin: return "admin";
   520→   520→    case PermissionLevel.delete: return "delete";
   521→   521→    case PermissionLevel.print: return "print";
   522→   522→    case PermissionLevel.export: return "export";
   523→   523→    case PermissionLevel.e_invoice: return "e_invoice";
   524→   524→    case PermissionLevel.bulk_input: return "bulk_input";
   525→   525→    case PermissionLevel.manage_prices: return "manage_prices";
   526→   526→    default: return "none";
   527→   527→  }
   528→   528→}
   529→   529→
   530→   530→/**
   531→   531→ * Check if a granted action covers a requested action.
   532→   532→ * Hierarchy: admin > approve > write > read > none
   533→   533→ * "admin" covers everything. "approve" covers write + read. etc.
   534→   534→ */
   535→   535→function actionCovers(granted: string, requested: string): boolean {
   536→   536→  // Special actions that don't follow the hierarchy (e.g., "print", "export")
   537→   537→  const SPECIAL_ACTIONS = ["print", "export", "e_invoice", "bulk_input", "manage_prices",
   538→   538→    "manage", "manage_periods", "reconcile", "file", "initiate", "convert",
   539→   539→    "generate", "calculate", "manage_permissions"];
   540→   540→
   541→   541→  if (granted === "admin") return true;
   542→   542→  if (granted === requested) return true;
   543→   543→
   544→   544→  // Special actions require explicit grant OR admin level
   545→   545→  if (SPECIAL_ACTIONS.includes(requested)) return granted === "admin";
   546→   546→
   547→   547→  // Hierarchy: approve > write > read
   548→   548→  const hierarchy: Record<string, number> = { none: 0, read: 1, write: 2, approve: 3, admin: 4 };
   549→   549→  const grantedLevel = hierarchy[granted] ?? 0;
   550→   550→  const requestedLevel = hierarchy[requested] ?? 0;
   551→   551→  return grantedLevel >= requestedLevel;
   552→   552→}
   553→   553→
   554→   554→/**
   555→   555→ * Check if a granted scope covers a requested scope.
   556→   556→ * Hierarchy: platform > company > team > own
   557→   557→ */
   558→   558→function scopeCovers(granted: PermissionScope, requested: PermissionScope): boolean {
   559→   559→  const hierarchy: Record<string, number> = { own: 1, team: 2, company: 3, platform: 4 };
   560→   560→  return hierarchy[granted] >= hierarchy[requested];
   561→   561→}

   563→   563→/**
   564→   564→ * Check legacy flat permission map.
   565→   565→ * Maps resource:action to the old permission keys.
   566→   566→ */
   567→   567→function checkLegacyPermission(
   568→   568→  perms: Record<string, number>,
   569→   569→  resource: string,
   570→   570→  action: string,
   571→   571→): boolean {
   572→   572→  const LEGACY_MAP: Record<string, Record<string, string>> = {
   573→   573→    invoice: { read: "create_invoice", write: "edit_invoice", delete: "delete_invoice", print: "print_invoice" },
   574→   574→    customer: { read: "view_customers", write: "edit_customer", delete: "delete_customer" },
   575→   575→    inventory: { read: "edit_inventory", write: "edit_inventory", manage_prices: "manage_wholesale_prices", bulk_input: "bulk_input" },
   576→   576→    report: { read: "reports_access", export: "export_data" },
   577→   577→    settings: { read: "settings_access", write: "settings_access" },
   578→   578→    accounting: { read: "finance_access", write: "finance_access" },
   579→   579→    employee: { read: "employee_management", write: "employee_management" },
   580→   580→    invoice_e_invoice: { submit: "e_invoicing_submit" },
   581→   581→  };
   582→   582→
   583→   583→  const resourceMap = LEGACY_MAP[resource];
   584→   584→  if (!resourceMap) return false;
   585→   585→  const permKey = resourceMap[action];
   586→   586→  if (!permKey) return false;
   587→   587→  return !!perms[permKey];
   588→   588→}

   590→   590→// ── Time Restriction Checking ────────────────────────────────────────────────
   591→   591→
   592→   592→/**
   593→   593→ * Check if the current time is within a time restriction window.
   594→   594→ */
   595→   595→export function isWithinTimeRestriction(restriction: TimeRestriction): boolean {
   596→   596→  const now = new Date();
   597→   597→
   598→   598→  // Check day-of-week restriction
   599→   599→  if (restriction.allowedDays && restriction.allowedDays.length > 0) {
   600→   600→    // ISO day: 1=Mon, 7=Sun. JS getDay: 0=Sun, 1=Mon, ... 6=Sat
   601→   601→    const jsDay = now.getDay();
   602→   602→    const isoDay = jsDay === 0 ? 7 : jsDay;
   603→   603→    if (!restriction.allowedDays.includes(isoDay)) return false;
   604→   604→  }
   605→   605→
   606→   606→  // Check hour restriction
   607→   607→  if (restriction.startHour !== undefined && restriction.endHour !== undefined) {
   608→   608→    const hour = now.getHours();
   609→   609→    if (hour < restriction.startHour || hour >= restriction.endHour) return false;
   610→   610→  }
   611→   611→
   612→   612→  // Check date range restriction
   613→   613→  if (restriction.startDate) {
   614→   614→    const start = new Date(restriction.startDate);
   615→   615→    if (now < start) return false;
   616→   616→  }
   617→   617→  if (restriction.endDate) {
   618→   618→    const end = new Date(restriction.endDate);
   619→   619→    if (now > end) return false;
   620→   620→  }
   621→   621→
   622→   622→  return true;
   623→   623→}

   625→   625→// ── Get Effective Permissions ────────────────────────────────────────────────
   626→   626→
   627→   627→/**
   628→   628→ * Compute the full effective permission set for a user.
   629→   629→ * Combines the legacy flat system with the new RBAC system.
   630→   630→ */
   631→   631→export function getEffectivePermissions(
   632→   632→  role: string,
   633→   633→  customPerms: Record<string, number> | null | undefined,
   634→   634→  isFounder: boolean,
   635→   635→  customRolePerms?: ResourcePermission[],
   636→   636→  timeRestrictions?: Record<string, TimeRestriction>,
   637→   637→): {
   638→   638→  flat: Record<string, number>;
   639→   639→  resources: ResourcePermission[];
   640→   640→  inheritanceChain: string[];
   641→   641→} {
   642→   642→  // Get legacy flat permissions
   643→   643→  const flat = computeEffectivePermissions(role, customPerms, isFounder);
   644→   644→
   645→   645→  // Get resource-level permissions from role hierarchy
   646→   646→  const roleDef = ROLE_DEFINITIONS[role] || customRoles.get(role);
   647→   647→  const inheritedPerms = roleDef ? getInheritedPermissions(role) : [];
   648→   648→  const resources = [...inheritedPerms, ...(customRolePerms || [])];
   649→   649→
   650→   650→  // Founder gets platform scope on all resources
   651→   651→  if (isFounder) {
   652→   652→    const founderPerms: ResourcePermission[] = AVAILABLE_RESOURCES.map((r) => ({
   653→   653→      resource: r,
   654→   654→      level: PermissionLevel.admin,
   655→   655→      scope: PermissionScope.platform,
   656→   656→    }));
   657→   657→    return { flat, resources: founderPerms, inheritanceChain: ["founder"] };
   658→   658→  }
   659→   659→
   660→   660→  const inheritanceChain = roleDef ? getInheritanceChain(role) : [role];
   661→   661→
   662→   662→  return { flat, resources, inheritanceChain };
   663→   663→}

   665→   665→// ── Validate Permission Changes ──────────────────────────────────────────────
   666→   666→
   667→   667→/**
   668→   668→ * Validate that a permission change is safe to apply.
   669→   669→ * Prevents:
   670→   670→ *   - Removing locked permissions from non-admin users
   671→   671→ *   - Granting platform scope to non-founders
   672→   672→ *   - Granting admin level on locked resources to non-admins
   673→   673→ *   - Creating roles with invalid inheritance
   674→   674→ */
   675→   675→export function validatePermissionChange(
   676→   676→  change: {
   677→   677→    type: "grant" | "revoke" | "modify";
   678→   678→    targetRole: string;
   679→   679→    resource: string;
   680→   680→    level: PermissionLevel;
   681→   681→    scope: PermissionScope;
   682→   682→  },
   683→   683→  actorRole: string,
   684→   684→  isFounder: boolean,
   685→   685→): { valid: boolean; reason?: string } {
   686→   686→  // Only admin or founder can modify permissions
   687→   687→  if (actorRole !== "admin" && !isFounder) {
   688→   688→    return { valid: false, reason: "Only admin or founder can modify permissions" };
   689→   689→  }
   690→   690→
   691→   691→  // Platform scope is founder-only
   692→   692→  if (change.scope === PermissionScope.platform && !isFounder) {
   693→   693→    return { valid: false, reason: "Platform scope requires founder access" };
   694→   694→  }
   695→   695→
   696→   696→  // Admin level on locked resources requires founder
   697→   697→  const LOCKED_RESOURCES = ["settings", "report", "audit", "role", "feature_flag"];
   698→   698→  if (LOCKED_RESOURCES.includes(change.resource) && change.level >= PermissionLevel.admin && !isFounder) {
   699→   699→    return { valid: false, reason: `Admin level on "${change.resource}" requires founder access` };
   700→   700→  }
   701→   701→
   702→   702→  // Cannot revoke permissions from a role higher than your own
   703→   703→  const actorDef = ROLE_DEFINITIONS[actorRole];
   704→   704→  const targetDef = ROLE_DEFINITIONS[change.targetRole] || customRoles.get(change.targetRole);
   705→   705→  if (actorDef && targetDef) {
   706→   706→    const actorChain = getInheritanceChain(actorRole);
   707→   707→    const targetChain = getInheritanceChain(change.targetRole);
   708→   708→    // If the target role's chain includes the actor's role, the actor can't revoke
   709→   709→    if (targetChain.includes(actorRole) && change.type === "revoke") {
   710→   710→      // Actually this is fine — you can revoke from roles below you
   711→   711→    }
   712→   712→  }
   713→   713→
   714→   714→  // Validate the resource exists
   715→   715→  if (!AVAILABLE_RESOURCES.includes(change.resource)) {
   716→   716→    return { valid: false, reason: `Unknown resource: "${change.resource}"` };
   717→   717→  }
   718→   718→
   719→   719→  return { valid: true };
   720→   720→}

   724→   724→function wouldCreateCircularInheritance(newRoleId: string, inheritsFrom: string): boolean {
   725→   725→  let current: string | null = inheritsFrom;
   726→   726→  const visited = new Set<string>();
   727→   727→
   728→   728→  while (current) {
   729→   729→    if (current === newRoleId) return true;
   730→   730→    if (visited.has(current)) return true; // existing circular chain
   731→   731→    visited.add(current);
   732→   732→    const def = ROLE_DEFINITIONS[current] || customRoles.get(current);
   733→   733→    current = def?.inheritsFrom ?? null;
   734→   734→  }
   735→   735→
   736→   736→  return false;
   737→   737→}
