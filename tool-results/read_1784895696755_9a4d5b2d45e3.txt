    84→export function AccountingView() {
    85→  const { activeCompany } = useBrand();
    86→  const [moduleTab, setModuleTab] = useState<ModuleTab>("core");
    87→
    88→  /* ─── Core sub-module state (preserved exactly as-is) ────────────────── */
    89→  const [tab, setTab] = useState<Tab>("accounts");
    90→  const [accounts, setAccounts] = useState<Account[]>([]);
    91→  const [entries, setEntries] = useState<JournalEntry[]>([]);
    92→  const [trial, setTrial] = useState<{ accounts: TrialRow[]; grandDebit: number; grandCredit: number; isBalanced: boolean } | null>(null);
    93→  const [fiscalPeriods, setFiscalPeriods] = useState<FiscalPeriod[]>([]);
    94→  const [costCenters, setCostCenters] = useState<CostCenter[]>([]);
    95→  const [agingData, setAgingData] = useState<AgingBucket[]>([]);
    96→  const [bankAccounts, setBankAccounts] = useState<BankAccount[]>([]);
    97→  const [loading, setLoading] = useState(true);
    98→  const [showForm, setShowForm] = useState(false);
    99→  const [selectedIds, setSelectedIds] = useState<Set<number>>(new Set());
   100→  const [currentPage, setCurrentPage] = useState(1);
   101→  const [bulkDeleting, setBulkDeleting] = useState(false);
   102→  const [reversingId, setReversingId] = useState<number | null>(null);
   103→  const [reverseConfirm, setReverseConfirm] = useState<JournalEntry | null>(null);
   104→
   105→  const load = useCallback(async () => {
   106→    if (!activeCompany) { setLoading(false); return; }
   107→    setLoading(true);
   108→    const slug = `companySlug=${encodeURIComponent(activeCompany.slug)}`;
   109→    try {
   110→      const [a, j] = await Promise.all([
   111→        authedFetch(`/api/accounting/accounts?${slug}`),
   112→        authedFetch(`/api/accounting/journal-entries?${slug}`),
   113→      ]);
   114→      if (!a.ok || !j.ok) {
   115→        const failed = !a.ok ? a : j;
   116→        const e = await failed.json().catch(() => ({}));
   117→        throw new Error((e as Record<string, unknown>)?.error as string || `فشل تحميل البيانات (${failed.status})`);
   118→      }
   119→      const [aD, jD] = await Promise.all([a.json(), j.json()]);
   120→      setAccounts((aD as { accounts?: Account[] }).accounts || []);
   121→      setEntries((jD as { entries?: JournalEntry[] }).entries || []);
   122→    } catch (err) {
   123→      toast.error(err instanceof Error ? err.message : "تعذّر تحميل بيانات المحاسبة");
   124→    } finally { setLoading(false); }
   125→  }, [activeCompany]);
   126→
   127→  const loadTrial = useCallback(async () => {
   128→    if (!activeCompany) return;
   129→    try {
   130→      const res = await authedFetch(`/api/accounting/trial-balance?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   131→      if (!res.ok) {
   132→        const e = await res.json().catch(() => ({}));
   133→        toast.error((e as Record<string, unknown>)?.error as string || "تعذّر تحميل ميزان المراجعة");
   134→        setTrial(null);
   135→        return;
   136→      }
   137→      setTrial(await res.json());
   138→    } catch (err) {
   139→      toast.error(err instanceof Error ? err.message : "تعذّر تحميل ميزان المراجعة");
   140→      setTrial(null);
   141→    }
   142→  }, [activeCompany]);
   143→
   144→  const loadFiscalPeriods = useCallback(async () => {
   145→    if (!activeCompany) return;
   146→    try {
   147→      const res = await authedFetch(`/api/accounting/fiscal-periods?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   148→      if (res.ok) { const d = await res.json(); setFiscalPeriods(d.periods || []); }
   149→      else { setFiscalPeriods([]); }
   150→    } catch { setFiscalPeriods([]); }
   151→  }, [activeCompany]);
   152→
   153→  const loadCostCenters = useCallback(async () => {
   154→    if (!activeCompany) return;
   155→    try {
   156→      const res = await authedFetch(`/api/accounting/cost-centers?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   157→      if (res.ok) { const d = await res.json(); setCostCenters(d.costCenters || []); }
   158→      else { setCostCenters([]); }
   159→    } catch { setCostCenters([]); }
   160→  }, [activeCompany]);
   161→
   162→  const loadAging = useCallback(async () => {
   163→    if (!activeCompany) return;
   164→    try {
   165→      const res = await authedFetch(`/api/accounting/aging?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   166→      if (res.ok) { const d = await res.json(); setAgingData(d.buckets || []); }
   167→      else { setAgingData([]); }
   168→    } catch { setAgingData([]); }
   169→  }, [activeCompany]);
   170→
   171→  const loadBankAccounts = useCallback(async () => {
   172→    if (!activeCompany) return;
   173→    try {
   174→      const res = await authedFetch(`/api/accounting/bank-accounts?companySlug=${encodeURIComponent(activeCompany.slug)}`);
   175→      if (res.ok) { const d = await res.json(); setBankAccounts(d.accounts || []); }
   176→      else { setBankAccounts([]); }
   177→    } catch { setBankAccounts([]); }
   178→  }, [activeCompany]);
   179→
   180→  useEffect(() => { load(); }, [load]);
   181→  useEffect(() => {
   182→    if (tab === "trial" && activeCompany) loadTrial();
   183→    if (tab === "fiscal-periods" && activeCompany) loadFiscalPeriods();
   184→    if (tab === "cost-centers" && activeCompany) loadCostCenters();
   185→    if (tab === "aging" && activeCompany) loadAging();
   186→    if (tab === "banking" && activeCompany) loadBankAccounts();
   187→    if (tab === "dashboard" && activeCompany) { load(); loadAging(); loadBankAccounts(); }
   188→  }, [tab, activeCompany, loadTrial, loadFiscalPeriods, loadCostCenters, loadAging, loadBankAccounts, load]);
   189→
   190→  const switchTab = (t: Tab) => {
   191→    setTab(t);
   192→    setShowForm(false);
   193→    setSelectedIds(new Set());
   194→    setCurrentPage(1);
   195→  };
   196→
   197→  const itemsForTab = (): Array<{ id: number }> => (tab === "accounts" ? accounts : entries);
   198→
   199→  const allItems = itemsForTab();
   200→  const totalPages = Math.max(1, Math.ceil(allItems.length / PAGE_SIZE));
   201→  const pageItems = allItems.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE);
   202→  const safePage = Math.min(currentPage, totalPages);
   203→
   204→  const toggleSelectAll = () => {
   205→    if (selectedIds.size === pageItems.length && pageItems.length > 0) setSelectedIds(new Set());
   206→    else setSelectedIds(new Set(pageItems.map((i) => i.id)));
   207→  };
   208→  const toggleRow = (id: number) => {
   209→    setSelectedIds((prev) => {
   210→      const next = new Set(prev);
   211→      if (next.has(id)) next.delete(id);
   212→      else next.add(id);
   213→      return next;
   214→    });
   215→  };
   216→  const handleBulkDelete = async () => {
   217→    if (selectedIds.size === 0) return;
   218→    if (!confirm(`حذف ${selectedIds.size} عنصر؟`)) return;
   219→    const endpoint = tab === "accounts" ? "/api/accounting/accounts" : "/api/accounting/journal-entries";
   220→    setBulkDeleting(true);
   221→    let okCount = 0, failCount = 0;
   222→    for (const id of selectedIds) {
   223→      try {
   224→        const res = await authedFetch(`${endpoint}/${id}`, { method: "DELETE" });
   225→        if (res.ok) okCount++;
   226→        else { failCount++; }
   227→      } catch { failCount++; }
   228→    }
   229→    setBulkDeleting(false);
   230→    setSelectedIds(new Set());
   231→    if (okCount > 0) toast.success(`تم حذف ${okCount} عنصر`);
   232→    if (failCount > 0) toast.error(`تعذّر حذف ${failCount} عنصر`);
   233→    load();
   234→  };
   235→  const handleDelete = async (id: number) => {
   236→    if (!confirm("حذف هذا العنصر؟")) return;
   237→    const endpoint = tab === "accounts" ? "/api/accounting/accounts" : "/api/accounting/journal-entries";
   238→    const res = await authedFetch(`${endpoint}/${id}`, { method: "DELETE" });
   239→    if (res.ok) { toast.success("تم الحذف"); load(); }
   240→    else { const e = await res.json().catch(() => ({})); toast.error(e.error || "تعذّر الحذف"); }
   241→  };
   242→
   243→  const handleReverse = async (entry: JournalEntry) => {
   244→    if (!activeCompany) return;
   245→    setReverseConfirm(entry);
   246→  };
   247→
   248→  const confirmReverse = async () => {
   249→    if (!reverseConfirm || !activeCompany) return;
   250→    const entry = reverseConfirm;
   251→    setReversingId(entry.id);
   252→    try {
   253→      const url = `/api/accounting/journal-entries/${entry.id}/reverse?companySlug=${encodeURIComponent(activeCompany.slug)}`;
   254→      const res = await authedFetch(url, { method: "POST" });
   255→      if (!res.ok) {
   256→        const e = await res.json().catch(() => ({}));
   257→        throw new Error(e.error || "تعذّر العكس");
   258→      }
   259→      const data = await res.json().catch(() => ({}));
   260→      const reversalId = data?.reversal?.id;
   261→      toast.success(reversalId ? `تم إنشاء قيد عكسي #${reversalId}` : "تم عكس القيد");
   262→      setReverseConfirm(null);
   263→      await load();
   264→    } catch (err) {
   265→      toast.error(err instanceof Error ? err.message : "خطأ أثناء العكس");
   266→    } finally { setReversingId(null); }
   267→  };
   268→
   269→  const pageBtnStyle = (disabled: boolean): string =>
   270→    disabled
   271→      ? "py-1.5 px-3 rounded-md bg-transparent text-muted-foreground border border-border text-[12px] font-bold cursor-not-allowed opacity-50"
   272→      : "py-1.5 px-3 rounded-md bg-card text-foreground border border-border text-[12px] font-bold cursor-pointer";
   273→
   274→  if (!activeCompany) return <div className="p-8 md:p-12 text-center text-muted-foreground">اختر شركة</div>;
   275→
   276→  const ACCOUNT_TYPES: Record<string, { label: string; color: string; badge: string }> = {
   277→    asset: { label: "أصول", color: "#10b981", badge: "bg-emerald-500/15 text-emerald-500" },
   278→    liability: { label: "خصوم", color: "#ef4444", badge: "bg-red-500/15 text-red-500" },
   279→    equity: { label: "حقوق ملكية", color: "#7c3aed", badge: "bg-violet-500/15 text-violet-500" },
   280→    revenue: { label: "إيرادات", color: "#3b82f6", badge: "bg-blue-500/15 text-blue-500" },
   281→    expense: { label: "مصروفات", color: "#f59e0b", badge: "bg-amber-500/15 text-amber-500" },
   282→    contra_revenue: { label: "مقابل إيرادات", color: "#9ca3af", badge: "bg-gray-400/15 text-gray-400" },
   283→    contra_asset: { label: "مقابل أصول", color: "#9ca3af", badge: "bg-gray-400/15 text-gray-400" },
   284→  };
   285→
   286→  const tabs: Array<{ key: Tab; label: string; icon?: React.ComponentType<{ size?: number }> }> = [
   287→    { key: "dashboard", label: "لوحة التحكم", icon: LayoutDashboard },
   288→    { key: "accounts", label: `الحسابات (${accounts.length})` },
   289→    { key: "journal", label: `القيود (${entries.length})` },
   290→    { key: "trial", label: "ميزان المراجعة", icon: Scale },
   291→    { key: "statements", label: "القوائم المالية", icon: FileBarChart },
   292→    { key: "fiscal-periods", label: "الفترات المالية", icon: Clock },
   293→    { key: "cost-centers", label: "مراكز التكلفة", icon: Building },
   294→    { key: "aging", label: "تقادم الذمم", icon: BarChart3 },
   295→    { key: "banking", label: "البنوك", icon: Landmark },
   296→  ];
   297→
   298→  // Dashboard metrics
   299→  const totalRevenue = accounts.filter(a => a.type === "revenue").reduce((s, a) => s + a.balance, 0);
   300→  const totalExpenses = accounts.filter(a => a.type === "expense").reduce((s, a) => s + a.balance, 0);
   301→  const totalAssets = accounts.filter(a => a.type === "asset").reduce((s, a) => s + a.balance, 0);
   302→  const totalLiabilities = accounts.filter(a => a.type === "liability").reduce((s, a) => s + a.balance, 0);
   303→  const netProfit = totalRevenue - totalExpenses;
   304→  const totalAR = agingData.reduce((s, b) => s + b.receivable, 0);
   305→  const totalAP = agingData.reduce((s, b) => s + b.payable, 0);
   306→  const totalCash = bankAccounts.reduce((s, b) => s + b.balance, 0);
   307→  const currentPeriod = fiscalPeriods.find(p => p.status === "open");
   308→
   309→  /* ─── Render ──────────────────────────────────────────────────────────── */
   310→  return (
   311→    <div className="flex flex-col gap-4">
   312→      {/* Module-level header */}
   313→      <div className="flex flex-wrap justify-between items-center gap-3">
   314→        <div><h1 className="text-xl md:text-2xl font-extrabold flex items-center gap-2"><Calculator size={20} /> المحاسبة</h1><p className="text-[13px] text-muted-foreground">{activeCompany.nameAr || activeCompany.name}</p></div>
   315→      </div>
   316→
   317→      {/* Module-level navigation tabs (top row, visually distinct) */}
   318→      <div className="flex gap-1.5 flex-wrap bg-muted/50 rounded-[12px] p-2">
   319→        {MODULE_TABS.map((mt) => {
   320→          const Icon = mt.icon;
   321→          const active = moduleTab === mt.key;
   322→          return (
   323→            <button
   324→              key={mt.key}
   325→              onClick={() => setModuleTab(mt.key)}
   326→              className={cn(
   327→                "py-2.5 px-4 rounded-[10px] text-[13px] font-bold cursor-pointer inline-flex items-center gap-2 transition-colors",
   328→                active
   329→                  ? "bg-primary text-primary-foreground shadow-sm"
   330→                  : "bg-card text-muted-foreground border border-border hover:bg-accent hover:text-accent-foreground",
   331→              )}
   332→            >
   333→              <Icon size={16} />
   334→              {mt.label}
   335→            </button>
   336→          );
   337→        })}
   338→      </div>
   339→
   340→      {/* ─── Module content ─────────────────────────────────────────────── */}
   341→      {moduleTab === "core" ? (
   342→        /* ─── Core sub-module: exact original inner content ───────────── */
   343→        <>
   344→          <div className="flex flex-wrap justify-between items-center gap-3">
   345→            {(tab === "accounts" || tab === "journal" || tab === "fiscal-periods" || tab === "cost-centers" || tab === "banking") && (
   346→              <button onClick={() => setShowForm(true)} className="inline-flex items-center gap-1.5 py-2.5 px-[18px] rounded-[10px] bg-primary text-primary-foreground border-none text-[13px] font-bold cursor-pointer"><Plus size={16} /> إضافة</button>
   347→            )}
   348→          </div>
   349→          <div className="flex gap-1.5 flex-wrap">
   350→            {tabs.map((t) => {
   351→              const Icon = t.icon;
   352→              return (
   353→                <button key={t.key} onClick={() => switchTab(t.key)} className={cn(
   354→                  "py-2 px-4 rounded-[10px] border border-border text-[12px] font-bold cursor-pointer inline-flex items-center gap-1.5",
   355→                  tab === t.key ? "bg-primary text-primary-foreground" : "bg-card text-muted-foreground",
   356→                )}>
   357→                  {Icon && <Icon size={14} />}
   358→                  {t.label}
   359→                </button>
   360→              );
   361→            })}
   362→          </div>
   363→
   364→          {loading ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : tab === "dashboard" ? (
   365→            <FinancialDashboard
   366→              totalRevenue={totalRevenue}
   367→              totalExpenses={totalExpenses}
   368→              netProfit={netProfit}
   369→              totalAssets={totalAssets}
   370→              totalLiabilities={totalLiabilities}
   371→              totalAR={totalAR}
   372→              totalAP={totalAP}
   373→              totalCash={totalCash}
   374→              currentPeriod={currentPeriod}
   375→              entriesCount={entries.length}
   376→              accountsCount={accounts.length}
   377→            />
   378→          ) : tab === "trial" ? (
   379→            <TrialBalanceTable data={trial} loading={loading} />
   380→          ) : tab === "statements" ? (
   381→            <FinancialStatements company={activeCompany} />
   382→          ) : tab === "fiscal-periods" ? (
   383→            showForm ? <FiscalPeriodForm company={activeCompany} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadFiscalPeriods(); }} /> : (
   384→              <FiscalPeriodsTable periods={fiscalPeriods} company={activeCompany} onRefresh={loadFiscalPeriods} />
   385→            )
   386→          ) : tab === "cost-centers" ? (
   387→            showForm ? <CostCenterForm company={activeCompany} costCenters={costCenters} onClose={() => setShowForm(false)} onSaved={() => { setShowForm(false); loadCostCenters(); }} /> : (

  1406→function JournalForm({ company, accounts, onClose, onSaved }: { company: { slug: string }; accounts: Account[]; onClose: () => void; onSaved: () => void }) {
  1407→  const [date, setDate] = useState(new Date().toISOString().slice(0, 10));
  1408→  const [description, setDescription] = useState("");
  1409→  const [reference, setReference] = useState("");
  1410→  const [status, setStatus] = useState("draft");
  1411→  const [lines, setLines] = useState<Array<{ accountId: number | null; debit: number; credit: number; description?: string }>>([{ accountId: null, debit: 0, credit: 0 }]);
  1412→  const [saving, setSaving] = useState(false);
  1413→
  1414→  const totalDebit = lines.reduce((s, l) => s + Number(l.debit || 0), 0);
  1415→  const totalCredit = lines.reduce((s, l) => s + Number(l.credit || 0), 0);
  1416→  const isBalanced = Math.abs(totalDebit - totalCredit) < 0.001;
  1417→
  1418→  const updateLine = (i: number, field: string, value: number | string) => { setLines((arr) => arr.map((l, idx) => idx === i ? { ...l, [field]: value } : l)); };
  1419→  const addLine = () => setLines((arr) => [...arr, { accountId: null, debit: 0, credit: 0 }]);
  1420→  const removeLine = (i: number) => setLines((arr) => arr.filter((_, idx) => idx !== i));
  1421→
  1422→  const submit = async () => {
  1423→    if (lines.length === 0) { toast.error("أضف بنداً واحداً على الأقل"); return; }
  1424→    if (!isBalanced) { toast.error("القيد غير متوازن — المدين ≠ الدائن"); return; }
  1425→    if (lines.some((l) => !l.accountId)) { toast.error("كل بند يجب أن يحدد حساباً"); return; }
  1426→    setSaving(true);
  1427→    try {
  1428→      const res = await authedFetch("/api/accounting/journal-entries", {
  1429→        method: "POST", headers: { "Content-Type": "application/json" },
  1430→        body: JSON.stringify({ date, description, reference, status, lines, companySlug: company.slug }),
  1431→      });
  1432→      if (!res.ok) { const e = await res.json().catch(() => ({})); throw new Error(e.error || "Failed"); }
  1433→      toast.success("تم إنشاء القيد"); onSaved();
  1434→    } catch (err) { toast.error(err instanceof Error ? err.message : "خطأ"); }
  1435→    finally { setSaving(false); }
  1436→  };
  1437→
  1438→  return (
  1439→    <div className="flex flex-col gap-4">
  1440→      <div className="bg-card rounded-[14px] border border-border p-5 flex flex-col gap-3.5">
  1441→        <h3 className="text-[15px] font-bold">قيد يومية جديد</h3>
  1442→        <div className="grid grid-cols-[repeat(auto-fit,minmax(160px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(200px,1fr))] gap-3">
  1443→          <div><label className={labelStyle}>التاريخ</label><input type="date" value={date} onChange={(e) => setDate(e.target.value)} className={inputStyle} dir="ltr" /></div>
  1444→          <div><label className={labelStyle}>المرجع</label><input value={reference} onChange={(e) => setReference(e.target.value)} className={inputStyle} dir="ltr" /></div>
  1445→          <div><label className={labelStyle}>الحالة</label>
  1446→            <select value={status} onChange={(e) => setStatus(e.target.value)} className={inputStyle}><option value="draft">مسودة</option><option value="posted">مُرحّل</option></select>
  1447→          </div>
  1448→        </div>

  1450→        <div>
  1451→          <div className="flex justify-between items-center mb-2">
  1452→            <label className={cn(labelStyle, "mb-0")}>البنود</label>

  1458→                <select value={l.accountId ?? ""} onChange={(e) => updateLine(i, "accountId", Number(e.target.value))} className={inputStyle}><option value="">— اختر حساب —</option>{accounts.map((a) => <option key={a.id} value={a.id}>{a.code} — {a.nameAr}</option>)}</select>

  1460→                <input type="number" placeholder="دائن" value={l.credit} onChange={(e) => updateLine(i, "credit", Number(e.target.value))} className={inputStyle} dir="ltr" />
