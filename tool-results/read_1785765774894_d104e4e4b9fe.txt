   161→/* ─── Bank Account Form ────────────────────────────────────────────────────── */
   162→function BankAccountFormView({ company, onClose, onSaved }: { company: { slug: string; currency?: string }; onClose: () => void; onSaved: () => void }) {
   163→  const [bankName, setBankName] = useState("");
   164→  const [accountName, setAccountName] = useState("");
   165→  const [accountNumber, setAccountNumber] = useState("");
   166→  const [iban, setIban] = useState("");
   167→  const [currency, setCurrency] = useState(company.currency || "KWD");
   168→  const [accountType, setAccountType] = useState("checking");
   169→  const [glAccountId, setGlAccountId] = useState<number | null>(null);
   170→  const [balance, setBalance] = useState(0);
   171→  const [saving, setSaving] = useState(false);
   172→
   173→  const glAccountsQuery = useAccounts(company.slug);
   174→  const glAccounts = (glAccountsQuery.data?.accounts || []) as Array<{ id: number; code: string; name: string; nameAr?: string }>;
   175→  const createBankAccountMutation = useCreateBankAccount();
   176→
   177→  const submit = () => {
   178→    if (!bankName || !accountName || !accountNumber) { toast.error("البنك واسم الحساب ورقم الحساب مطلوبة"); return; }
   179→    setSaving(true);
   180→    createBankAccountMutation.mutate(
   181→      { name: accountName, bankName, accountName, accountNumber, iban, currency, accountType, glAccountId, balance, companySlug: company.slug },
   182→      {
   183→        onSuccess: () => { toast.success("تم إنشاء الحساب البنكي"); onSaved(); setSaving(false); },
   184→        onError: (err) => { toast.error(err.message || "خطأ"); setSaving(false); },
   185→      },
   186→    );
   187→  };
   188→
   189→  return (
   190→    <div className="bg-card rounded-[14px] border border-border p-5 flex flex-col gap-3.5">
   191→      <h3 className="text-[15px] font-bold flex items-center gap-2"><Landmark size={16} /> حساب بنكي جديد</h3>
   192→      <div className="grid grid-cols-[repeat(auto-fit,minmax(160px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(200px,1fr))] gap-3">
   193→        <div><label className={labelStyle}>البنك *</label><input value={bankName} onChange={(e) => setBankName(e.target.value)} className={inputStyle} /></div>
   194→        <div><label className={labelStyle}>اسم الحساب *</label><input value={accountName} onChange={(e) => setAccountName(e.target.value)} className={inputStyle} /></div>
   195→        <div><label className={labelStyle}>رقم الحساب *</label><input value={accountNumber} onChange={(e) => setAccountNumber(e.target.value)} className={inputStyle} dir="ltr" /></div>
   196→        <div><label className={labelStyle}>IBAN</label><input value={iban} onChange={(e) => setIban(e.target.value)} className={inputStyle} dir="ltr" /></div>
   197→        <div><label className={labelStyle}>العملة</label>
   198→          <select value={currency} onChange={(e) => setCurrency(e.target.value)} className={inputStyle}>
   199→            <option value="KWD">KWD</option><option value="SAR">SAR</option><option value="AED">AED</option><option value="USD">USD</option><option value="EUR">EUR</option>
   200→          </select>
   201→        </div>
   202→        <div><label className={labelStyle}>نوع الحساب</label>
   203→          <select value={accountType} onChange={(e) => setAccountType(e.target.value)} className={inputStyle}>
   204→            <option value="checking">جاري</option><option value="savings">توفير</option><option value="overdraft">سحب على المكشوف</option><option value="loan">قرض</option>
   205→          </select>
   206→        </div>
   207→        <div><label className={labelStyle}>الحساب المالي (GL)</label>
   208→          <select value={glAccountId ?? ""} onChange={(e) => setGlAccountId(e.target.value ? Number(e.target.value) : null)} className={inputStyle}>
   209→            <option value="">— اختر —</option>
   210→            {glAccounts.map((a) => <option key={a.id} value={a.id}>{a.code} — {a.nameAr}</option>)}
   211→          </select>
   212→        </div>
   213→        <div><label className={labelStyle}>الرصيد الافتتاحي</label><input type="number" value={balance} onChange={(e) => setBalance(Number(e.target.value))} className={inputStyle} dir="ltr" /></div>
   214→      </div>
   215→      <div className="flex gap-2.5 justify-end">
   216→        <button onClick={onClose} className="py-2.5 px-5 rounded-md bg-transparent text-muted-foreground border border-border text-[13px] font-bold cursor-pointer">إلغاء</button>
   217→        <button onClick={submit} disabled={saving} className="py-2.5 px-6 rounded-md bg-primary text-primary-foreground border-none text-[13px] font-extrabold cursor-pointer disabled:opacity-70">{saving ? "جارٍ…" : "حفظ"}</button>
   218→      </div>
   219→    </div>
   220→  );

   223→/* ─── Reconciliation ─────────────────────────────────────────────────────────── */
   224→function ReconciliationView({ company }: { company: { slug: string } }) {
   225→  const [selectedAccountId, setSelectedAccountId] = useState<number | null>(null);
   226→  const [actionId, setActionId] = useState<number | null>(null);
   227→  const [completing, setCompleting] = useState(false);
   228→
   229→  const slug = encodeURIComponent(company.slug);
   230→
   231→  const bankAccountsQuery = useBankAccountsList(slug);
   232→  const bankAccounts = bankAccountsQuery.data?.bankAccounts ?? [];
   233→  const reconciliationQuery = useBankReconciliation(slug, selectedAccountId);
   234→  const items = reconciliationQuery.data?.items ?? [];
   235→  const loading = reconciliationQuery.isLoading && !!selectedAccountId;
   236→  const matchMutation = useMatchBankReconciliation();
   237→  const completeMutation = useCompleteBankReconciliation();
   238→
   239→  const loadItems = () => { reconciliationQuery.refetch(); };
   240→
   241→  const totalBank = items.reduce((s, i) => s + i.bankAmount, 0);
   242→  const totalBook = items.reduce((s, i) => s + i.bookAmount, 0);
   243→  const totalDiff = totalBank - totalBook;
   244→  const matchedCount = items.filter(i => i.status === "matched").length;
   245→  const unmatchedCount = items.filter(i => i.status !== "matched").length;
   246→
   247→  const handleMatch = (id: number) => {
   248→    setActionId(id);
   249→    matchMutation.mutate(
   250→      { id, companySlug: company.slug, bankAccountId: selectedAccountId! },
   251→      {
   252→        onSuccess: () => { toast.success("تم المطابقة"); reconciliationQuery.refetch(); setActionId(null); },
   253→        onError: (err) => { toast.error(err.message || "تعذّر المطابقة"); setActionId(null); },
   254→      },
   255→    );
   256→  };
   257→
   258→  const handleComplete = () => {
   259→    if (!selectedAccountId) return;
   260→    if (unmatchedCount > 0 && !confirm(`هناك ${unmatchedCount} عنصر غير مطابق. إتمام المطابقة؟`)) return;
   261→    setCompleting(true);
   262→    completeMutation.mutate(
   263→      { companySlug: company.slug, bankAccountId: selectedAccountId },
   264→      {
   265→        onSuccess: () => { toast.success("تم إتمام المطابقة البنكية"); reconciliationQuery.refetch(); setCompleting(false); },
   266→        onError: (err) => { toast.error(err.message || "تعذّر إتمام المطابقة"); setCompleting(false); },
   267→      },
   268→    );
   269→  };
   270→
   271→  return (
   272→    <div className="flex flex-col gap-4">
   273→      <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex gap-3 items-center flex-wrap">
   274→        <div className="flex items-center gap-1.5">
   275→          <label className="text-[11px] font-bold text-muted-foreground">الحساب البنكي</label>
   276→          <select value={selectedAccountId ?? ""} onChange={(e) => setSelectedAccountId(e.target.value ? Number(e.target.value) : null)} className={cn(inputStyle, "w-auto min-w-[200px]")}>
   277→            <option value="">— اختر حساب —</option>
   278→            {bankAccounts.map((a) => <option key={a.id} value={a.id}>{a.bankName} — {a.accountNumber}</option>)}
   279→          </select>
   280→        </div>
   281→        <button onClick={loadItems} disabled={!selectedAccountId || loading} className="py-2 px-4 rounded-sm bg-accent text-accent-foreground border border-border text-[12px] font-bold cursor-pointer disabled:opacity-70 inline-flex items-center gap-1.5">
   282→          <RefreshCw size={12} className={loading ? "animate-spin" : ""} /> {loading ? "جارٍ…" : "عرض"}
   283→        </button>
   284→        {selectedAccountId && unmatchedCount === 0 && items.length > 0 && (
   285→          <button onClick={handleComplete} disabled={completing} className="py-2 px-4 rounded-sm bg-emerald-600 text-white border-none text-[12px] font-bold cursor-pointer disabled:opacity-70 inline-flex items-center gap-1.5">
   286→            <CheckCircle2 size={12} /> {completing ? "جارٍ…" : "إتمام المطابقة"}
   287→          </button>
   288→        )}
   289→      </div>
   290→
   291→      {selectedAccountId && items.length > 0 && (
   292→        <div className="grid grid-cols-[repeat(auto-fit,minmax(140px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(180px,1fr))] gap-3">
   293→          <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   294→            <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-emerald-500/20 text-emerald-500"><Landmark size={18} /></div>
   295→            <div><div className="text-[11px] text-muted-foreground">إجمالي البنك</div><div className="text-lg font-extrabold [direction:ltr] text-end">{fmt(totalBank)}</div></div>
   296→          </div>
   297→          <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   298→            <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-blue-500/20 text-blue-500"><FileText size={18} /></div>
   299→            <div><div className="text-[11px] text-muted-foreground">إجمالي الكتب</div><div className="text-lg font-extrabold [direction:ltr] text-end">{fmt(totalBook)}</div></div>
   300→          </div>
   301→          <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   302→            <div className={cn("w-10 h-10 rounded-sm flex items-center justify-center", totalDiff === 0 ? "bg-emerald-500/20 text-emerald-500" : "bg-red-500/20 text-red-500")}>
   303→              {totalDiff === 0 ? <CheckCircle2 size={18} /> : <AlertTriangle size={18} />}
   304→            </div>
   305→            <div><div className="text-[11px] text-muted-foreground">الفرق</div><div className={cn("text-lg font-extrabold [direction:ltr] text-end", totalDiff === 0 ? "text-emerald-500" : "text-red-500")}>{fmt(totalDiff)}</div></div>
   306→          </div>
   307→          <div className="bg-card rounded-[14px] border border-border py-3.5 px-4 flex items-center gap-3">
   308→            <div className="w-10 h-10 rounded-sm flex items-center justify-center bg-emerald-700/20 text-emerald-700"><CheckCircle2 size={18} /></div>
   309→            <div><div className="text-[11px] text-muted-foreground">مطابق / غير مطابق</div><div className="text-lg font-extrabold [direction:ltr] text-end"><span className="text-emerald-500">{matchedCount}</span> / <span className="text-amber-500">{unmatchedCount}</span></div></div>
   310→          </div>
   311→        </div>
   312→      )}
   313→
   314→      <div className="bg-card rounded-[14px] border border-border overflow-hidden">
   315→        {!selectedAccountId ? <div className="p-12 text-center text-muted-foreground">اختر حساب بنكي لعرض عناصر المطابقة</div> : items.length === 0 ? <Empty label="عناصر المطابقة" /> : (
   316→          <div className="overflow-x-auto garfix-scroll">
   317→            <table className="table-enterprise w-full border-collapse">

   429→/* ─── Transfer Form ────────────────────────────────────────────────────────── */
   430→function TransferFormView({ accounts, company, onClose, onSaved }: { accounts: BankAccount[]; company: { slug: string }; onClose: () => void; onSaved: () => void }) {
   431→  const [fromAccountId, setFromAccountId] = useState<number | null>(null);
   432→  const [toAccountId, setToAccountId] = useState<number | null>(null);
   433→  const [amount, setAmount] = useState(0);
   434→  const [currency, setCurrency] = useState("KWD");
   435→  const [date, setDate] = useState(new Date().toISOString().slice(0, 10));
   436→  const [description, setDescription] = useState("");
   437→  const [saving, setSaving] = useState(false);
   438→  const createBankTransferMutation = useCreateBankTransfer();
   439→
   440→  const fromAccount = accounts.find(a => a.id === fromAccountId);
   441→  const toAccount = accounts.find(a => a.id === toAccountId);
   442→
   443→  // Auto-set currency when selecting source account
   444→  useEffect(() => {
   445→     
   446→    if (fromAccount) setCurrency(fromAccount.currency);
   447→  }, [fromAccount]);
   448→
   449→  const submit = () => {
   450→    if (!fromAccountId || !toAccountId || amount <= 0) { toast.error("اختر الحسابات والمبلغ"); return; }
   451→    if (fromAccountId === toAccountId) { toast.error("لا يمكن التحويل إلى نفس الحساب"); return; }
   452→    setSaving(true);
   453→    createBankTransferMutation.mutate(
   454→      { fromAccountId, toAccountId, amount, currency, date, description, companySlug: company.slug },
   455→      {
   456→        onSuccess: () => { toast.success("تم إنشاء التحويل"); onSaved(); setSaving(false); },
   457→        onError: (err) => { toast.error(err.message || "خطأ"); setSaving(false); },
   458→      },
   459→    );
   460→  };
   461→
   462→  return (
   463→    <div className="bg-card rounded-[14px] border border-border p-5 flex flex-col gap-3.5">
   464→      <h3 className="text-[15px] font-bold flex items-center gap-2"><ArrowRightLeft size={16} /> تحويل جديد</h3>
   465→      <div className="grid grid-cols-[repeat(auto-fit,minmax(160px,1fr))] sm:grid-cols-[repeat(auto-fit,minmax(200px,1fr))] gap-3">
   466→        <div><label className={labelStyle}>من حساب *</label>
   467→          <select value={fromAccountId ?? ""} onChange={(e) => setFromAccountId(e.target.value ? Number(e.target.value) : null)} className={inputStyle}>
   468→            <option value="">— اختر —</option>
   469→            {accounts.map((a) => <option key={a.id} value={a.id}>{a.bankName} — {a.accountNumber} ({a.currency})</option>)}
   470→          </select>
   471→          {fromAccount && <div className="text-[11px] text-muted-foreground mt-1">الرصيد: <span className={cn("font-bold", fromAccount.balance >= 0 ? "text-emerald-500" : "text-red-500")}>{fmt(fromAccount.balance)} {fromAccount.currency}</span></div>}
   472→        </div>
   473→        <div><label className={labelStyle}>إلى حساب *</label>
   474→          <select value={toAccountId ?? ""} onChange={(e) => setToAccountId(e.target.value ? Number(e.target.value) : null)} className={inputStyle}>
   475→            <option value="">— اختر —</option>
   476→            {accounts.filter(a => a.id !== fromAccountId).map((a) => <option key={a.id} value={a.id}>{a.bankName} — {a.accountNumber} ({a.currency})</option>)}
   477→          </select>
   478→          {toAccount && <div className="text-[11px] text-muted-foreground mt-1">الرصيد: <span className={cn("font-bold", toAccount.balance >= 0 ? "text-emerald-500" : "text-red-500")}>{fmt(toAccount.balance)} {toAccount.currency}</span></div>}
   479→        </div>
   480→        <div><label className={labelStyle}>المبلغ *</label><input type="number" value={amount} onChange={(e) => setAmount(Number(e.target.value))} className={inputStyle} dir="ltr" min={0} /></div>
   481→        <div><label className={labelStyle}>العملة</label>
   482→          <select value={currency} onChange={(e) => setCurrency(e.target.value)} className={inputStyle}>
   483→            <option value="KWD">KWD</option><option value="SAR">SAR</option><option value="AED">AED</option><option value="USD">USD</option><option value="EUR">EUR</option>
   484→          </select>
   485→        </div>
   486→        <div><label className={labelStyle}>التاريخ</label><input type="date" value={date} onChange={(e) => setDate(e.target.value)} className={inputStyle} dir="ltr" /></div>
   487→        <div><label className={labelStyle}>الوصف</label><input value={description} onChange={(e) => setDescription(e.target.value)} className={inputStyle} placeholder="وصف التحويل" /></div>
   488→      </div>
   489→      <div className="flex gap-2.5 justify-end">
   490→        <button onClick={onClose} className="py-2.5 px-5 rounded-md bg-transparent text-muted-foreground border border-border text-[13px] font-bold cursor-pointer">إلغاء</button>
   491→        <button onClick={submit} disabled={saving} className="py-2.5 px-6 rounded-md bg-primary text-primary-foreground border-none text-[13px] font-extrabold cursor-pointer disabled:opacity-70">{saving ? "جارٍ…" : "حفظ"}</button>
   492→      </div>
   493→    </div>
   494→  );
   495→}
