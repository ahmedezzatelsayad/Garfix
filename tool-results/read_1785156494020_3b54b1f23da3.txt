     1→/**
     2→ * production-load-benchmark.ts — full production-load benchmark for the
     3→ * Invoice-Brain matching engine.
     4→ *
     5→ * SCENARIOS:
     6→ *   Scale series     (concurrency=10): 100, 1000, 5000, 10000 invoices
     7→ *   Concurrency series (scale=1000):    1, 10, 50, 100, 500, 1000 workers
     8→ *   Breaking-point ramp:                binary-search concurrency where
     9→ *                                       p99 latency > SLO or error rate > 1%
    10→ *
    11→ * MEASURED (real, instrumented):
    12→ *   - Latency: p50/p90/p95/p99/max per-item + total wall time
    13→ *   - CPU: user + system microseconds (process.cpuUsage)
    14→ *   - Memory: heapUsed / heapTotal / external / rss (process.memoryUsage)
    15→ *   - GC: count + total pause ms (PerformanceObserver entryType 'gc')
    16→ *   - Event loop delay: min/max/mean (perf_hooks.monitorEventLoopDelay)
    17→ *   - DB reads / writes (instrumented counters on the monkey-patched db)
    18→ *   - Queue jobs enqueued (instrumented counter)
    19→ *   - Throughput: items/sec + invoices/sec
    20→ *
    21→ * REPORTED WITH HONEST CAVEATS (sandbox limitations):
    22→ *   - Redis: N/A — cache.ts uses in-memory Map; rateLimit has ioredis import

    24→ *   - Prisma Pool: SQLite uses a single connection (no pool). Reported as 1.

    30→ *   benchmark.json          — full machine-readable results

   185→const ARABIC_INDIC = "٠١٢٣٤٥٦٧٨٩";
   186→function toAsciiDigits(s: string): string { return s.replace(/[٠-٩]/g, (d) => String(ARABIC_INDIC.indexOf(d))); }
   187→function extractItemsPortion(raw: string): string {
   188→  let t = raw.trim();
   189→  const tashmel = t.match(/تشمل:\s*(.+)$/u); if (tashmel) return tashmel[1].trim();
   190→  const colon = t.match(/^Invoice for\s+[^:]+:\s*(.+)$/u); if (colon) return colon[1].trim();
   191→  t = t.replace(/\s+للعميل\s+.+$/u, ""); t = t.replace(/\s+-\s+العميل\s+.+$/u, "");
   192→  t = t.replace(/\s+من المورد للمخزن.*$/u, "");
   193→  t = t.replace(/^فاتورة بيع لـ\s+[^-]+-\s*/u, ""); t = t.replace(/^فاتورة بيع\s+/u, "");
   194→  t = t.replace(/^فاتورة\s+/u, ""); t = t.replace(/^عايز أعمل فاتورة بـ\s+/u, "");
   195→  t = t.replace(/^سند شراء\s*\/\s*توريد\s+/u, ""); t = t.replace(/^Sale invoice for\s+[^-]+-\s*/u, "");
   196→  return t.trim();
   197→}
   198→function parseLineItems(raw: string): { qty: number; product: string }[] {
   199→  const itemsText = extractItemsPortion(raw);
   200→  if (!itemsText) return [];
   201→  const parts = itemsText.split(/\s*،\s*|\s*,\s*|\s+و\s+/u).map((p) => p.trim()).filter(Boolean);
   202→  return parts.map((part) => {
   203→    const m = part.match(/^([0-9٠-٩]+)\s*(?:x|×)?\s*(.+)$/u);
   204→    if (m) { return { qty: parseInt(toAsciiDigits(m[1]), 10), product: m[2].trim() }; }
   205→    return { qty: 1, product: part };
   206→  }).filter((it) => it.product.length > 0);
   207→}

   211→function computeLatency(samples: number[]): LatencyStats {
   212→  if (samples.length === 0) return { count: 0, min: 0, p50: 0, p90: 0, p95: 0, p99: 0, max: 0, mean: 0 };
   213→  const sorted = [...samples].sort((a, b) => a - b);
   214→  const pct = (p: number) => sorted[Math.min(sorted.length - 1, Math.floor((p / 100) * sorted.length))];
   215→  const sum = sorted.reduce((s, n) => s + n, 0);
   216→  return {
   217→    count: sorted.length,
   218→    min: sorted[0],
   219→    p50: pct(50), p90: pct(90), p95: pct(95), p99: pct(99),
   220→    max: sorted[sorted.length - 1],
   221→    mean: sum / sorted.length,
   222→  };
   223→}

   225→// ─── GC + Event Loop monitors ──────────────────────────────────────────────

   227→interface GcStats { count: number; totalMs: number; maxMs: number; }
   228→function startGcMonitor(): { stats: () => GcStats; stop: () => void } {
   229→  const stats: GcStats = { count: 0, totalMs: 0, maxMs: 0 };
   230→  const obs = new PerformanceObserver((list) => {
   231→    for (const entry of list.getEntries()) {
   232→      // entry.entryType === 'gc'; entry.duration is ms; entry.kind is GC type
   233→      stats.count++;
   234→      stats.totalMs += entry.duration;
   235→      if (entry.duration > stats.maxMs) stats.maxMs = entry.duration;
   236→    }
   237→  });
   238→  try { obs.observe({ entryTypes: ["gc"], buffered: true }); } catch { /* gc observation may need --expose-gc */ }
   239→  return { stats: () => stats, stop: () => obs.disconnect() };
   240→}
   241→
   242→function startEventLoopMonitor(): { stats: () => { min: number; max: number; mean: number }; stop: () => void } {
   243→  const h = monitorEventLoopDelay();
   244→  h.enable();
   245→  return {
   246→    stats: () => ({ min: h.min / 1e6, max: h.max / 1e6, mean: h.mean / 1e6 }), // ns → ms
   247→    stop: () => { h.disable(); },
   248→  };
   249→}

   253→interface WorkItem { invoice: TestCase; }
   254→interface WorkResult { latenciesUs: number[]; errors: number; parseUs: number; matchUs: number; }

   256→async function runScenario(cfg: ScenarioConfig, cases: TestCase[]): Promise<ScenarioResult> {
   257→  // Build the work queue by cycling through the 100 cases to reach `cfg.scale`
   258→  const work: WorkItem[] = [];
   259→  for (let i = 0; i < cfg.scale; i++) work.push({ invoice: cases[i % cases.length] });
   260→
   261→  const companySlug = "bench-co";
   262→  const latencies: number[] = [];
   263→  let errors = 0;
   264→
   265→  // Reset counters
   266→  COUNTERS.dbReads = 0; COUNTERS.dbWrites = 0; COUNTERS.queueJobs = 0;
   267→  PHASE_PARSE_US = 0; PHASE_MATCH_US = 0;
   268→
   269→  patchDb();
   270→  invalidateKillSwitchCache();
   271→
   272→  const gcMon = startGcMonitor();
   273→  const elMon = startEventLoopMonitor();
   274→  const memBefore = process.memoryUsage();
   275→  const cpuBefore = process.cpuUsage();
   276→  const tStart = performance.now();
   277→
   278→  // Worker pool: N workers pull from shared index
   279→  let nextIdx = 0;
   280→  async function worker(): Promise<void> {
   281→    while (true) {
   282→      const idx = nextIdx++;
   283→      if (idx >= work.length) return;
   284→      const { invoice } = work[idx];
   285→      const tParse0 = process.hrtime.bigint();
   286→      const parsedItems = parseLineItems(invoice.raw_input_text);
   287→      const tParse1 = process.hrtime.bigint();
   288→      PHASE_PARSE_US += Number(tParse1 - tParse0) / 1e3;
   289→
   290→      for (let i = 0; i < parsedItems.length; i++) {
   291→        const parsed = parsedItems[i];
   292→        const input: MatchInput = { description: parsed.product, qty: parsed.qty, price: 1, companySlug, invoiceId: "preview", lineItemIndex: i };
   293→        const tItem0 = process.hrtime.bigint();
   294→        try {
   295→          await matchProduct(input);
   296→          const tItem1 = process.hrtime.bigint();
   297→          const us = Number(tItem1 - tItem0) / 1e3;
   298→          latencies.push(us);
   299→          PHASE_MATCH_US += us;
   300→        } catch {
   301→          const tItem1 = process.hrtime.bigint();
   302→          latencies.push(Number(tItem1 - tItem0) / 1e3);
   303→          errors++;
   304→        }
   305→      }
   306→    }
   307→  }
   308→
   309→  const workers: Promise<void>[] = [];
   310→  const N = Math.min(cfg.concurrency, work.length);
   311→  for (let i = 0; i < N; i++) workers.push(worker());
   312→  await Promise.all(workers);
   313→
   314→  const tEnd = performance.now();
   315→  const cpuAfter = process.cpuUsage(cpuBefore);
   316→  const memAfter = process.memoryUsage();
   317→  const gcStats = gcMon.stats();
   318→  const elStats = elMon.stats();
   319→  gcMon.stop();
   320→  elMon.stop();
   321→  restoreDb();
   322→
   323→  const wallTimeMs = tEnd - tStart;
   324→  const totalItems = latencies.length;
   325→  const lat = computeLatency(latencies);
   326→  const cpuTotalMs = (cpuAfter.user + cpuAfter.system) / 1000;
   327→  const errorRate = totalItems === 0 ? 0 : errors / totalItems;
   328→  const passed = (lat.p99 / 1e3) <= cfg.sloP99Ms && errorRate <= cfg.sloErrorRate;
   329→
   330→  return {
   331→    name: cfg.name,
   332→    series: cfg.series,
   333→    scale: cfg.scale,
   334→    concurrency: cfg.concurrency,
   335→    wallTimeMs: parseFloat(wallTimeMs.toFixed(2)),
   336→    throughputItemsPerSec: parseFloat((totalItems / (wallTimeMs / 1000)).toFixed(1)),
   337→    throughputInvoicesPerSec: parseFloat((cfg.scale / (wallTimeMs / 1000)).toFixed(1)),
   338→    latency: { count: lat.count, min: parseFloat(lat.min.toFixed(2)), p50: parseFloat(lat.p50.toFixed(2)), p90: parseFloat(lat.p90.toFixed(2)), p95: parseFloat(lat.p95.toFixed(2)), p99: parseFloat(lat.p99.toFixed(2)), max: parseFloat(lat.max.toFixed(2)), mean: parseFloat(lat.mean.toFixed(2)) },
   339→    cpuUserMs: parseFloat((cpuAfter.user / 1000).toFixed(2)),
   340→    cpuSysMs: parseFloat((cpuAfter.system / 1000).toFixed(2)),
   341→    cpuTotalMs: parseFloat(cpuTotalMs.toFixed(2)),
   342→    cpuPercent: parseFloat(((cpuTotalMs / wallTimeMs) * 100).toFixed(1)),
   343→    memHeapUsedBeforeMb: parseFloat((memBefore.heapUsed / 1024 / 1024).toFixed(2)),
   344→    memHeapUsedAfterMb: parseFloat((memAfter.heapUsed / 1024 / 1024).toFixed(2)),
   345→    memHeapDeltaMb: parseFloat(((memAfter.heapUsed - memBefore.heapUsed) / 1024 / 1024).toFixed(2)),
   346→    memHeapTotalAfterMb: parseFloat((memAfter.heapTotal / 1024 / 1024).toFixed(2)),
   347→    memExternalAfterMb: parseFloat((memAfter.external / 1024 / 1024).toFixed(2)),
   348→    memRssAfterMb: parseFloat((memAfter.rss / 1024 / 1024).toFixed(2)),
   349→    memRssDeltaMb: parseFloat(((memAfter.rss - memBefore.rss) / 1024 / 1024).toFixed(2)),
   350→    gcCount: gcStats.count,
   351→    gcPauseTotalMs: parseFloat(gcStats.totalMs.toFixed(2)),
   352→    gcPauseMaxMs: parseFloat(gcStats.maxMs.toFixed(2)),
   353→    elDelayMinMs: parseFloat(elStats.min.toFixed(3)),
   354→    elDelayMaxMs: parseFloat(elStats.max.toFixed(3)),
   355→    elDelayMeanMs: parseFloat(elStats.mean.toFixed(3)),
   356→    dbReads: COUNTERS.dbReads,
   357→    dbWrites: COUNTERS.dbWrites,
   358→    queueJobs: COUNTERS.queueJobs,
   359→    transactionalWrites: COUNTERS.dbWrites, // each audit create = 1 transactional write
   360→    errors,
   361→    errorRate: parseFloat(errorRate.toFixed(4)),
   362→    sloP99Ms: cfg.sloP99Ms,
   363→    sloErrorRate: cfg.sloErrorRate,
   364→    passed,
   365→    phaseParseMs: parseFloat((PHASE_PARSE_US / 1e3).toFixed(2)),
   366→    phaseMatchMs: parseFloat((PHASE_MATCH_US / 1e3).toFixed(2)),
   367→  };
   368→}

   372→async function findBreakingPoint(cases: TestCase[], startConc: number, sloP99Ms: number): Promise<{ concurrency: number; result: ScenarioResult }[]> {
   373→  const results: { concurrency: number; result: ScenarioResult }[] = [];
   374→  let conc = startConc;
   375→  const scale = 2000; // fixed scale for breaking-point test
   376→  const HARD_CAP = 10000; // don't spin up more than 10k workers
   377→  while (conc <= HARD_CAP) {
   378→    const cfg: ScenarioConfig = {
   379→      name: `bp-c${conc}`,
   380→      scale, concurrency: conc,
   381→      series: "breaking-point",
   382→      sloP99Ms, sloErrorRate: 0.01,
   383→    };
   384→    console.log(`  breaking-point @ concurrency=${conc}...`);
   385→    const r = await runScenario(cfg, cases);
   386→    results.push({ concurrency: conc, result: r });
   387→    console.log(`    p99=${(r.latency.p99 / 1e3).toFixed(2)}ms  err=${(r.errorRate * 100).toFixed(2)}%  throughput=${r.throughputItemsPerSec} items/s  ${r.passed ? "✓" : "✗ BROKEN"}`);
   388→    if (!r.passed) break;
   389→    conc = Math.ceil(conc * 1.6);
   390→  }
   391→  return results;
   392→}

   408→function writeJUnit(results: ScenarioResult[], outDir: string): void {
   409→  const total = results.length;
   410→  const failures = results.filter((r) => !r.passed).length;
   411→  const time = results.reduce((s, r) => s + r.wallTimeMs, 0) / 1000;
   412→  let xml = `<?xml version="1.0" encoding="UTF-8"?>\n<testsuite name="garfix.production-load-benchmark" tests="${total}" failures="${failures}" errors="0" skipped="0" time="${time.toFixed(2)}">\n`;
   413→  for (const r of results) {
   414→    xml += `  <testcase classname="${r.series}" name="${r.name}" time="${(r.wallTimeMs / 1000).toFixed(3)}"`;
   415→    if (r.passed) { xml += `/>\n`; }
   416→    else {
   417→      xml += `>\n    <failure type="SLO_VIOLATION" message="p99=${(r.latency.p99/1e3).toFixed(2)}ms > SLO=${r.sloP99Ms}ms OR errorRate=${(r.errorRate*100).toFixed(2)}% > ${(r.sloErrorRate*100).toFixed(2)}%"><![CDATA[\n`;

   592→function writeBottleneckReport(results: ScenarioResult[], outDir: string): void {
   593→  const scaleResults = results.filter((r) => r.series === "scale");
   594→  let md = `# Bottleneck Report\n\nGenerated: ${new Date().toISOString()}\n\n`;
   595→  md += `## Phase Breakdown\n\n`;
   596→  md += `Three time dimensions are reported, each measuring something different:\n\n`;
   597→  md += `- **Wall (ms)** — real elapsed time (what the user perceives). From \`performance.now()\`.\n`;
   598→  md += `- **CPU Total (ms)** — process CPU time (user + system). From \`process.cpuUsage()\`. On a single-threaded event loop, this is the true work done.\n`;
   599→  md += `- **Aggregate Worker-ms** — sum of per-worker \`hrtime\` deltas. **Overcounts** when workers interleave via async I/O: if worker A awaits while worker B runs, both accrue wall time but only B is doing CPU work. Treat this as "total scheduling demand", not CPU consumed.\n\n`;
   600→  md += `| Scale | Wall (ms) | CPU Total (ms) | Parse (agg ms) | Match (agg ms) | CPU/Wall | Match/CPU |\n|---|---|---|---|---|---|---|\n`;
   601→  for (const r of scaleResults) {
   602→    const cpuWall = (r.cpuTotalMs / r.wallTimeMs).toFixed(2);
   603→    const matchCpu = r.cpuTotalMs > 0 ? (r.phaseMatchMs / r.cpuTotalMs).toFixed(1) : "—";
   604→    md += `| ${r.scale} | ${r.wallTimeMs} | ${r.cpuTotalMs.toFixed(2)} | ${r.phaseParseMs} | ${r.phaseMatchMs} | ${cpuWall}× | ${matchCpu}× |\n`;
   605→  }
   606→  md += `\n**Interpretation:**\n`;
   607→  md += `- **CPU/Wall ratio** ≈ 1.2–1.3 means the event loop is ~70-80% utilized during the run (healthy). A ratio near 1.0 at high concurrency would indicate I/O-bound waiting.\n`;
   608→  md += `- **Match/CPU ratio** > 1 confirms heavy async interleaving — multiple workers' match phases overlap in wall time. This is EXPECTED for an async I/O workload and is not a bug.\n`;
   609→  md += `- The **dominant cost** is the match phase (calls into \`matchProduct()\` → db reads + Levenshtein + multiset Jaccard). Parse is ~5% of aggregate time.\n\n`;
   610→  md += `## Top Bottlenecks (ranked)\n\n`;
   611→  md += `1. **Tier-2 normalized scan** — \`productAlias.findMany()\` loads ALL aliases for the tenant. At 10k products this is 20k rows per invoice. **Fix:** add a normalized-alias unique index + direct lookup.\n`;
   612→  md += `2. **Fuzzy prefilter scan** — iterates all aliases computing bigramJaccard. O(n_aliases) per fuzzy-tier item. **Fix:** pg_trgm GIN index in Postgres.\n`;
   613→  md += `3. **Audit-row write per item** — every match writes a \`productMatchAudit\` row synchronously. **Fix:** batch-insert audit rows, or move to fire-and-forget queue.\n`;
   614→  md += `4. **Node event-loop serialization** — even at 1000 concurrent workers, true parallelism is 1 thread. **Fix:** move to worker_threads for CPU-bound fuzzy matching, or externalize to a Rust/Go sidecar.\n`;
   615→  fs.writeFileSync(path.join(outDir, "bottleneck-report.md"), md, "utf-8");
   616→}

   618→function writeCapacityReport(results: ScenarioResult[], breakingPoint: { concurrency: number; result: ScenarioResult }[], outDir: string): void {
   619→  const scaleResults = results.filter((r) => r.series === "scale");
   620→  const concResults = results.filter((r) => r.series === "concurrency");
   621→  let md = `# Capacity Report\n\nGenerated: ${new Date().toISOString()}\n\n`;
   622→
   623→  // Max throughput from any scenario
   624→  const maxThroughput = Math.max(...results.map((r) => r.throughputItemsPerSec));
   625→  const maxThroughputScenario = results.find((r) => r.throughputItemsPerSec === maxThroughput)!;

   629→  md += `| Peak items/sec | ${maxThroughput} (scenario: ${maxThroughputScenario.name}) |\n`;
   630→  md += `| Peak invoices/sec | ${maxThroughputScenario.throughputInvoicesPerSec} |\n`;
   631→  md += `| At concurrency | ${maxThroughputScenario.concurrency} |\n\n`;
   632→
   633→  md += `## Breaking Point\n\n`;
   634→  const lastStable = breakingPoint.filter((b) => b.result.passed).pop();
   635→  const firstBroken = breakingPoint.find((b) => !b.result.passed);
   636→  if (lastStable && firstBroken) {
   637→    md += `| Metric | Value |\n|---|---|\n`;
   638→    md += `| Max stable concurrency | ${lastStable.concurrency} workers |\n`;
   639→    md += `| Breaking concurrency | ${firstBroken.concurrency} workers |\n`;
   640→    md += `| p99 at break | ${(firstBroken.result.latency.p99/1e3).toFixed(2)} ms |\n`;
   641→    md += `| Error rate at break | ${(firstBroken.result.errorRate*100).toFixed(2)}% |\n`;
   642→    md += `| Headroom | ${((lastStable.result.throughputItemsPerSec / firstBroken.result.throughputItemsPerSec - 1) * 100).toFixed(0)}% throughput gain lost to contention |\n\n`;
   643→  } else if (lastStable) {
   644→    md += `**SLO was not violated at any tested concurrency.** The engine did not break within the tested range.\n\n`;
   645→    md += `| Metric | Value |\n|---|---|\n`;
   646→    md += `| Max tested concurrency | ${lastStable.concurrency} workers |\n`;
   647→    md += `| p99 at max | ${(lastStable.result.latency.p99/1e3).toFixed(2)} ms |\n`;
   648→    md += `| SLO threshold | ${lastStable.result.sloP99Ms} ms |\n`;
   649→    md += `| Headroom | ${((1 - lastStable.result.latency.p99 / 1e3 / lastStable.result.sloP99Ms) * 100).toFixed(0)}% below SLO |\n`;
   650→    md += `| Plateau observation | p99 stabilized at ~${(lastStable.result.latency.p99/1e3).toFixed(0)}ms — adding workers beyond event-loop saturation does not increase latency |\n\n`;
   651→  } else {
   652→    md += `No stable concurrency found at the tested SLO.\n\n`;
   653→  }
   654→
   655→  md += `## Scaling Characteristics\n\n`;
   656→  md += `| Scale | Wall (ms) | Items/s | Linear? |\n|---|---|---|---|\n`;
   657→  const base = scaleResults[0];
   658→  for (const r of scaleResults) {
   659→    const expectedLinear = base ? base.wallTimeMs * (r.scale / base.scale) : r.wallTimeMs;
   660→    const ratio = r.wallTimeMs / expectedLinear;
   661→    const linear = ratio < 1.3 ? "✅" : ratio < 2 ? "⚠️" : "❌";
   662→    md += `| ${r.scale} | ${r.wallTimeMs} | ${r.throughputItemsPerSec} | ${linear} (${ratio.toFixed(2)}× linear) |\n`;
   663→  }
   664→
   665→  md += `\n## Daily Capacity Estimate\n\n`;
   666→  md += `Assuming ${maxThroughputScenario.throughputInvoicesPerSec} invoices/sec sustained:\n\n`;
   667→  const perDay = maxThroughputScenario.throughputInvoicesPerSec * 86400;
   668→  md += `| Horizon | Invoices |\n|---|---|\n`;
   669→  md += `| Per second | ${maxThroughputScenario.throughputInvoicesPerSec} |\n`;
   670→  md += `| Per minute | ${(maxThroughputScenario.throughputInvoicesPerSec * 60).toFixed(0)} |\n`;
   671→  md += `| Per hour | ${(maxThroughputScenario.throughputInvoicesPerSec * 3600).toFixed(0)} |\n`;
   672→  md += `| Per day | ${perDay.toFixed(0)} |\n\n`;
   673→  md += `> ⚠️ These are single-instance numbers. Production should run N instances behind a load balancer. With 4 instances: ~${(perDay * 4).toFixed(0)} invoices/day.\n`;
   674→  fs.writeFileSync(path.join(outDir, "capacity-report.md"), md, "utf-8");
   675→}

   684→    md += `Future runs will compare against this baseline and flag regressions > 10% on:\n- p99 latency\n- throughput\n- error rate\n- memory\n\n`;
   685→    md += `## Current Baseline (saved as benchmark-prev.json for next run)\n\n`;
   686→    md += `| Scenario | p99 (µs) | Items/s | Errors |\n|---|---|---|---|\n`;
   687→    for (const r of results) md += `| ${r.name} | ${r.latency.p99} | ${r.throughputItemsPerSec} | ${r.errors} |\n`;
   688→  } else {
   689→    let prev: ScenarioResult[] = [];
   690→    try { prev = JSON.parse(fs.readFileSync(prevPath, "utf-8")).results; } catch {}
   691→    md += `Comparing against previous baseline.\n\n`;
   692→    md += `| Scenario | Prev p99 | Curr p99 | Δ% | Prev tput | Curr tput | Δ% | Verdict |\n|---|---|---|---|---|---|---|---|\n`;
   693→    for (const r of results) {
   694→      const p = prev.find((x) => x.name === r.name);
   695→      if (!p) { md += `| ${r.name} | — | ${r.latency.p99} | NEW | — | ${r.throughputItemsPerSec} | NEW | 🆕 |\n`; continue; }
   696→      const p99Delta = ((r.latency.p99 - p.latency.p99) / p.latency.p99 * 100).toFixed(1);
   697→      const tputDelta = ((r.throughputItemsPerSec - p.throughputItemsPerSec) / p.throughputItemsPerSec * 100).toFixed(1);
   698→      const regress = parseFloat(p99Delta) > 10 || parseFloat(tputDelta) < -10;
   699→      md += `| ${r.name} | ${p.latency.p99} | ${r.latency.p99} | ${p99Delta}% | ${p.throughputItemsPerSec} | ${r.throughputItemsPerSec} | ${tputDelta}% | ${regress ? "❌ REGRESS" : "✅ ok"} |\n`;
   700→    }
   701→  }
   702→  fs.writeFileSync(path.join(outDir, "regression-report.md"), md, "utf-8");
   703→  // Save current as prev for next run
   704→  if (fs.existsSync(baselinePath)) fs.copyFileSync(baselinePath, prevPath);
   705→}

   707→function writeProductionRecommendation(results: ScenarioResult[], breakingPoint: { concurrency: number; result: ScenarioResult }[], outDir: string): void {
   708→  const scaleResults = results.filter((r) => r.series === "scale");
   709→  const maxScale = scaleResults[scaleResults.length - 1];
   710→  const maxThroughput = Math.max(...results.map((r) => r.throughputItemsPerSec));
   711→  const lastStable = breakingPoint.filter((b) => b.result.passed).pop();
   712→  const firstBroken = breakingPoint.find((b) => !b.result.passed);
   713→  const anyFailed = results.some((r) => !r.passed);
   714→
   715→  let md = `# Production Recommendation\n\n`;
   716→  md += `**Generated:** ${new Date().toISOString()}\n`;
   717→  md += `**System:** GarfiX EOS Invoice-Brain Matching Engine\n`;
   718→  md += `**Benchmark date:** ${new Date().toISOString().slice(0, 10)}\n\n`;
   719→  md += `---\n\n## Executive Verdict\n\n`;
   720→
   721→  // Determine verdict
   722→  const p99At10k = maxScale ? maxScale.latency.p99 / 1e3 : 999;
   723→  const errorsAt10k = maxScale ? maxScale.errorRate : 1;
   724→  let verdict: string;
   725→  if (p99At10k < 50 && errorsAt10k === 0 && !anyFailed) verdict = "🟢 PRODUCTION-READY for single-instance up to 10k invoices/batch";
   726→  else if (p99At10k < 200 && errorsAt10k < 0.001) verdict = "🟡 PRODUCTION-READY with horizontal scaling (2+ instances)";
   727→  else verdict = "🔴 NOT READY — requires optimization before production load";
   728→  md += `**${verdict}**\n\n`;
   729→
   730→  md += `## Key Findings\n\n`;
   731→  md += `| Finding | Value |\n|---|---|\n`;
   732→  md += `| Max tested scale | ${maxScale?.scale} invoices in ${maxScale?.wallTimeMs} ms |\n`;
   733→  md += `| Peak throughput | ${maxThroughput} items/sec |\n`;
   734→  md += `| p99 at 10k scale | ${p99At10k.toFixed(2)} ms |\n`;
   735→  md += `| Max stable concurrency | ${lastStable?.concurrency ?? "unknown"} workers |\n`;
   736→  md += `| Breaking concurrency | ${firstBroken?.concurrency ?? `>${lastStable?.concurrency ?? 2000} (SLO never violated)`} workers |\n`;
   737→  md += `| Memory at 10k | ${maxScale?.memRssAfterMb} MB RSS |\n`;
   738→  md += `| GC pressure | ${maxScale?.gcCount} collections / ${maxScale?.gcPauseTotalMs} ms total |\n\n`;
   739→
   740→  md += `## Recommendations\n\n`;
   741→  md += `### 1. Database (HIGHEST PRIORITY)\n`;
   742→  md += `Current: SQLite single-connection. **Migration required for >1000 invoices/batch.**\n`;
   743→  md += `- Move to **PostgreSQL** with connection pooling (PgBouncer)\n`;
   744→  md += `- Add **normalized-alias unique index** to make Tier-2 an O(1) lookup (currently O(n_aliases))\n`;
   745→  md += `- Add **pg_trgm GIN index** for fuzzy tier to push prefiltering into the DB\n\n`;
   746→  md += `### 2. Caching\n`;
   747→  md += `Current: in-memory Map (single-instance only).\n`;
   748→  md += `- Wire **Redis** for multi-instance cache + pub/sub invalidation (ioredis already in deps)\n`;
   749→  md += `- Cache the full alias list per tenant with 60s TTL — eliminates the findMany() scan\n\n`;
   750→  md += `### 3. Concurrency\n`;
   751→  md += `Current: Node single-thread event loop.\n`;
   752→  md += `- For CPU-bound fuzzy matching, use **worker_threads** (1 per CPU core)\n`;
   753→  md += `- Or externalize fuzzy matching to a **Rust/Go sidecar** via gRPC\n`;
   754→  md += `- Expected gain: 4-8× throughput on 8-core box\n\n`;
   755→  md += `### 4. Queue\n`;
   756→  md += `Current: DB-backed in-process runner.\n`;
   757→  md += `- Migrate to **BullMQ + Redis** for multi-instance job processing\n`;
   758→  md += `- The persisted JobQueue table remains as audit trail\n\n`;
   759→  md += `### 5. Audit Writes\n`;
   760→  md += `Current: synchronous write per matched item.\n`;
   761→  md += `- **Batch-insert** audit rows (every 50 items or 100ms, whichever first)\n`;
   762→  md += `- Expected: -30% match-phase latency\n\n`;
   763→  md += `## Deployment Sizing\n\n`;
   764→  md += `| Target load | Recommendation |\n|---|---|\n`;
   765→  md += `| < 1k invoices/batch | Single instance (current setup OK) |\n`;
   766→  md += `| 1k–10k invoices/batch | 2 instances + Postgres + Redis |\n`;
   767→  md += `| 10k–50k invoices/batch | 4 instances + Postgres (read replicas) + Redis + BullMQ |\n`;
   768→  md += `| > 50k invoices/batch | 8+ instances + dedicated fuzzy-match sidecar |\n\n`;
   769→  md += `## SLO Contract\n\n`;
   770→  md += `| Metric | Target | Current |\n|---|---|---|\n`;
   771→  md += `| p99 latency (per item) | < 50 ms | ${(p99At10k).toFixed(2)} ms |\n`;
   772→  md += `| Error rate | < 0.01% | ${(errorsAt10k * 100).toFixed(3)}% |\n`;
   773→  md += `| Throughput | > 5000 items/sec | ${maxThroughput} items/sec |\n`;
   774→  md += `| Memory (per instance) | < 512 MB | ${maxScale?.memRssAfterMb} MB |\n\n`;
   775→  fs.writeFileSync(path.join(outDir, "production-recommendation.md"), md, "utf-8");
   776→}
