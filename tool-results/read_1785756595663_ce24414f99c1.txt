    93→function formatNumber(num: number): string {
    94→  if (num >= 1_000_000) return `${(num / 1_000_000).toFixed(1)}M`;
    95→  if (num >= 1_000) return `${(num / 1_000).toFixed(1)}K`;
    96→  return num.toString();
    97→}
    98→
    99→function formatTime(ms: number): string {
   100→  if (ms < 1000) return `${ms}ms`;
   101→  return `${(ms / 1000).toFixed(1)}s`;
   102→}
   103→
   104→function getStatusColor(status: string): string {
   105→  switch (status) {
   106→    case 'healthy': return 'text-green-600 bg-green-50 border-green-200';
   107→    case 'degraded': return 'text-yellow-600 bg-yellow-50 border-yellow-200';
   108→    case 'critical': return 'text-red-600 bg-red-50 border-red-200';
   109→    default: return 'text-gray-600 bg-gray-50 border-gray-200';
   110→  }
   111→}
   112→
   113→function getAlertIcon(level: string): string {
   114→  switch (level) {
   115→    case 'info': return 'ℹ️';
   116→    case 'warning': return '⚠️';
   117→    case 'error': return '❌';
   118→    case 'critical': return '🚨';
   119→    default: return '📋';
   120→  }
   121→}
   122→
   123→function getAlertBg(level: string): string {
   124→  switch (level) {
   125→    case 'info': return 'bg-blue-50 border-blue-200';
   126→    case 'warning': return 'bg-yellow-50 border-yellow-200';
   127→    case 'error': return 'bg-red-50 border-red-200';
   128→    case 'critical': return 'bg-red-100 border-red-300';
   129→    default: return 'bg-gray-50 border-gray-200';
   130→  }

   332→function UtilizationGauge({ value, max = 100 }: { value: number; max?: number }) {
   333→  const percentage = Math.min((value / max) * 100, 100);
   334→  const rotation = (percentage / 100) * 180 - 90; // -90 to 90 degrees
   335→  
   336→  let color = '#22c55e'; // green
   337→  if (percentage > 75) color = '#ef4444'; // red
   338→  else if (percentage > 50) color = '#eab308'; // yellow
   339→
   340→  return (
   341→    <div className="relative w-32 h-16 mx-auto overflow-hidden">
   342→      <svg viewBox="0 0 100 50" className="w-full h-full">
   343→        {/* Background arc */}
   344→        <path
   345→          d="M 10 45 A 40 40 0 0 1 90 45"
   346→          fill="none"
   347→          stroke="#e5e7eb"
   348→          strokeWidth="8"
   349→          strokeLinecap="round"
   350→        />
   351→        {/* Value arc */}
   352→        <path
   353→          d="M 10 45 A 40 40 0 0 1 90 45"
   354→          fill="none"
   355→          stroke={color}
   356→          strokeWidth="8"
   357→          strokeLinecap="round"
   358→          strokeDasharray={`${percentage * 1.256} 126`}
   359→          transform="rotate(-90 50 45)"
   360→        />
   361→      </svg>
   362→      <div className="absolute inset-0 flex items-end justify-center pb-1">
   363→        <span className="text-lg font-bold" style={{ color }}>
   364→          {Math.round(percentage)}%
   365→        </span>
   366→      </div>
   367→    </div>
   368→  );
   369→}

   373→export default function AIDashboardPage() {
   374→  const [data, setData] = useState<AIMetricsData | null>(null);
   375→  const [loading, setLoading] = useState(true);
   376→  const [error, setError] = useState<string | null>(null);
   377→  const [lastUpdate, setLastUpdate] = useState<Date | null>(null);
   378→  const [autoRefresh, setAutoRefresh] = useState(true);
   379→  const [activeTab, setActiveTab] = useState<'overview' | 'keys' | 'workers' | 'queue'>('overview');
   380→
   381→  // Fetch metrics
   382→  const fetchMetrics = useCallback(async () => {
   383→    try {
   384→      setError(null);
   385→      const response = await fetch('/api/ai/metrics');
   386→      
   387→      if (!response.ok) {
   388→        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
   389→      }
   390→      
   391→      const result: AIMetricsData = await response.json();
   392→      
   393→      if (result.success) {
   394→        setData(result);
   395→        setLastUpdate(new Date());
   396→      } else {
   397→        throw new Error((result as any).error || 'Failed to fetch metrics');
   398→      }
   399→    } catch (err) {
   400→      setError(err instanceof Error ? err.message : 'Unknown error');
   401→    } finally {
   402→      setLoading(false);
   403→    }
   404→  }, []);
   405→
   406→  // Initial fetch + auto-refresh
   407→  useEffect(() => {
   408→    fetchMetrics();
   409→    
   410→    if (autoRefresh) {
   411→      const interval = setInterval(fetchMetrics, 10_000); // 10 seconds
   412→      return () => clearInterval(interval);
   413→    }
   414→  }, [fetchMetrics, autoRefresh]);
   415→
   416→  // Manual refresh handler
   417→  const handleRefresh = () => {
   418→    setLoading(true);
   419→    fetchMetrics();
   420→  };
   421→
   422→  // Reset quotas handler
   423→  const handleResetQuotas = async () => {
   424→    if (!confirm('هل أنت متأكد من إعادة تعيين الحصص اليومية؟')) return;
   425→    
   426→    try {
   427→      const response = await fetch('/api/ai/metrics?action=reset-quotas', { method: 'POST' });
   428→      const result = await response.json();
   429→      
   430→      if (result.success) {
   431→        alert('✅ تم إعادة تعيين الحصص بنجاح');
   432→        fetchMetrics();
   433→      } else {
   434→        alert(`❌ خطأ: ${result.error}`);
   435→      }
   436→    } catch (err) {
   437→      alert(`❌ خطأ: ${err instanceof Error ? err.message : 'Unknown error'}`);
   438→    }
   439→  };
   440→
   441→  // Loading state
   442→  if (loading && !data) {
   443→    return (
   444→      <main className="min-h-screen bg-gray-50 p-4 md:p-8" dir="rtl">
   445→        <div className="max-w-7xl mx-auto">
   446→          <div className="flex items-center justify-center min-h-[60vh]">
   447→            <div className="text-center">
   448→              <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-violet-600 mx-auto mb-4"></div>
   449→              <p className="text-gray-500 font-medium">جاري تحميل بيانات الذكاء الاصطناعي...</p>
   450→              <p className="text-sm text-gray-400 mt-2">AI Metrics Loading...</p>
   451→            </div>
   452→          </div>
   453→        </div>
   454→      </main>
   455→    );
   456→  }
   457→
   458→  // Error state
   459→  if (error && !data) {
   460→    return (
   461→      <main className="min-h-screen bg-gray-50 p-4 md:p-8" dir="rtl">
   462→        <div className="max-w-7xl mx-auto">
   463→          <div className="flex items-center justify-center min-h-[60vh]">
   464→            <div className="text-center bg-red-50 rounded-xl p-8 max-w-md">
   465→              <div className="text-5xl mb-4">😵</div>
   466→              <h2 className="text-xl font-bold text-red-800 mb-2">خطأ في تحميل البيانات</h2>
   467→              <p className="text-red-600 mb-4">{error}</p>
   468→              <button
   469→                onClick={handleRefresh}
   470→                className="px-6 py-2 bg-violet-600 text-white rounded-lg hover:bg-violet-700 transition-colors"
   471→              >
   472→                🔄 إعادة المحاولة
   473→              </button>
   474→            </div>
   475→          </div>
   476→        </div>
   477→      </main>
   478→    );
   479→  }
   480→
   481→  return (
   482→    <main className="min-h-screen bg-gray-50 p-4 md:p-8" dir="rtl">
   483→      <div className="max-w-7xl mx-auto space-y-6">
   484→        
   485→        {/* ── Header ──────────────────────────────────────── */}
   486→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   487→          <div>
   488→            <h1 className="text-2xl md:text-3xl font-bold text-gray-900 flex items-center gap-3">
   489→              <span className="text-4xl">🤖</span>
   490→              لوحة تحكم الذكاء الاصطناعي
   491→            </h1>
   492→            <p className="text-gray-500 mt-1">GarfiX AI Enterprise Dashboard</p>
   493→          </div>
   494→          
   495→          <div className="flex items-center gap-3">
   496→            {/* Auto-refresh toggle */}
   497→            <label className="flex items-center gap-2 bg-white px-3 py-2 rounded-lg border cursor-pointer">
   498→              <input
   499→                type="checkbox"
   500→                checked={autoRefresh}
   501→                onChange={(e) => setAutoRefresh(e.target.checked)}
   502→                className="rounded text-violet-600"
   503→              />
   504→              <span className="text-sm text-gray-600">تحديث تلقائي</span>
   505→            </label>
   506→            
   507→            {/* Refresh button */}
   508→            <button
   509→              onClick={handleRefresh}
   510→              disabled={loading}
   511→              className="px-4 py-2 bg-violet-600 text-white rounded-lg hover:bg-violet-700 disabled:opacity-50 transition-colors flex items-center gap-2"
   512→            >
   513→              <span className={loading ? 'animate-spin' : ''}>🔄</span>
   514→              تحديث
   515→            </button>
   516→
   517→            {/* Reset button (admin only) */}
   518→            <button
   519→              onClick={handleResetQuotas}
   520→              className="px-4 py-2 bg-gray-200 text-gray-700 rounded-lg hover:bg-gray-300 transition-colors text-sm"
   521→            >
   522→              ↺ إعادة الحصص
   523→            </button>
   524→          </div>
   525→        </div>
   526→
   527→        {/* Last update time */}
   528→        {lastUpdate && (
   529→          <p className="text-xs text-gray-400">
   530→            آخر تحديث: {lastUpdate.toLocaleTimeString('ar-EG')} • 
   531→            {autoRefresh && ' تحديث كل 10 ثوانٍ'}
   532→          </p>
   533→        )}
   534→
   535→        {/* ── Pool Status Banner ───────────────────────── */}
   536→        {data?.data.pool && (
