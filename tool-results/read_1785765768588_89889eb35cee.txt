    22→/* ─── Interfaces ──────────────────────────────────────────────────────────── */
    23→interface Voucher {
    24→  id: number; voucherType: string; date: string; amount: number;
    25→  currency: string; payee: string; payer: string; status: string;
    26→  amountTextAr?: string;
    27→}
    28→interface Quotation {
    29→  id: number; clientName: string; date: string; validUntil: string;
    30→  lineItems: LineItem[]; totalAmount: number; status: string;
    31→}
    32→interface PurchaseOrder {
    33→  id: number; supplierName: string; date: string; expectedDelivery: string;
    34→  lineItems: LineItem[]; totalAmount: number; status: string;
    35→}
    36→interface LineItem { id?: number; description: string; quantity: number; unitPrice: number; total: number; }
    37→interface OpeningBalance { id: number; accountId: number; accountCode: string; accountNameAr: string; amount: number; posted: boolean; }
    38→interface Commission { id: number; salesperson: string; totalSales: number; commissionAmount: number; posted: boolean; }
    39→interface ProfitDistribution { id: number; partnerName: string; ownershipPercent: number; profitShare: number; posted: boolean; }
    40→
    41→type Tab = "vouchers" | "quotations" | "purchase-orders" | "opening-balances" | "commissions" | "profit-distribution";
    42→

    45→const tdStyle = "py-2.5 px-3 text-[13px]";

    47→const inputStyle = "w-full py-2 px-3 rounded-sm bg-background border border-border text-foreground text-[13px] outline-none focus-ring";
    48→const labelStyle = "block text-[11px] font-semibold text-muted-foreground mb-1";
    49→// DS v4.0: Added focus-ring for selects
    50→const selectStyle = "w-full py-2 px-3 rounded-sm bg-background border border-border text-foreground text-[13px] outline-none cursor-pointer focus-ring";
    51→

    56→/* ─── Main Component ────────────────────────────────────────────────────────── */
    57→export function VouchersDetailView() {
    58→  const { activeCompany } = useBrand();
    59→  const [tab, setTab] = useState<Tab>("vouchers");
    60→  const [showForm, setShowForm] = useState(false);
    61→
    62→  /* Voucher form */
    63→  const [vType, setVType] = useState("receipt");
    64→  const [vDate, setVDate] = useState("");
    65→  const [vAmount, setVAmount] = useState("");
    66→  const [vCurrency, setVCurrency] = useState("KWD");
    67→  const [vPayee, setVPayee] = useState("");
    68→  const [vPayer, setVPayer] = useState("");
    69→
    70→  /* Quotation form */
    71→  const [qClient, setQClient] = useState("");
    72→  const [qDate, setQDate] = useState("");
    73→  const [qValidUntil, setQValidUntil] = useState("");
    74→  const [qLineItems, setQLineItems] = useState<LineItem[]>([{ description: "", quantity: 1, unitPrice: 0, total: 0 }]);
    75→
    76→  /* PO form */
    77→  const [poSupplier, setPoSupplier] = useState("");
    78→  const [poDate, setPoDate] = useState("");
    79→  const [poDelivery, setPoDelivery] = useState("");
    80→  const [poLineItems, setPoLineItems] = useState<LineItem[]>([{ description: "", quantity: 1, unitPrice: 0, total: 0 }]);
    81→
    82→  /* Opening Balance form */
    83→  const [obAccountId, setObAccountId] = useState("");
    84→  const [obAmount, setObAmount] = useState("");
    85→
    86→  /* Commission dates */
    87→  const [commFrom, setCommFrom] = useState("");
    88→  const [commTo, setCommTo] = useState("");
    89→
    90→  const slug = activeCompany ? encodeURIComponent(activeCompany.slug) : "";
    91→
    92→  // TanStack Query hooks
    93→  const vouchersQuery = useVouchers(slug);
    94→  const quotationsQuery = useQuotations(slug);
    95→  const purchaseOrdersQuery = usePurchaseOrders(slug);
    96→  const openingBalancesQuery = useOpeningBalances(slug);
    97→  const commissionsQuery = useCommissions(slug, commFrom, commTo);
    98→  const profitDistributionQuery = useProfitDistribution(slug);
    99→
   100→  const vouchers = vouchersQuery.data?.vouchers ?? [];
   101→  const quotations = quotationsQuery.data?.quotations ?? [];
   102→  const purchaseOrders = purchaseOrdersQuery.data?.purchaseOrders ?? [];
   103→  const openingBalances = openingBalancesQuery.data?.openingBalances ?? [];
   104→  const commissions = commissionsQuery.data?.commissions ?? [];
   105→  const profitDistributions = profitDistributionQuery.data?.distributions ?? [];
   106→
   107→  const loading = (tab === "vouchers" && vouchersQuery.isLoading) || (tab === "quotations" && quotationsQuery.isLoading) || (tab === "purchase-orders" && purchaseOrdersQuery.isLoading) || (tab === "opening-balances" && openingBalancesQuery.isLoading) || (tab === "commissions" && commissionsQuery.isLoading) || (tab === "profit-distribution" && profitDistributionQuery.isLoading);
   108→
   109→  const switchTab = (t: Tab) => { setTab(t); setShowForm(false); };
   110→
   111→  /* ── Line Items helper ───────────────────────────────────────────────────── */
   112→  const updateLineItem = (items: LineItem[], idx: number, field: keyof LineItem, value: string | number): LineItem[] => {
   113→    const updated = [...items];
   114→    const item = { ...updated[idx] };
   115→    if (field === "description") item.description = String(value);
   116→    else if (field === "quantity") item.quantity = Number(value);
   117→    else if (field === "unitPrice") item.unitPrice = Number(value);
   118→    else if (field === "total") item.total = Number(value);
   119→    item.total = item.quantity * item.unitPrice;
   120→    updated[idx] = item;
   121→    return updated;
   122→  };
   123→
   124→  /* ── Mutation hooks ──────────────────────────────────────────────────────── */
   125→  const createVoucherMutation = useCreateVoucher();
   126→  const approveVoucherMutation = useApproveVoucher();
   127→  const cancelVoucherMutation = useCancelVoucher();
   128→  const createQuotationMutation = useCreateQuotation();
   129→  const convertQuotationMutation = useConvertQuotationToInvoice();
   130→  const createPurchaseOrderMutation = useCreatePurchaseOrder();
   131→  const createOpeningBalanceMutation = useCreateOpeningBalance();
   132→  const postOpeningBalancesMutation = usePostOpeningBalances();
   133→  const postCommissionMutation = usePostCommission();
   134→  const postProfitDistributionMutation = usePostProfitDistribution();
   135→
   136→  /* ── Create Voucher ─────────────────────────────────────────────────────── */
   137→  const handleCreateVoucher = () => {
   138→    if (!activeCompany || !vAmount || !vDate) { toast.error("يرجى ملء جميع الحقول المطلوبة"); return; }
   139→    createVoucherMutation.mutate(
   140→      { companySlug: activeCompany.slug, voucherType: vType, date: vDate, amount: parseFloat(vAmount), currency: vCurrency, payee: vPayee, payer: vPayer },
   141→      {
   142→        onSuccess: () => { toast.success("تم إنشاء السند"); setShowForm(false); resetVoucherForm(); vouchersQuery.refetch(); },
   143→        onError: (err) => { toast.error(err.message || "تعذّر إنشاء السند"); },
   144→      },
   145→    );
   146→  };
   147→  const resetVoucherForm = () => { setVType("receipt"); setVDate(""); setVAmount(""); setVCurrency("KWD"); setVPayee(""); setVPayer(""); };
   148→
   149→  /* ── Approve / Cancel Voucher ────────────────────────────────────────────── */
   150→  const handleApproveVoucher = (id: number) => {
   151→    if (!activeCompany) return;
   152→    approveVoucherMutation.mutate(
   153→      { id, companySlug: activeCompany.slug },
   154→      {
   155→        onSuccess: () => { toast.success("تم اعتماد السند"); vouchersQuery.refetch(); },
   156→        onError: (err) => { toast.error(err.message || "تعذّر اعتماد السند"); },
   157→      },
   158→    );
   159→  };
   160→  const handleCancelVoucher = (id: number) => {
   161→    if (!activeCompany) return;
   162→    cancelVoucherMutation.mutate(
   163→      { id, companySlug: activeCompany.slug },
   164→      {
   165→        onSuccess: () => { toast.success("تم إلغاء السند"); vouchersQuery.refetch(); },
   166→        onError: (err) => { toast.error(err.message || "تعذّر إلغاء السند"); },
   167→      },
   168→    );
   169→  };
   170→
   171→  /* ── Create Quotation ────────────────────────────────────────────────────── */
   172→  const handleCreateQuotation = () => {
   173→    if (!activeCompany || !qClient || !qDate) { toast.error("يرجى ملء جميع الحقول المطلوبة"); return; }
   174→    createQuotationMutation.mutate(
   175→      { companySlug: activeCompany.slug, clientName: qClient, date: qDate, validUntil: qValidUntil, lineItems: qLineItems, totalAmount: qLineItems.reduce((s, li) => s + li.total, 0) },
   176→      {
   177→        onSuccess: () => { toast.success("تم إنشاء عرض السعر"); setShowForm(false); quotationsQuery.refetch(); },
   178→        onError: (err) => { toast.error(err.message || "تعذّر إنشاء عرض السعر"); },
   179→      },
   180→    );
   181→  };
   182→
   183→  /* ── Convert Quotation to Invoice ─────────────────────────────────────────── */
   184→  const handleConvertToInvoice = (id: number) => {
   185→    if (!activeCompany) return;
   186→    convertQuotationMutation.mutate(
   187→      { id, companySlug: activeCompany.slug },
   188→      {
   189→        onSuccess: () => { toast.success("تم التحويل إلى فاتورة"); quotationsQuery.refetch(); },
   190→        onError: (err) => { toast.error(err.message || "تعذّر التحويل إلى فاتورة"); },
   191→      },
   192→    );
   193→  };
   194→
   195→  /* ── Create PO ────────────────────────────────────────────────────────────── */
   196→  const handleCreatePO = () => {
   197→    if (!activeCompany || !poSupplier || !poDate) { toast.error("يرجى ملء جميع الحقول المطلوبة"); return; }
   198→    createPurchaseOrderMutation.mutate(
   199→      { companySlug: activeCompany.slug, supplierName: poSupplier, date: poDate, expectedDelivery: poDelivery, lineItems: poLineItems, totalAmount: poLineItems.reduce((s, li) => s + li.total, 0) },
   200→      {
   201→        onSuccess: () => { toast.success("تم إنشاء أمر الشراء"); setShowForm(false); purchaseOrdersQuery.refetch(); },
   202→        onError: (err) => { toast.error(err.message || "تعذّر إنشاء أمر الشراء"); },
   203→      },
   204→    );
   205→  };
   206→
   207→  /* ── Create Opening Balance ──────────────────────────────────────────────── */
   208→  const handleCreateOB = () => {
   209→    if (!activeCompany || !obAccountId || !obAmount) { toast.error("يرجى ملء جميع الحقول المطلوبة"); return; }
   210→    createOpeningBalanceMutation.mutate(
   211→      { companySlug: activeCompany.slug, accountId: parseInt(obAccountId), amount: parseFloat(obAmount) },
   212→      {
   213→        onSuccess: () => { toast.success("تم إنشاء الرصيد الافتتاحي"); setShowForm(false); setObAccountId(""); setObAmount(""); openingBalancesQuery.refetch(); },
   214→        onError: (err) => { toast.error(err.message || "تعذّر إنشاء الرصيد الافتتاحي"); },
   215→      },
   216→    );
   217→  };
   218→
   219→  /* ── Post all opening balances ────────────────────────────────────────────── */
   220→  const handlePostAllOB = () => {
   221→    if (!activeCompany) return;
   222→    if (!confirm("ترحيل جميع الأرصدة الافتتاحية؟ لا يمكن التراجع عن هذا الإجراء.")) return;
   223→    postOpeningBalancesMutation.mutate(
   224→      { companySlug: activeCompany.slug },
   225→      {
   226→        onSuccess: () => { toast.success("تم ترحيل جميع الأرصدة الافتتاحية"); openingBalancesQuery.refetch(); },
   227→        onError: (err) => { toast.error(err.message || "تعذّر ترحيل الأرصدة الافتتاحية"); },
   228→      },
   229→    );
   230→  };
   231→
   232→  /* ── Post Commission as JE ────────────────────────────────────────────────── */
   233→  const handlePostCommission = (id: number) => {
   234→    if (!activeCompany) return;
   235→    postCommissionMutation.mutate(
   236→      { id, companySlug: activeCompany.slug },
   237→      {
   238→        onSuccess: () => { toast.success("تم ترحيل العمولة كقيد يومية"); commissionsQuery.refetch(); },
   239→        onError: (err) => { toast.error(err.message || "تعذّر ترحيل العمولة"); },
   240→      },
   241→    );
   242→  };
   243→
   244→  /* ── Post Profit Distribution as JE ───────────────────────────────────────── */
   245→  const handlePostProfitDist = (id: number) => {
   246→    if (!activeCompany) return;
   247→    postProfitDistributionMutation.mutate(
   248→      { id, companySlug: activeCompany.slug },
   249→      {
   250→        onSuccess: () => { toast.success("تم ترحيل التوزيع كقيد يومية"); profitDistributionQuery.refetch(); },
   251→        onError: (err) => { toast.error(err.message || "تعذّر ترحيل توزيع الأرباح"); },
   252→      },
   253→    );
   254→  };
   255→
   256→  if (!activeCompany) return <div className="p-12 text-center text-muted-foreground">اختر شركة</div>;
   257→
   258→  const tabs: Array<{ key: Tab; label: string; icon: React.ComponentType<{ size?: number }> }> = [
   259→    { key: "vouchers", label: "سندات", icon: Receipt },
   260→    { key: "quotations", label: "عروض أسعار", icon: FileText },
   261→    { key: "purchase-orders", label: "أوامر الشراء", icon: ShoppingCart },
   262→    { key: "opening-balances", label: "أرصدة افتتاحية", icon: BookOpen },
   263→    { key: "commissions", label: "عمولات", icon: Percent },
   264→    { key: "profit-distribution", label: "توزيع أرباح", icon: HandCoins },
   265→  ];
   266→
   267→  const fmt = (n: number) => n.toLocaleString("ar-EG", { maximumFractionDigits: 3 });
   268→
   269→  return (
   270→    <div className="flex flex-col gap-4">
   271→      {/* Header - DS v4.0: Emerald accent color */}
   272→      <div className="flex flex-wrap justify-between items-center gap-3">
   273→        <div>
   274→          <h1 className="text-2xl font-extrabold flex items-center gap-2 text-[#047857]"><Receipt size={20} /> السندات والتفاصيل</h1>
   275→          <p className="text-[13px] text-muted-foreground">{activeCompany.nameAr || activeCompany.name}</p>
   276→        </div>
   277→        {!showForm && (
   278→          // DS v4.0: Action button with active-press
   279→          <button onClick={() => setShowForm(true)} className="inline-flex items-center gap-1.5 py-2.5 px-[18px] rounded-[10px] bg-primary text-primary-foreground border-none text-[13px] font-bold cursor-pointer active-press duration-150"><Plus size={16} /> إضافة</button>
   280→        )}
   281→      </div>
   282→
   283→      {/* Tabs - DS v4.0: Filter buttons with active-press */}
   284→      <div className="flex gap-1.5 flex-wrap">
   285→        {tabs.map((t) => { const Icon = t.icon; return (
   286→          <button key={t.key} onClick={() => switchTab(t.key)} className={cn(
   287→            "py-2 px-4 rounded-[10px] border border-border text-[12px] font-bold cursor-pointer inline-flex items-center gap-1.5 active-press duration-150",
   288→            tab === t.key ? "bg-primary text-primary-foreground" : "bg-card text-muted-foreground",
   289→          )}>{Icon && <Icon size={14} />} {t.label}</button>
   290→        ); })}
   291→      </div>
   292→
   293→      {/* Commissions date filter - DS v4.0: Panel with hover-lift */}
   294→      {tab === "commissions" && !showForm && (
   295→        <div className="flex gap-3 items-end bg-card rounded-[10px] border border-border p-3 hover-lift duration-120">
   296→          <div><label className={labelStyle}>من</label><input value={commFrom} onChange={(e) => setCommFrom(e.target.value)} className={cn(inputStyle, "w-36")} type="date" /></div>
   297→          <div><label className={labelStyle}>إلى</label><input value={commTo} onChange={(e) => setCommTo(e.target.value)} className={cn(inputStyle, "w-36")} type="date" /></div>
   298→        </div>
   299→      )}
   300→
   301→      {loading ? <div className="p-12 text-center text-muted-foreground">جارٍ التحميل…</div> : (() => {
   302→        /* ── Vouchers Tab ─────────────────────────────────────────────────────── */
   303→        if (tab === "vouchers") {
   304→          if (showForm) return (
   305→            // DS v4.0: Form card with hover-lift
   306→            <div className="bg-card rounded-[14px] border border-border p-5 hover-lift duration-120">
   307→              <div className="flex items-center justify-between mb-4">
   308→                <h2 className="text-base font-bold">سند جديد</h2>
   309→                <button onClick={() => { setShowForm(false); resetVoucherForm(); }} className="w-7 h-7 rounded-md border border-border flex items-center justify-center cursor-pointer"><X size={14} /></button>
   310→              </div>
   311→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   312→                <div><label className={labelStyle}>نوع السند *</label>
   313→                  <select value={vType} onChange={(e) => setVType(e.target.value)} className={selectStyle}>
   314→                    <option value="receipt">سند قبض</option><option value="payment">سند دفع</option>
   315→                  </select>
   316→                </div>
   317→                <div><label className={labelStyle}>التاريخ *</label><input value={vDate} onChange={(e) => setVDate(e.target.value)} className={inputStyle} type="date" /></div>
   318→                <div><label className={labelStyle}>المبلغ *</label><input value={vAmount} onChange={(e) => setVAmount(e.target.value)} className={inputStyle} type="number" placeholder="0.000" /></div>
   319→                <div><label className={labelStyle}>العملة</label>
   320→                  <select value={vCurrency} onChange={(e) => setVCurrency(e.target.value)} className={selectStyle}>
   321→                    <option value="KWD">KWD</option><option value="USD">USD</option><option value="EUR">EUR</option><option value="SAR">SAR</option>
   322→                  </select>
   323→                </div>
   324→                <div><label className={labelStyle}>المستفيد</label><input value={vPayee} onChange={(e) => setVPayee(e.target.value)} className={inputStyle} placeholder="اسم المستفيد" /></div>
   325→                <div><label className={labelStyle}>الدافع</label><input value={vPayer} onChange={(e) => setVPayer(e.target.value)} className={inputStyle} placeholder="اسم الدافع" /></div>
   326→              </div>
   327→              {/* DS v4.0: Action buttons with active-press */}
   328→              <div className="flex gap-2 justify-end mt-5">
   329→                <button onClick={() => { setShowForm(false); resetVoucherForm(); }} className="px-4 py-2 rounded-md border border-border bg-transparent text-foreground text-sm font-semibold cursor-pointer active-press duration-150">إلغاء</button>
   330→                <button onClick={handleCreateVoucher} className="px-4 py-2 rounded-md bg-primary text-primary-foreground text-sm font-bold cursor-pointer inline-flex items-center gap-1.5 active-press duration-150"><Receipt size={14} /> إنشاء</button>
   331→              </div>
   332→            </div>
   333→          );
   334→          if (vouchers.length === 0) return <Empty label="سندات" />;
   335→          // DS v4.0: Vouchers table with enterprise styling and kpi-card-gold totals
   336→          return (
   337→            <div className="bg-card rounded-[14px] border border-border overflow-hidden hover-lift duration-120">
   338→              {/* DS v4.0: KPI Summary Cards for Vouchers */}
   339→              {vouchers.length > 0 && (
   340→                <div className="grid grid-cols-1 md:grid-cols-4 gap-3 p-4 border-b border-border">
   341→                  {/* Total Vouchers - kpi-card */}
   342→                  <div className="kpi-card">
   343→                    <div className="flex items-center gap-2">
   344→                      <FileText size={16} className="text-[#047857]" />
   345→                      <div>
   346→                        <p className="text-[10px] text-muted-foreground">إجمالي السندات</p>
   347→                        <p className="text-sm font-bold text-[#047857]">{vouchers.length}</p>
   348→                      </div>
   349→                    </div>
   350→                  </div>
   351→                  {/* Total Debit (Receipts) - kpi-card-gold (Premium financial data!) */}
   352→                  <div className="kpi-card-gold">
   353→                    <div className="flex items-center gap-2">
   354→                      <ArrowRight size={16} className="text-[#d4a574]" />
   355→                      <div>
   356→                        <p className="text-[10px] opacity-80">إجمالي القبض</p>
   357→                        <p className="text-sm font-bold">{fmt(vouchers.filter(v => v.voucherType === 'receipt').reduce((sum, v) => sum + v.amount, 0))}</p>
   358→                      </div>
   359→                    </div>
   360→                  </div>
   361→                  {/* Total Credit (Payments) - kpi-card */}
   362→                  <div className="kpi-card">
   363→                    <div className="flex items-center gap-2">
   364→                      <CreditCard size={16} className="text-red-500" />
   365→                      <div>
   366→                        <p className="text-[10px] text-muted-foreground">إجمالي الدفع</p>
   367→                        <p className="text-sm font-bold text-red-500">{fmt(vouchers.filter(v => v.voucherType === 'payment').reduce((sum, v) => sum + v.amount, 0))}</p>
   368→                      </div>
   369→                    </div>
   370→                  </div>
   371→                  {/* Net Balance - kpi-card-gold */}
   372→                  <div className="kpi-card-gold">
   373→                    <div className="flex items-center gap-2">
