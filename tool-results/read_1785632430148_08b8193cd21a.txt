   266→  computeLayoutHash(formatType) {
   267→    // Simplified layout hash computation
   268→    // In real system, this would analyze actual document structure
   269→    let hash = 0;
   270→    for (let i = 0; i < formatType.length; i++) {
   271→      const char = formatType.charCodeAt(i);
   272→      hash = ((hash << 5) - hash) + char;
   273→      hash = hash & hash; // Convert to 32bit integer
   274→    }
   275→    return `layout_${Math.abs(hash).toString(16).padStart(8, '0')}`;
   276→  }

   306→  applyOcrErrors() {
   307→    const errorTypes = [
   308→      { type: 'char_substitution', field: 'supplierName', pattern: /[أإآ]/g, replacement: 'ا' },
   309→      { type: 'char_omission', field: 'trn', pattern: /\d/g, probability: 0.1 },
   310→      { type: 'space_insertion', field: 'amount', pattern: /\d/g, insertSpace: true },
   311→      { type: 'wrong_language', field: 'anchorWords', swapToEnglish: true },
   312→      { type: 'position_drift', field: 'fieldPositions', driftAmount: 5 }
   313→    ];
   314→    
   315→    let errorCount = 0;
   316→    
   317→    for (const invoice of this.invoices) {
   318→      if (Math.random() > this.ocrErrorRate) continue;
   319→      
   320→      invoice.hasOcrError = true;
   321→      const errorType = errorTypes[Math.floor(Math.random() * errorTypes.length)];
   322→      
   323→      switch (errorType.type) {
   324→        case 'char_substitution':
   325→          if (invoice.expectedSupplierName.includes('أ') || invoice.expectedSupplierName.includes('إ')) {
   326→            invoice.ocrErrors.push({
   327→              type: 'char_substitution',
   328→              field: 'supplierName',
   329→              original: invoice.expectedSupplierName,
   330→              corrupted: invoice.expectedSupplierName.replace(/[أإآ]/g, 'ا')
   331→            });
   332→            errorCount++;
   333→          }
   334→          break;
   335→          
   336→        case 'char_omission':
   337→          const originalTrn = invoice.expectedTrn;
   338→          let corruptedTrn = '';
   339→          for (const char of originalTrn) {
   340→            if (/\d/.test(char) && Math.random() < 0.1) {
   341→              continue; // Skip this digit
   342→            }
   343→            corruptedTrn += char;
   344→          }
   345→          if (corruptedTrn !== originalTrn) {
   346→            invoice.ocrErrors.push({
   347→              type: 'char_omission',
   348→              field: 'trn',
   349→              original: originalTrn,
   350→              corrupted: corruptedTrn
   351→            });
   352→            errorCount++;
   353→          }
   354→          break;
   355→          
   356→        case 'position_drift':
   357→          invoice.ocrErrors.push({
   358→            type: 'position_drift',
   359→            field: 'fieldPositions',
   360→            driftAmount: errorType.driftAmount
   361→          });
   362→          // Apply drift to positions
   363→          if (invoice.fieldPositions) {
   364→            for (const field of Object.keys(invoice.fieldPositions)) {
   365→              invoice.fieldPositions[field].x += (Math.random() - 0.5) * errorType.driftAmount;
   366→              invoice.fieldPositions[field].y += (Math.random() - 0.5) * errorType.driftAmount;
   367→            }
   368→          }
   369→          errorCount++;
   370→          break;
   371→      }
   372→    }
   373→    
   374→    console.log(`   🔤 Applied OCR errors: ${errorCount} invoices affected (${(errorCount / this.invoices.length * 100).toFixed(1)}%)`);
   375→  }
   376→  
   377→  getStats() {
   378→    const formatDistribution = {};
   379→    for (const inv of this.invoices) {
   380→      formatDistribution[inv.formatType] = (formatDistribution[inv.formatType] || 0) + 1;
   381→    }
   382→    
   383→    const ocrErrorCount = this.invoices.filter(i => i.hasOcrError).length;
   384→    const sharedLayoutCount = this.invoices.filter(i => i.hasSharedLayout).length;
   385→    
   386→    return {
   387→      totalSuppliers: this.suppliers.length,
   388→      totalInvoices: this.invoices.length,
   389→      ocrErrorRate: (ocrErrorCount / this.invoices.length * 100).toFixed(1) + '%',
   390→      sharedLayoutRate: (sharedLayoutCount / this.invoices.length * 100).toFixed(1) + '%',
   391→      formatDistribution,
   392→      tenants: [...new Set(this.suppliers.map(s => s.tenantId))].length
   393→    };
   394→  }

   401→class SemanticFingerprintEngine {
   402→  constructor() {
   403→    this.patternStore = new EnhancedPatternStore();
   404→    this.decisionTraces = []; // Store all decision traces
   405→    this.stats = {
   406→      totalProcessed: 0,
   407→      patternMatched: 0,
   408→      aiFallback: 0,
   409→      falseMatches: 0,
   410→      correctRejects: 0,
   411→      stagePass: { layout: 0, semantic: 0, supplier: 0, crossField: 0 },
   412→      stageFail: { layout: 0, semantic: 0, supplier: 0, crossField: 0 }
   413→    };
   414→  }
   415→
   416→  /**
   417→   * PHASE 1: Supplier Ownership Gate
   418→   * Patterns are now scoped to tenantId + supplierId
   419→   */
   420→  processInvoice(invoice, knownTenantId = null, knownSupplierCandidates = []) {
   421→    const startTime = performance.now();
   422→    const trace = new DecisionTrace(invoice.id);
   423→    
   424→    try {
   425→      trace.addStep('start', 'Received invoice for processing', {
   426→        invoiceId: invoice.id,
   427→        formatType: invoice.formatType,
   428→        hasOcrError: invoice.hasOcrError
   429→      });
   430→
   431→      // ===== STAGE 1: Layout Fingerprint (Original method) =====
   432→      const layoutResult = this.stage1_LayoutValidation(invoice, trace);
   433→      if (!layoutResult.pass) {
   434→        this.recordAiFallback(invoice, trace, 'layout_validation_failed');
   435→        return this.buildResult(invoice, 'AI_FALLBACK', trace, startTime, layoutResult.confidence);
   436→      }
   437→      
   438→      trace.addStep('stage1_pass', '✓ Layout validation passed', { confidence: layoutResult.confidence });
   439→
   440→      // ===== PHASE 1: Supplier Ownership Gate =====
   441→      // If we know the tenant/supplier upfront, filter candidates immediately
   442→      let candidatePatterns = layoutResult.candidates;
   443→      
   444→      if (knownTenantId) {
   445→        trace.addStep('ownership_gate', 'Applying Supplier Ownership Gate', { tenantId: knownTenantId });
   446→        
   447→        candidatePatterns = candidatePatterns.filter(p => p.tenantId === knownTenantId);
   448→        trace.addStep('ownership_filter', `Filtered to ${candidatePatterns.length} patterns for tenant`, {
   449→          before: layoutResult.candidates.length,
   450→          after: candidatePatterns.length
   451→        });
   452→        
   453→        if (knownSupplierCandidates.length > 0) {
   454→          candidatePatterns = candidatePatterns.filter(p => 
   455→            knownSupplierCandidates.includes(p.supplierId)
   456→          );
   457→          trace.addStep('supplier_filter', `Filtered to ${candidatePatterns.length} patterns for supplier candidates`, {
   458→            candidates: knownSupplierCandidates
   459→          });
   460→        }
   461→      }
   462→      
   463→      if (candidatePatterns.length === 0) {
   464→        this.stats.correctRejects++;
   465→        trace.addStep('ownership_reject', '✗ No matching pattern for this tenant/supplier');
   466→        this.recordAiFallback(invoice, trace, 'no_pattern_for_owner');
   467→        return this.buildResult(invoice, 'AI_FALLBACK', trace, startTime, 0.1);
   468→      }
   469→
   470→      // ===== STAGE 2: Semantic Fingerprint Validation =====
   471→      const semanticResult = this.stage2_SemanticValidation(invoice, candidatePatterns, trace);
   472→      if (!semanticResult.pass) {
   473→        this.recordAiFallback(invoice, trace, 'semantic_validation_failed');
   474→        return this.buildResult(invoice, 'AI_FALLBACK', trace, startTime, semanticResult.confidence);
   475→      }
   476→      
   477→      trace.addStep('stage2_pass', '✓ Semantic validation passed', { 
   478→        confidence: semanticResult.confidence,
   479→        matchedPattern: semanticResult.bestMatch?.supplierId 
   480→      });
   481→
   482→      // ===== STAGE 3: Supplier Identity Validation =====
   483→      const supplierResult = this.stage3_SupplierIdentityValidation(invoice, semanticResult.bestMatch, trace);
   484→      if (!supplierResult.pass) {
   485→        // THIS IS THE KEY FIX: Reject even high-confidence matches if supplier identity doesn't match
   486→        this.stats.correctRejects++; // This was a potential false match, correctly rejected!
   487→        trace.addStep('stage3_fail', '✗ Supplier identity validation FAILED - Potential False Match prevented!', {
   488→          reason: supplierResult.reason,
   489→          expectedSupplier: invoice.expectedSupplierName,
   490→          patternSupplier: semanticResult.bestMatch?.supplierName
   491→        });
   492→        
   493→        this.recordAiFallback(invoice, trace, 'supplier_identity_mismatch');
   494→        return this.buildResult(invoice, 'AI_FALLBACK', trace, startTime, supplierResult.confidence);
   495→      }
   496→      
   497→      trace.addStep('stage3_pass', '✓ Supplier identity validated', { supplierName: supplierResult.supplierName });
   498→
   499→      // ===== STAGE 4: Cross-Field Validation =====
   500→      const crossFieldResult = this.stage4_CrossFieldValidation(invoice, semanticResult.bestMatch, trace);
   501→      if (!crossFieldResult.pass) {
   502→        this.recordAiFallback(invoice, trace, 'cross_field_validation_failed');
   503→        return this.buildResult(invoice, 'AI_FALLBACK', trace, startTime, crossFieldResult.confidence);
   504→      }
   505→      
   506→      trace.addStep('stage4_pass', '✓ Cross-field validation passed');
   507→
   508→      // ===== FINAL ACCEPTANCE =====
   509→      const finalConfidence = this.calculateFinalConfidence([
   510→        layoutResult.confidence,
   511→        semanticResult.confidence,
   512→        supplierResult.confidence,
   513→        crossFieldResult.confidence
   514→      ]);
   515→      
   516→      trace.addStep('accepted', '🎉 INVOICE ACCEPTED - All validations passed', {
   517→        finalConfidence: finalConfidence,
   518→        matchedSupplier: semanticResult.bestMatch?.supplierName,
   519→        patternId: semanticResult.bestMatch?.id
   520→      });
   521→      
   522→      this.stats.patternMatched++;
   523→      this.stats.stagePass.layout++;
   524→      this.stats.stagePass.semantic++;
   525→      this.stats.stagePass.supplier++;
   526→      this.stats.stagePass.crossField++;
   527→      
   528→      // Check if this is actually a correct match or false match
   529→      const isCorrectMatch = semanticResult.bestMatch?.supplierId === invoice.supplierId;
   530→      if (!isCorrectMatch) {
   531→        this.stats.falseMatches++;
   532→        trace.addStep('FALSE_MATCH_DETECTED', '⚠️ FALSE MATCH - Wrong supplier matched!', {
   533→          expected: invoice.expectedSupplierName,
   534→          actual: semanticResult.bestMatch?.supplierName
   535→        });
   536→      }
   537→      
   538→      return this.buildResult(
   539→        invoice,
   540→        isCorrectMatch ? 'PATTERN_MATCH' : 'FALSE_MATCH',
   541→        trace, 
   542→        startTime, 
   543→        finalConfidence,
   544→        semanticResult.bestMatch
   545→      );
   546→
   547→    } catch (error) {
   548→      trace.addStep('error', `Processing error: ${error.message}`);
   549→      this.recordAiFallback(invoice, trace, 'processing_error');
   550→      return this.buildResult(invoice, 'AI_FALLBACK', trace, startTime, 0);
   551→    }
   552→  }

   554→  /**
   555→   * Stage 1: Layout-based Fingerprint Matching (Original Method)
   556→   */
   557→  stage1_LayoutValidation(invoice, trace) {
   558→    const layoutFingerprint = invoice.layoutHash;
   559→    const candidates = this.patternStore.findByLayout(layoutFingerprint);
   560→    
   561→    if (candidates.length === 0) {
   562→      this.stats.stageFail.layout++;
   563→      return { pass: false, confidence: 0, candidates: [] };
   564→    }
   565→    
   566→    // Calculate layout similarity score
   567→    const bestCandidate = candidates.reduce((best, current) => {
   568→      const similarity = this.calculateLayoutSimilarity(invoice, current);
   569→      return similarity > best.score ? { pattern: current, score: similarity } : best;
   570→    }, { pattern: null, score: 0 });
   571→    
   572→    this.stats.stagePass.layout++;
   573→    return {
   574→      pass: bestCandidate.score > 0.6,
   575→      confidence: bestCandidate.score,
   576→      candidates: candidates
   577→    };
   578→  }
   579→
   580→  /**
   581→   * Stage 2: Semantic Fingerprint Validation (PHASE 2)
   582→   * Uses multiple dimensions beyond just layout
   583→   */
   584→  stage2_SemanticValidation(invoice, candidates, trace) {
   585→    const scores = [];
   586→    
   587→    for (const pattern of candidates) {
   588→      const semanticScore = this.calculateSemanticScore(invoice, pattern, trace);
   589→      scores.push({ pattern, score: semanticScore });
   590→    }
   591→    
   592→    // Sort by score descending
   593→    scores.sort((a, b) => b.score - a.score);
   594→    const bestMatch = scores[0];
   595→    
   596→    if (!bestMatch || bestMatch.score < 0.5) {
   597→      this.stats.stageFail.semantic++;
   598→      return { pass: false, confidence: bestMatch?.score || 0, bestMatch: null };
   599→    }
   600→    
   601→    this.stats.stagePass.semantic++;
   602→    return { pass: true, confidence: bestMatch.score, bestMatch: bestMatch.pattern };
   603→  }
   604→
   605→  /**
   606→   * Calculate Semantic Score (Multi-dimensional fingerprint)
   607→   */
   608→  calculateSemanticScore(invoice, pattern, trace) {
   609→    let totalScore = 0;
   610→    const weights = {
   611→      fieldTopology: 0.25,   // Field positions matter A LOT
   612→      headerSignature: 0.20, // Header text matching
   613→      currency: 0.15,        // Currency must match
   614→      taxStructure: 0.15,    // VAT/tax structure
   615→      anchorWords: 0.25      // Key identifying words
   616→    };
   617→    
   618→    // 1. Field Topology Comparison
   619→    const topologyScore = this.compareFieldTopology(invoice.fieldPositions, pattern.fieldPositions);
   620→    trace.addDetail('field_topology', `Topology score: ${topologyScore.toFixed(3)}`);
   621→    totalScore += weights.fieldTopology * topologyScore;
   622→    
   623→    // 2. Header Signature Comparison
   624→    const headerScore = this.compareHeaderSignatures(invoice.headerSignature, pattern.headerSignature);
   625→    trace.addDetail('header_signature', `Header score: ${headerScore.toFixed(3)}`);
   626→    totalScore += weights.headerSignature * headerScore;
   627→    
   628→    // 3. Currency Check (strict match required)
   629→    const currencyScore = invoice.currency === pattern.currency ? 1.0 : 0.0;
   630→    trace.addDetail('currency_check', `Currency: ${invoice.currency} vs ${pattern.currency} = ${currencyScore}`);
   631→    totalScore += weights.currency * currencyScore;
   632→    
   633→    // 4. Tax Structure Comparison
   634→    const taxScore = this.compareTaxStructures(invoice.taxStructure, pattern.taxStructure);
   635→    trace.addDetail('tax_structure', `Tax structure score: ${taxScore.toFixed(3)}`);
   636→    totalScore += weights.taxStructure * taxScore;
   637→    
   638→    // 5. Anchor Words Overlap
   639→    const anchorScore = this.compareAnchorWords(invoice.anchorWords, pattern.anchorWords);
   640→    trace.addDetail('anchor_words', `Anchor words score: ${anchorScore.toFixed(3)}`);
   641→    totalScore += weights.anchorWords * anchorScore;
   642→    
   643→    return Math.min(1.0, totalScore);
   644→  }
   645→
   646→  /**
   647→   * Stage 3: Supplier Identity Validation (PHASE 1 Enhancement)
   648→   * CRITICAL: Prevents cross-supplier false matches
   649→   */
   650→  stage3_SupplierIdentityValidation(invoice, pattern, trace) {
   651→    if (!pattern) {
   652→      return { pass: false, confidence: 0, reason: 'no_pattern' };
   653→    }
   654→    
   655→    let confidence = 1.0;
   656→    const checks = [];
   657→    
   658→    // Check 1: If invoice has expected supplier info, verify it matches
   659→    if (invoice.expectedSupplierName) {
   660→      const nameSimilarity = this.calculateStringSimilarity(
   661→        invoice.expectedSupplierName.toLowerCase(),
   662→        pattern.supplierName?.toLowerCase() || ''
   663→      );
   664→      checks.push({ check: 'supplier_name_match', score: nameSimilarity });
   665→      
   666→      // Strict threshold for supplier name
   667→      if (nameSimilarity < 0.7) {
   668→        this.stats.stageFail.supplier++;
   669→        return {
   670→          pass: false,
   671→          confidence: nameSimilarity,
   672→          reason: `supplier_name_mismatch (${nameSimilarity.toFixed(2)} < 0.7)`,
   673→          supplierName: pattern.supplierName
   674→        };
   675→      }
   676→      confidence *= (0.5 + 0.5 * nameSimilarity);
   677→    }
   678→    
   679→    // Check 2: TRN verification
   680→    if (invoice.expectedTrn && pattern.trn) {
   681→      const trnMatch = invoice.expectedTrn === pattern.trn ? 1.0 : 0.0;
   682→      checks.push({ check: 'trn_match', score: trnMatch });
   683→      
   684→      if (trnMatch < 0.9) {
   685→        this.stats.stageFail.supplier++;
   686→        return {
   687→          pass: false,
   688→          confidence: trnMatch,
   689→          reason: 'trn_mismatch',
   690→          supplierName: pattern.supplierName
   691→        };
   692→      }
   693→      confidence *= (0.7 + 0.3 * trnMatch);
   694→    }
   695→    
   696→    // Check 3: Tenant isolation (from Phase 1)
   697→    if (invoice.tenantId && pattern.tenantId) {
   698→      const tenantMatch = invoice.tenantId === pattern.tenantId ? 1.0 : 0.0;
   699→      checks.push({ check: 'tenant_match', score: tenantMatch });
   700→      
   701→      if (tenantMatch === 0) {
   702→        this.stats.stageFail.supplier++;
   703→        return {
   704→          pass: false,
   705→          confidence: 0,
   706→          reason: 'tenant_mismatch',
   707→          supplierName: pattern.supplierName
   708→        };
   709→      }
   710→    }
   711→    
   712→    trace.addDetail('supplier_identity', 'Supplier identity checks passed', checks);
   713→    this.stats.stagePass.supplier++;
   714→    return { pass: true, confidence, supplierName: pattern.supplierName };
   715→  }
   716→
   717→  /**
   718→   * Stage 4: Cross-Field Validation
   719→   * Ensures extracted data is internally consistent
   720→   */
   721→  stage4_CrossFieldValidation(invoice, pattern, trace) {
   722→    const validations = [];
   723→    
   724→    // Validation 1: Amount range check
   725→    if (pattern.typicalAmountRange) {
   726→      const [min, max] = pattern.typicalAmountRange;
   727→      const inRange = invoice.amount >= min && invoice.amount <= max;
   728→      validations.push({ check: 'amount_range', passed: inRange, value: invoice.amount, range: [min, max] });
   729→      
   730→      if (!inRange) {
   731→        this.stats.stageFail.crossField++;
   732→        return { pass: false, confidence: 0.3, reason: 'amount_out_of_range' };
   733→      }
   734→    }
   735→    
   736→    // Validation 2: Date reasonableness
   737→    const invoiceDate = new Date(invoice.date);
   738→    const now = new Date();
   739→    const daysDiff = (now - invoiceDate) / (1000 * 60 * 60 * 24);
   740→    const reasonableDate = daysDiff >= 0 && daysDiff <= 365 * 2; // Within 2 years
   741→    validations.push({ check: 'date_reasonable', passed: reasonableDate, daysOld: daysDiff });
   742→    
   743→    // Validation 3: Format consistency
   744→    const formatConsistent = invoice.formatType === pattern.formatType;
   745→    validations.push({ check: 'format_consistent', passed: formatConsistent });
   746→    
   747→    trace.addDetail('cross_field', 'Cross-field validations completed', validations);
   748→    
   749→    const overallScore = validations.filter(v => v.passed).length / validations.length;
   750→    
   751→    if (overallScore < 0.75) {
   752→      this.stats.stageFail.crossField++;
   753→      return { pass: false, confidence: overallScore, reason: 'cross_field_inconsistency' };
   754→    }
   755→    
   756→    this.stats.stagePass.crossField++;
   757→    return { pass: true, confidence: overallScore };
   758→  }

   760→  // ========== HELPER METHODS ==========
   761→  
   762→  calculateLayoutSimilarity(invoice, pattern) {
   763→    if (invoice.layoutHash === pattern.layoutHash) return 0.95;
   764→    // Simple hash comparison - in reality would be more sophisticated
   765→    return 0.6 + Math.random() * 0.35;
   766→  }
   767→  
   768→  compareFieldTopology(pos1, pos2) {
   769→    if (!pos1 || !pos2) return 0.5;
   770→    
   771→    let totalDistance = 0;
   772→    let fieldCount = 0;
   773→    
   774→    for (const field of Object.keys(pos1)) {
   775→      if (pos2[field]) {
   776→        const dx = pos1[field].x - pos2[field].x;
   777→        const dy = pos1[field].y - pos2[field].y;
   778→        totalDistance += Math.sqrt(dx * dx + dy * dy);
   779→        fieldCount++;
   780→      }
   781→    }
   782→    
   783→    if (fieldCount === 0) return 0.5;
   784→    
   785→    const avgDistance = totalDistance / fieldCount;
   786→    // Convert distance to similarity (lower distance = higher similarity)
   787→    return Math.max(0, Math.min(1, 1 - avgDistance / 20));
   788→  }
   789→  
   790→  compareHeaderSignatures(sig1, sig2) {
   791→    if (!sig1 || !sig2) return 0.5;
   792→    
   793→    let matches = 0;
   794→    for (const line of sig1) {
   795→      if (sig2.some(s => s.includes(line) || line.includes(s))) {
   796→        matches++;
   797→      }
   798→    }
   799→    
   800→    return matches / Math.max(sig1.length, sig2.length);
   801→  }
   802→  
   803→  compareTaxStructures(tax1, tax2) {
   804→    if (!tax1 || !tax2) return 0.5;
   805→    
   806→    let score = 0;
   807→    if (tax1.hasVat === tax2.hasVat) score += 0.4;
   808→    if (tax1.vatRate === tax2.vatRate) score += 0.3;
   809→    if (tax1.showSeparateLine === tax2.showSeparateLine) score += 0.3;
   810→    
   811→    return score;
   812→  }
   813→  
   814→  compareAnchorWords(anchors1, anchors2) {
   815→    if (!anchors1 || !anchors2) return 0.5;
   816→    
   817→    const set1 = new Set(anchors1.filter(a => a));
   818→    const set2 = new Set(anchors2.filter(a => a));
   819→    
   820→    let intersection = 0;
   821→    for (const word of set1) {
   822→      if (set2.has(word)) intersection++;
   823→    }
   824→    
   825→    const union = new Set([...set1, ...set2]).size;
   826→    return union > 0 ? intersection / union : 0.5;
   827→  }
   828→  
   829→  calculateStringSimilarity(str1, str2) {
   830→    if (str1 === str2) return 1.0;
   831→    if (!str1 || !str2) return 0.0;
   832→    
   833→    // Simple Levenshtein-like similarity
   834→    const longer = str1.length > str2.length ? str1 : str2;
   835→    const shorter = str1.length > str2.length ? str2 : str1;
   836→    
   837→    if (longer.length === 0) return 1.0;
   838→    
   839→    const costs = [];
   840→    for (let i = 0; i <= longer.length; i++) {
   841→      let lastValue = i;
   842→      for (let j = 0; j <= shorter.length; j++) {
   843→        if (i === 0) {
   844→          costs[j] = j;
   845→        } else if (j > 0) {
   846→          let newValue = costs[j - 1];
   847→          if (longer.charAt(i - 1) !== shorter.charAt(j - 1)) {
   848→            newValue = Math.min(Math.min(newValue, lastValue), costs[j]) + 1;
   849→          }
   850→          costs[j - 1] = lastValue;
   851→          lastValue = newValue;
   852→        }
   853→      }
   854→      if (i > 0) costs[shorter.length] = lastValue;
   855→    }
   856→    
   857→    return (longer.length - costs[shorter.length]) / longer.length;
   858→  }
   859→  
   860→  calculateFinalConfidence(scores) {
   861→    if (scores.length === 0) return 0;
   862→    const product = scores.reduce((a, b) => a * b, 1);
   863→    return Math.pow(product, 1 / scores.length); // Geometric mean
   864→  }
   865→  
   866→  recordAiFallback(invoice, trace, reason) {
   867→    this.stats.aiFallback++;
   868→    trace.addStep('ai_fallback', `→ AI Fallback triggered: ${reason}`, {
   869→      invoiceId: invoice.id,
   870→      reason: reason
   871→    });
   872→  }
   873→  
   874→  buildResult(invoice, outcome, trace, startTime, confidence, pattern = null) {
   875→    const processingTime = performance.now() - startTime;
   876→    this.stats.totalProcessed++;
   877→    
   878→    invoice.processedAt = new Date().toISOString();
   879→    invoice.processingTimeMs = processingTime;
   880→    invoice.decisionTrace = trace;
   881→    invoice.matchOutcome = outcome;
   882→    invoice.matchConfidence = confidence;
   883→    invoice.matchedPattern = pattern;
   884→    
   885→    // Store trace for analysis
   886→    this.decisionTraces.push({
   887→      invoiceId: invoice.id,
   888→      outcome: outcome,
   889→      confidence: confidence,
   890→      processingTimeMs: processingTime,
   891→      trace: trace.getTrace(),
   892→      timestamp: new Date().toISOString()
   893→    });
   894→    
   895→    return {
   896→      outcome,
   897→      confidence,
   898→      processingTimeMs: processingTime,
   899→      trace: trace.getTrace(),
   900→      pattern: pattern
   901→    };
   902→  }
   903→  
   904→  getStats() {
   905→    const total = this.stats.totalProcessed || 1;
   906→    return {
   907→      ...this.stats,
   908→      falseMatchRate: ((this.stats.falseMatches / total) * 100).toFixed(2) + '%',
   909→      aiFallbackRate: ((this.stats.aiFallback / total) * 100).toFixed(2) + '%',
   910→      patternHitRate: ((this.stats.patternMatched / total) * 100).toFixed(2) + '%'
   911→    };
   912→  }
   913→}

   916→// SECTION 3: ENHANCED PATTERN STORE (With Ownership)

   918→
   919→class EnhancedPatternStore {
   920→  constructor() {
   921→    this.patterns = new Map(); // key: patternId, value: pattern object
   922→    this.layoutIndex = new Map(); // key: layoutHash, value: [patternIds]
   923→    this.tenantIndex = new Map(); // key: tenantId, value: [patternIds]
   924→    this.supplierIndex = new Map(); // key: supplierId, value: patternId
   925→  }
   926→  
   927→  addPattern(pattern) {
   928→    // Validate required ownership fields
   929→    if (!pattern.tenantId || !pattern.supplierId) {
   930→      throw new Error('Pattern must have tenantId and supplierId (Supplier Ownership Gate)');
   931→    }
   932→    
   933→    const patternId = `pattern_${pattern.supplierId}_${pattern.version || 1}`;
   934→    pattern.id = patternId;
   935→    pattern.createdAt = new Date().toISOString();
   936→    
   937→    this.patterns.set(patternId, pattern);
   938→    
   939→    // Index by layout
   940→    if (!this.layoutIndex.has(pattern.layoutHash)) {
   941→      this.layoutIndex.set(pattern.layoutHash, []);
   942→    }
   943→    this.layoutIndex.get(pattern.layoutHash).push(patternId);
   944→    
   945→    // Index by tenant
   946→    if (!this.tenantIndex.has(pattern.tenantId)) {
   947→      this.tenantIndex.set(pattern.tenantId, []);
   948→    }
   949→    this.tenantIndex.get(pattern.tenantId).push(patternId);
   950→    
   951→    // Index by supplier
   952→    this.supplierIndex.set(pattern.supplierId, patternId);
   953→    
   954→    return patternId;
   955→  }
   956→  
   957→  findByLayout(layoutHash) {
   958→    const patternIds = this.layoutIndex.get(layoutHash) || [];
   959→    return patternIds.map(id => this.patterns.get(id)).filter(Boolean);
   960→  }
   961→  
   962→  findByTenant(tenantId) {
   963→    const patternIds = this.tenantIndex.get(tenantId) || [];
   964→    return patternIds.map(id => this.patterns.get(id)).filter(Boolean);
   965→  }
   966→  
   967→  findBySupplier(supplierId) {
   968→    const patternId = this.supplierIndex.get(supplierId);
   969→    return patternId ? this.patterns.get(patternId) : null;
   970→  }
   971→  
   972→  getPatternCount() {
   973→    return this.patterns.size;
   974→  }
   975→  
   976→  getStats() {
   977→    return {
   978→      totalPatterns: this.patterns.size,
   979→      uniqueLayouts: this.layoutIndex.size,
   980→      tenants: this.tenantIndex.size,
   981→      suppliers: this.supplierIndex.size
   982→    };
   983→  }
   984→}

   990→class DecisionTrace {
   991→  constructor(invoiceId) {
   992→    this.invoiceId = invoiceId;
   993→    this.steps = [];
   994→    this.details = [];
   995→    this.startedAt = new Date().toISOString();
   996→    this.completedAt = null;
   997→  }
   998→  
   999→  addStep(stepName, message, data = null) {
  1000→    const step = {
  1001→      step: stepName,
  1002→      message: message,
  1003→      timestamp: new Date().toISOString(),
  1004→      data: data
  1005→    };
  1006→    this.steps.push(step);
  1007→    return this;
  1008→  }
  1009→  
  1010→  addDetail(category, message, data = null) {
  1011→    const detail = {
  1012→      category: category,
  1013→      message: message,
  1014→      timestamp: new Date().toISOString(),
  1015→      data: data
  1016→    };
  1017→    this.details.push(detail);
  1018→    return this;
  1019→  }
  1020→  
  1021→  complete(outcome) {
  1022→    this.completedAt = new Date().toISOString();
  1023→    this.outcome = outcome;
  1024→    return this;
  1025→  }
  1026→  
  1027→  getTrace() {
  1028→    return {
  1029→      invoiceId: this.invoiceId,
  1030→      startedAt: this.startedAt,
  1031→      completedAt: this.completedAt,
  1032→      outcome: this.outcome,
  1033→      steps: this.steps,
  1034→      details: this.details,
  1035→      summary: this.generateSummary()
  1036→    };
  1037→  }
  1038→  
  1039→  generateSummary() {
  1040→    const passedStages = this.steps.filter(s => s.step.includes('pass')).map(s => s.step);
  1041→    const failedStage = this.steps.find(s => s.step.includes('fail'));
  1042→    
  1043→    return {
  1044→      totalSteps: this.steps.length,
  1045→      stagesPassed: passedStages,
  1046→      failedAt: failedStage?.step || null,
  1047→      finalOutcome: this.outcome
  1048→    };
  1049→  }
  1050→  
  1051→  // Pretty print for debugging
  1052→  prettyPrint() {
  1053→    console.log(`\n📋 Decision Trace: ${this.invoiceId}`);
  1054→    console.log('═'.repeat(60));
  1055→    
  1056→    for (const step of this.steps) {
  1057→      const icon = step.step.includes('pass') ? '✅' : 
  1058→                   step.step.includes('fail') ? '❌' :
  1059→                   step.step.includes('fallback') ? '🔄' : '➡️';

  1134→  buildPatternStore(suppliers) {
  1135→    // Use same layout hash computation as invoice generator for consistency

  1139→        const char = formatType.charCodeAt(i);
  1140→        hash = ((hash << 5) - hash) + char;
  1141→        hash = hash & hash;
  1142→      }
  1143→      return `layout_${Math.abs(hash).toString(16).padStart(8, '0')}`;
  1144→    };

  1198→  async runBaselineTest(invoices) {
  1199→    // Simulate ORIGINAL method (layout-only fingerprinting)
  1200→    let patternMatched = 0;
  1201→    let aiFallback = 0;
  1202→    let falseMatches = 0;
  1203→    let collisions = 0;
  1204→    
  1205→    for (const invoice of invoices) {
  1206→      // Original simple method: match by layout only
  1207→      const candidates = this.engine.patternStore.findByLayout(invoice.layoutHash);
  1208→      
  1209→      if (candidates.length === 0) {
  1210→        aiFallback++;
  1211→        continue;
  1212→      }
  1213→      
  1214→      if (candidates.length > 1) {
  1215→        // Random selection simulates collision/false match possibility
  1216→        const randomPick = candidates[Math.floor(Math.random() * candidates.length)];
  1217→        if (randomPick.supplierId !== invoice.supplierId) {
  1218→          falseMatches++;
  1219→        } else {
  1220→          patternMatched++;
  1221→        }
  1222→      } else {
  1223→        if (candidates[0].supplierId !== invoice.supplierId) {
  1224→          falseMatches++; // False match even with single candidate (wrong pattern stored)
  1225→        } else {
  1226→          patternMatched++;
  1227→        }
  1228→      }
  1229→    }
  1230→

  1350→  determineFalseMatchCause(trace) {
  1351→    const traceData = trace.trace;
  1352→    
  1353→    // Analyze trace to find root cause
  1354→    if (traceData.details?.some(d => d.category === 'supplier_identity')) {
  1355→      return 'weak_supplier_validation';

  1357→    if (traceData.details?.some(d => d.category === 'currency_check' && d.message?.includes('vs'))) {
  1358→      return 'currency_mismatch_ignored';
  1359→    }
  1360→    if (traceData.steps?.some(s => s.data?.hasSharedLayout)) {
  1361→      return 'similar_layout';
  1362→    }
  1363→    if (traceData.steps?.some(s => s.message?.includes('OCR'))) {
  1364→      return 'ocr_error';
  1365→    }

  1367→    return 'unknown';

  1370→  determineFailedStage(trace) {

  1376→      if (step.step.includes('stage4') || step.step.includes('cross_field')) return 'Stage 4: Cross-Field';
  1377→    }
  1378→    return 'Unknown';
  1379→  }
