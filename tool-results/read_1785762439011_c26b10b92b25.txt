    26→// ─── AppUser (replaces User — matches db.appUser) ───
    27→
    28→// AppUser — the canonical user model used across all auth + RBAC code.
    29→// Fields `passwordHash`, `displayName`, `companies`, `permissions` are
    30→// intentionally NOT camelCased — the auth API routes (register/login/me/
    31→// refresh/change-password/forgot/reset) and src/lib/auth.ts read/write
    32→// them by these exact names. JSON-encoded fields (`companies`, `permissions`)
    33→// store arrays/objects as TEXT to avoid join tables for the simple RBAC
    34→// model. `role` defaults to "employee" (NOT "user") so new signups match
    35→// ROLE_DEFAULTS in AuthContext.
    36→model AppUser {
    37→  uid           String   @id @default(cuid())
    38→  email         String   @unique
    39→  passwordHash  String
    40→  displayName   String   @default("")
    41→  role          String   @default("employee")
    42→  companies     String   @default("[]") // JSON array of company slugs
    43→  permissions   String   @default("{}") // JSON object of permission → 0/1
    44→  emailVerified Boolean  @default(false)
    45→  tokenVersion  Int      @default(0)
    46→  createdAt     DateTime @default(now())
    47→  updatedAt     DateTime @updatedAt
    48→
    49→  mFASecrets MFASecret[]
    50→  sessions   SessionRegistry[]
    51→
    52→  @@map("app_users")
    53→}
