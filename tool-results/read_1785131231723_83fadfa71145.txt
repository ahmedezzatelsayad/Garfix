     2→ * rbac.ts — Enterprise-grade Role-Based Access Control with granular permissions.
     3→ *
     4→ * Enhancements over the legacy permissions.ts:
     5→ *   - Permission hierarchy (inherits from parent roles)
     6→ *   - Resource-level permissions (invoice:read, invoice:write, etc.)

   331→export function createCustomRole(input: CustomRoleInput): RoleDefinition {
   332→  // Validate parent role exists
   333→  const parent = ROLE_DEFINITIONS[input.inheritsFrom] || customRoles.get(input.inheritsFrom);
   334→  if (!parent) {
   335→    throw new Error(`Parent role "${input.inheritsFrom}" does not exist`);
   336→  }
   337→  // Validate no circular inheritance
   338→  if (wouldCreateCircularInheritance(input.id, input.inheritsFrom)) {
   339→    throw new Error("Circular inheritance detected");
   340→  }
   341→  // Validate resource names
   342→  for (const rp of input.resourcePermissions) {
   343→    if (!AVAILABLE_RESOURCES.includes(rp.resource)) {
   344→      throw new Error(`Unknown resource: "${rp.resource}"`);
   345→    }
   346→  }
   347→
   348→  const role: RoleDefinition = {
   349→    id: input.id,
   350→    label: input.label,
   351→    labelAr: input.labelAr,
   352→    inheritsFrom: input.inheritsFrom,
   353→    resourcePermissions: input.resourcePermissions,
   354→    flatOverrides: input.flatOverrides,
   355→    timeRestrictions: input.timeRestrictions,
   356→    isBuiltIn: false,
   357→    color: "#6366f1",
   358→  };
   359→  customRoles.set(input.id, role);
   360→  return role;
   361→}

   363→export function deleteCustomRole(id: string): boolean {
   364→  const role = customRoles.get(id);
   365→  if (!role) return false;
   366→  if (role.isBuiltIn) throw new Error("Cannot delete built-in roles");
   367→  // Check no other custom role inherits from this one
   368→  const allCustom = Array.from(customRoles.entries());
   369→  for (const [cid, crole] of allCustom) {
   370→    if (crole.inheritsFrom === id) {
   371→      throw new Error(`Role "${cid}" inherits from "${id}" — delete or reassign first`);
   372→    }
   373→  }
   374→  customRoles.delete(id);
   375→  return true;
   376→}

   378→export function updateCustomRole(id: string, updates: Partial<CustomRoleInput>): RoleDefinition | undefined {
   379→  const existing = customRoles.get(id);
   380→  if (!existing) return undefined;
   381→  if (existing.isBuiltIn) throw new Error("Cannot modify built-in roles");
   382→
   383→  const updated: RoleDefinition = {
   384→    ...existing,
   385→    ...updates,
   386→    isBuiltIn: false,
   387→  };
   388→
   389→  // Validate inheritance if changed
   390→  if (updates.inheritsFrom) {
   391→    const parent = ROLE_DEFINITIONS[updates.inheritsFrom] || customRoles.get(updates.inheritsFrom);
   392→    if (!parent) throw new Error(`Parent role "${updates.inheritsFrom}" does not exist`);
   393→    if (wouldCreateCircularInheritance(id, updates.inheritsFrom)) {
   394→      throw new Error("Circular inheritance detected");
   395→    }
   396→  }
   397→
   398→  // Validate resources if changed
   399→  if (updates.resourcePermissions) {
   400→    for (const rp of updates.resourcePermissions) {
   401→      if (!AVAILABLE_RESOURCES.includes(rp.resource)) {
   402→        throw new Error(`Unknown resource: "${rp.resource}"`);
   403→      }
   404→    }
   405→  }
   406→
   407→  customRoles.set(id, updated);
   408→  return updated;
   409→}

   411→// ── Permission Hierarchy ─────────────────────────────────────────────────────
   412→
   413→/**
   414→ * Walk the inheritance chain and collect all permissions from ancestor roles.
   415→ * Permissions are merged: higher-level permissions override lower-level ones,
   416→ * wider scopes override narrower ones for the same level.
   417→ */
   418→export function getInheritedPermissions(roleId: string): ResourcePermission[] {
   419→  const chain = getInheritanceChain(roleId);
   420→  const collected: ResourcePermission[] = [];
   421→
   422→  // Process from root to leaf — all permissions are collected.
   423→  // The checkPermission function handles deduplication by finding
   424→  // the best matching permission (preferring wider scope & higher level).
   425→  for (const rid of chain) {
   426→    const def = ROLE_DEFINITIONS[rid] || customRoles.get(rid);
   427→    if (!def) continue;
   428→    collected.push(...def.resourcePermissions);
   429→  }
   430→
   431→  return collected;
   432→}
   433→
   434→/**
   435→ * Get the full inheritance chain from root to the given role.
   436→ */
   437→export function getInheritanceChain(roleId: string): string[] {
   438→  const chain: string[] = [];
   439→  let current: string | null = roleId;
   440→
   441→  while (current) {
   442→    chain.unshift(current); // add at beginning (root first)
   443→    const def = ROLE_DEFINITIONS[current] || customRoles.get(current);
   444→    current = def?.inheritsFrom ?? null;
   445→  }
   446→
   447→  return chain;
   448→}

   450→// ── Permission Checking ──────────────────────────────────────────────────────
   451→
   452→/**
   453→ * Check if a user has a specific permission on a resource at a given scope.
   454→ *
   455→ * This is the core authorization check. It considers:
   456→ *   1. The user's role and inherited permissions
   457→ *   2. Custom per-user overrides (from flat permission system)
   458→ *   3. Time-based restrictions
   459→ *   4. Scope requirements
   460→ *   5. Founder override (full access)
   461→ */
   462→export function checkPermission(
   463→  userPerms: Record<string, number>,
   464→  role: string,
   465→  resource: string,
   466→  action: string,
   467→  scope: PermissionScope,
   468→  isFounder = false,
   469→  customRolePerms?: ResourcePermission[],
   470→  timeRestrictions?: Record<string, TimeRestriction>,
   471→): boolean {
   472→  // Founder always has full access
   473→  if (isFounder) return true;
   474→
   475→  // Admin role has company-wide access to everything
   476→  if (role === "admin") {
   477→    if (scope === PermissionScope.platform) return false; // only founder
   478→    return true;
   479→  }
   480→
   481→  // Collect all resource permissions for the role (inherited + custom)
   482→  const roleDef = ROLE_DEFINITIONS[role] || customRoles.get(role);
   483→  const inheritedPerms = roleDef ? getInheritedPermissions(role) : [];
   484→  const allPerms = [...inheritedPerms, ...(customRolePerms || [])];
   485→
   486→  // Check time restrictions
   487→  if (timeRestrictions?.[resource] || roleDef?.timeRestrictions?.[resource]) {
   488→    const restriction = timeRestrictions?.[resource] || roleDef?.timeRestrictions?.[resource];
   489→    if (restriction && !isWithinTimeRestriction(restriction)) {
   490→      return false;
   491→    }
   492→  }
   493→
   494→  // Find matching permission
   495→  const matchingPerm = allPerms.find((rp) => {
   496→    if (rp.resource !== resource) return false;
   497→    const permAction = levelToAction(rp.level);
   498→    // Check if the granted action covers the requested action
   499→    if (!actionCovers(permAction, action)) return false;
   500→    // Check if the granted scope covers the requested scope
   501→    if (!scopeCovers(rp.scope, scope)) return false;
   502→    return true;
   503→  });
   504→
   505→  if (matchingPerm) return true;
   506→
   507→  // Fall back to legacy flat permission system
   508→  return checkLegacyPermission(userPerms, resource, action);
   509→}

   511→/**
   512→ * Map PermissionLevel to action string.
   513→ */
   514→function levelToAction(level: PermissionLevel): string {
   515→  switch (level) {
   516→    case PermissionLevel.read: return "read";
   517→    case PermissionLevel.write: return "write";
   518→    case PermissionLevel.approve: return "approve";
   519→    case PermissionLevel.admin: return "admin";
   520→    case PermissionLevel.delete: return "delete";
   521→    case PermissionLevel.print: return "print";
   522→    case PermissionLevel.export: return "export";
   523→    case PermissionLevel.e_invoice: return "e_invoice";
   524→    case PermissionLevel.bulk_input: return "bulk_input";
   525→    case PermissionLevel.manage_prices: return "manage_prices";
   526→    default: return "none";
   527→  }
   528→}
   529→
   530→/**
   531→ * Check if a granted action covers a requested action.
   532→ * Hierarchy: admin > approve > write > read > none
   533→ * "admin" covers everything. "approve" covers write + read. etc.
   534→ */
   535→function actionCovers(granted: string, requested: string): boolean {
   536→  // Special actions that don't follow the hierarchy (e.g., "print", "export")
   537→  const SPECIAL_ACTIONS = ["print", "export", "e_invoice", "bulk_input", "manage_prices",
   538→    "manage", "manage_periods", "reconcile", "file", "initiate", "convert",
   539→    "generate", "calculate", "manage_permissions"];
   540→
   541→  if (granted === "admin") return true;
   542→  if (granted === requested) return true;
   543→
   544→  // Special actions require explicit grant OR admin level
   545→  if (SPECIAL_ACTIONS.includes(requested)) return granted === "admin";
   546→
   547→  // Hierarchy: approve > write > read
   548→  const hierarchy: Record<string, number> = { none: 0, read: 1, write: 2, approve: 3, admin: 4 };
   549→  const grantedLevel = hierarchy[granted] ?? 0;
   550→  const requestedLevel = hierarchy[requested] ?? 0;
   551→  return grantedLevel >= requestedLevel;
   552→}
   553→
   554→/**
   555→ * Check if a granted scope covers a requested scope.
   556→ * Hierarchy: platform > company > team > own
   557→ */
   558→function scopeCovers(granted: PermissionScope, requested: PermissionScope): boolean {
   559→  const hierarchy: Record<string, number> = { own: 1, team: 2, company: 3, platform: 4 };
   560→  return hierarchy[granted] >= hierarchy[requested];
   561→}

   590→// ── Time Restriction Checking ────────────────────────────────────────────────
   591→
   592→/**
   593→ * Check if the current time is within a time restriction window.
   594→ */
   595→export function isWithinTimeRestriction(restriction: TimeRestriction): boolean {
   596→  const now = new Date();
   597→
   598→  // Check day-of-week restriction
   599→  if (restriction.allowedDays && restriction.allowedDays.length > 0) {
   600→    // ISO day: 1=Mon, 7=Sun. JS getDay: 0=Sun, 1=Mon, ... 6=Sat
   601→    const jsDay = now.getDay();
   602→    const isoDay = jsDay === 0 ? 7 : jsDay;
   603→    if (!restriction.allowedDays.includes(isoDay)) return false;
   604→  }
   605→
   606→  // Check hour restriction
   607→  if (restriction.startHour !== undefined && restriction.endHour !== undefined) {
   608→    const hour = now.getHours();
   609→    if (hour < restriction.startHour || hour >= restriction.endHour) return false;
   610→  }
   611→
   612→  // Check date range restriction
   613→  if (restriction.startDate) {
   614→    const start = new Date(restriction.startDate);
   615→    if (now < start) return false;
   616→  }
   617→  if (restriction.endDate) {
   618→    const end = new Date(restriction.endDate);
   619→    if (now > end) return false;
   620→  }
   621→
   622→  return true;
   623→}

   627→/**
   628→ * Compute the full effective permission set for a user.
   629→ * Combines the legacy flat system with the new RBAC system.
   630→ */
   631→export function getEffectivePermissions(
   632→  role: string,
   633→  customPerms: Record<string, number> | null | undefined,
   634→  isFounder: boolean,
   635→  customRolePerms?: ResourcePermission[],
   636→  timeRestrictions?: Record<string, TimeRestriction>,
   637→): {
   638→  flat: Record<string, number>;
   639→  resources: ResourcePermission[];
   640→  inheritanceChain: string[];
   641→} {
   642→  // Get legacy flat permissions
   643→  const flat = computeEffectivePermissions(role, customPerms, isFounder);
   644→
   645→  // Get resource-level permissions from role hierarchy
   646→  const roleDef = ROLE_DEFINITIONS[role] || customRoles.get(role);
   647→  const inheritedPerms = roleDef ? getInheritedPermissions(role) : [];
   648→  const resources = [...inheritedPerms, ...(customRolePerms || [])];
   649→
   650→  // Founder gets platform scope on all resources
   651→  if (isFounder) {
   652→    const founderPerms: ResourcePermission[] = AVAILABLE_RESOURCES.map((r) => ({
   653→      resource: r,
   654→      level: PermissionLevel.admin,
   655→      scope: PermissionScope.platform,
   656→    }));
   657→    return { flat, resources: founderPerms, inheritanceChain: ["founder"] };
   658→  }
   659→
   660→  const inheritanceChain = roleDef ? getInheritanceChain(role) : [role];
   661→
   662→  return { flat, resources, inheritanceChain };
   663→}

   665→// ── Validate Permission Changes ──────────────────────────────────────────────
   666→
   667→/**
   668→ * Validate that a permission change is safe to apply.
   669→ * Prevents:
   670→ *   - Removing locked permissions from non-admin users
   671→ *   - Granting platform scope to non-founders
   672→ *   - Granting admin level on locked resources to non-admins
   673→ *   - Creating roles with invalid inheritance
   674→ */
   675→export function validatePermissionChange(
   676→  change: {
   677→    type: "grant" | "revoke" | "modify";
   678→    targetRole: string;
   679→    resource: string;
   680→    level: PermissionLevel;
   681→    scope: PermissionScope;
   682→  },
   683→  actorRole: string,
   684→  isFounder: boolean,
   685→): { valid: boolean; reason?: string } {
   686→  // Only admin or founder can modify permissions
   687→  if (actorRole !== "admin" && !isFounder) {
   688→    return { valid: false, reason: "Only admin or founder can modify permissions" };
   689→  }
   690→
   691→  // Platform scope is founder-only
   692→  if (change.scope === PermissionScope.platform && !isFounder) {
   693→    return { valid: false, reason: "Platform scope requires founder access" };
   694→  }
   695→
   696→  // Admin level on locked resources requires founder
   697→  const LOCKED_RESOURCES = ["settings", "report", "audit", "role", "feature_flag"];
   698→  if (LOCKED_RESOURCES.includes(change.resource) && change.level >= PermissionLevel.admin && !isFounder) {
   699→    return { valid: false, reason: `Admin level on "${change.resource}" requires founder access` };
   700→  }
   701→
   702→  // Cannot revoke permissions from a role higher than your own
   703→  const actorDef = ROLE_DEFINITIONS[actorRole];
   704→  const targetDef = ROLE_DEFINITIONS[change.targetRole] || customRoles.get(change.targetRole);
   705→  if (actorDef && targetDef) {
   706→    const actorChain = getInheritanceChain(actorRole);
   707→    const targetChain = getInheritanceChain(change.targetRole);
   708→    // If the target role's chain includes the actor's role, the actor can't revoke
   709→    if (targetChain.includes(actorRole) && change.type === "revoke") {
   710→      // Actually this is fine — you can revoke from roles below you
   711→    }
   712→  }
   713→
   714→  // Validate the resource exists
   715→  if (!AVAILABLE_RESOURCES.includes(change.resource)) {
   716→    return { valid: false, reason: `Unknown resource: "${change.resource}"` };
   717→  }
   718→
   719→  return { valid: true };
   720→}

   722→// ── Circular Inheritance Check ───────────────────────────────────────────────
   723→
   724→function wouldCreateCircularInheritance(newRoleId: string, inheritsFrom: string): boolean {
   725→  let current: string | null = inheritsFrom;
   726→  const visited = new Set<string>();
   727→
   728→  while (current) {
   729→    if (current === newRoleId) return true;
   730→    if (visited.has(current)) return true; // existing circular chain
   731→    visited.add(current);
   732→    const def = ROLE_DEFINITIONS[current] || customRoles.get(current);
   733→    current = def?.inheritsFrom ?? null;
   734→  }
   735→
   736→  return false;
   737→}

   754→/**
   755→ * Convert a flat permission map to ResourcePermission array.
   756→ * Useful for migrating existing users to the new system.
   757→ */
   758→export function flatToResourcePerms(flat: Record<string, number>): ResourcePermission[] {
   759→  const perms: ResourcePermission[] = [];
   760→
   761→  const FLAT_TO_RESOURCE: Record<string, { resource: string; action: string; scope: PermissionScope }> = {
   762→    create_invoice: { resource: "invoice", action: "write", scope: PermissionScope.own },
   763→    print_invoice: { resource: "invoice", action: "print", scope: PermissionScope.own },
   764→    edit_invoice: { resource: "invoice", action: "write", scope: PermissionScope.own },
   765→    delete_invoice: { resource: "invoice", action: "delete", scope: PermissionScope.own },
   766→    view_customers: { resource: "customer", action: "read", scope: PermissionScope.company },
   767→    edit_customer: { resource: "customer", action: "write", scope: PermissionScope.own },
   768→    delete_customer: { resource: "customer", action: "delete", scope: PermissionScope.own },
   769→    edit_inventory: { resource: "inventory", action: "write", scope: PermissionScope.own },
   770→    manage_wholesale_prices: { resource: "inventory", action: "manage_prices", scope: PermissionScope.own },
   771→    bulk_input: { resource: "inventory", action: "bulk_input", scope: PermissionScope.own },
   772→    export_data: { resource: "report", action: "export", scope: PermissionScope.company },
   773→    reports_access: { resource: "report", action: "read", scope: PermissionScope.company },
   774→    settings_access: { resource: "settings", action: "write", scope: PermissionScope.company },
   775→    finance_access: { resource: "accounting", action: "write", scope: PermissionScope.company },
   776→    employee_management: { resource: "employee", action: "write", scope: PermissionScope.company },
   777→    e_invoicing_submit: { resource: "invoice", action: "e_invoice", scope: PermissionScope.company },
   778→  };
   779→
   780→  for (const [key, value] of Object.entries(flat)) {
   781→    if (value && FLAT_TO_RESOURCE[key]) {
   782→      const mapping = FLAT_TO_RESOURCE[key];
   783→      perms.push({
   784→        resource: mapping.resource,
   785→        level: actionToLevel(mapping.action),
   786→        scope: mapping.scope,
   787→      });
   788→    }
   789→  }
   790→
   791→  return perms;
   792→}
   793→
   794→function actionToLevel(action: string): PermissionLevel {
   795→  switch (action) {
   796→    case "read": return PermissionLevel.read;
   797→    case "write": return PermissionLevel.write;
   798→    case "approve": return PermissionLevel.approve;
   799→    case "admin": return PermissionLevel.admin;
   800→    case "delete": return PermissionLevel.delete;
   801→    case "print": return PermissionLevel.print;
   802→    case "export": return PermissionLevel.export;
   803→    case "e_invoice": return PermissionLevel.e_invoice;
   804→    case "bulk_input": return PermissionLevel.bulk_input;
   805→    case "manage_prices": return PermissionLevel.manage_prices;
   806→    default: return PermissionLevel.write; // unknown special actions default to write
   807→  }
   808→}
