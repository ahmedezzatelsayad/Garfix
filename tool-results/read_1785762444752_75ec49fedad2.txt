   151→export default function CompaniesPerFeatureAIPage() {
   152→  // ── State ───────────────────────────────────────────────
   153→  
   154→  const [companies, setCompanies] = useState<CompanyData[]>([]);
   155→  const [isLoading, setIsLoading] = useState(true);

   159→  const [selectedCompany, setSelectedCompany] = useState<CompanyData | null>(null);
   160→  const [showConfigModal, setShowConfigModal] = useState(false);
   161→  const [aiConfig, setAiConfig] = useState<CompanyAIConfigData | null>(null);
   162→  const [isSaving, setIsSaving] = useState(false);
   163→  const [isTesting, setIsTesting] = useState(false);
   164→  const [testingFeature, setTestingFeature] = useState<string | null>(null);
   165→  
   166→  // Form state (per-feature)
   167→  const [featureKeys, setFeatureKeys] = useState<Record<string, string>>({
   168→    chat: '',
   169→    invoice: '',
   170→    parse: '',
   171→    memory: '',
   172→  });
   173→  const [showKeys, setShowKeys] = useState<Record<string, boolean>>({
   174→    chat: false,
   175→    invoice: false,
   176→    parse: false,
   177→    memory: false,
   178→  });
   179→  const [testResults, setTestResults] = useState<Record<string, TestResult | null>>({});
   180→  
   181→  // Alert state
   182→  const [alert, setAlert] = useState<{ type: 'success' | 'error'; message: string } | null>(null);
   183→
   184→  // ── Effects ─────────────────────────────────────────────
   185→  
   186→  useEffect(() => {
   187→    fetchCompanies();
   188→  }, []);
   189→
   190→  // ── API Calls ────────────────────────────────────────────
   191→
   192→  /**
   193→   * Fetch all companies with their AI status
   194→   */
   195→  const fetchCompanies = async () => {
   196→    try {
   197→      setIsLoading(true);
   198→      const response = await fetch('/api/founder-panel/companies?includeAIStatus=true');
   199→      const data = await response.json();
   200→      
   201→      if (data.success) {
   202→        setCompanies(data.data.companies || []);
   203→      } else {
   204→        setAlert({ type: 'error', message: data.error || 'فشل تحميل الشركات' });
   205→      }
   206→    } catch (error) {
   207→      console.error('Fetch companies error:', error);
   208→      setAlert({ type: 'error', message: 'خطأ في الاتصال بالخادم' });
   209→    } finally {
   210→      setIsLoading(false);
   211→    }
   212→  };
   213→
   214→  /**
   215→   * Fetch AI config for a specific company
   216→   */
   217→  const fetchCompanyAIConfig = async (companyId: string) => {
   218→    try {
   219→      const response = await fetch(`/api/founder-panel/ai-config?companySlug=${companyId}`);
   220→      const data = await response.json();
   221→      
   222→      if (data.success) {
   223→        setAiConfig(data.data);
   224→        
   225→        // Initialize form keys (masked if exists)
   226→        setFeatureKeys({
   227→          chat: data.data.chat?.hasApiKey ? '••••••••' : '',
   228→          invoice: data.data.invoice?.hasApiKey ? '••••••••' : '',
   229→          parse: data.data.parse?.hasApiKey ? '••••••••' : '',
   230→          memory: data.data.memory?.hasApiKey ? '••••••••' : '',
   231→        });
   232→        
   233→        return data.data;
   234→      }
   235→      return null;
   236→    } catch (error) {
   237→      console.error('Fetch AI config error:', error);
   238→      return null;
   239→    }
   240→  };
   241→
   242→  /**
   243→   * Save all feature keys for a company
   244→   */
   245→  const saveAllFeatures = async () => {
   246→    if (!selectedCompany || !aiConfig) {
   247→      setAlert({ type: 'error', message: 'يرجى اختيار شركة أولاً' });
   248→      return;
   249→    }
   250→
   251→    try {
   252→      setIsSaving(true);
   253→      
   254→      const response = await fetch('/api/founder-panel/ai-config', {
   255→        method: 'PUT',
   256→        headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({
   258→          chat: {
   259→            apiKey: featureKeys.chat,
   260→            model: aiConfig.chat.model,
   261→            enabled: true,
   262→            rateLimitRpm: aiConfig.chat.rateLimitRpm,
   263→          },
   264→          invoice: {
   265→            apiKey: featureKeys.invoice,
   266→            model: aiConfig.invoice.model,
   267→            enabled: true,
   268→            rateLimitRpm: aiConfig.invoice.rateLimitRpm,
   269→          },
   270→          parse: {
   271→            apiKey: featureKeys.parse,
   272→            model: aiConfig.parse.model,
   273→            enabled: true,
   274→            rateLimitRpm: aiConfig.parse.rateLimitRpm,
   275→          },
   276→          memory: {
   277→            apiKey: featureKeys.memory,
   278→            model: aiConfig.memory.model,
   279→            enabled: true,
   280→            rateLimitRpm: aiConfig.memory.rateLimitRpm,
   281→          },
   282→          systemPrompt: '',
   283→          costOptimization: 'balanced',
   284→          notifyHighUsage: true,
   285→          usageNotificationThreshold: 80,
   286→        }),
   287→      });
   288→
   289→      const data = await response.json();
   290→      
   291→      if (data.success) {
   292→        setAlert({ type: 'success', message: `✅ تم حفظ إعدادات AI لـ ${selectedCompany.nameAr || selectedCompany.name} بنجاح!` });
   293→        setShowConfigModal(false);
   294→        resetForm();
   295→        fetchCompanies();
   296→      } else {
   297→        setAlert({ type: 'error', message: data.error || 'فشل الحفظ' });
   298→      }
   299→    } catch (error) {
   300→      console.error('Save error:', error);
   301→      setAlert({ type: 'error', message: 'خطأ في الاتصال' });
   302→    } finally {
   303→      setIsSaving(false);
   304→    }
   305→  };
   306→
   307→  /**
   308→   * Test API connection for a specific feature
   309→   */
   310→  const testFeatureConnection = async (featureKey: string) => {
   311→    const apiKey = featureKeys[featureKey];
   312→    
   313→    if (!apiKey || apiKey === '••••••••') {
   314→      setAlert({ type: 'error', message: 'يرجى إدخال المفتاح أولاً' });
   315→      return;
   316→    }
   317→
   318→    try {
   319→      setIsTesting(true);
   320→      setTestingFeature(featureKey);
   321→      
   322→      const feature = FEATURES.find(f => f.key === featureKey);
   323→      const model = aiConfig?.[featureKey as keyof CompanyAIConfigData]?.model || 'gemini-2.0-flash';
   324→      
   325→      const response = await fetch('/api/founder-panel/ai-config/test', {
   326→        method: 'POST',
   327→        headers: { 'Content-Type': 'application/json' },
   328→        body: JSON.stringify({
   329→          feature: featureKey,
   330→          apiKey: apiKey.trim(),
   331→          model,
   332→        }),
   333→      });
   334→
   335→      const data = await response.json();
   336→      
   337→      if (data.success) {
   338→        setTestResults(prev => ({
   339→          ...prev,
   340→          [featureKey]: data.data,
   341→        }));
   342→        
   343→        if (data.data.success) {
   344→          setAlert({ type: 'success', message: `✅ ${feature?.label}: الاتصال ناجح! (${data.data.latencyMs}ms)` });
   345→        } else {
   346→          setAlert({ type: 'error', message: `❌ ${feature?.label}: ${data.data.error}` });
   347→        }
   348→      }
   349→    } catch (error) {
   350→      console.error('Test error:', error);
   351→      setTestResults(prev => ({
   352→        ...prev,
   353→        [featureKey]: { success: false, latencyMs: 0, model: '', error: 'خطأ في الاتصال', feature: featureKey },
   354→      }));
   355→    } finally {
   356→      setIsTesting(false);
   357→      setTestingFeature(null);
   358→    }
   359→  };
   360→
   361→  // ── Handlers ─────────────────────────────────────────────
   362→
   363→  /**
   364→   * Open modal to configure AI for a company

   366→  const openConfigModal = async (company: CompanyData) => {
   367→    setSelectedCompany(company);
   368→    setTestResults({});
   369→    setAlert(null);
   370→    
   371→    await fetchCompanyAIConfig(company.id);
   372→    setShowConfigModal(true);
   373→  };
   374→
   375→  /**
   376→   * Reset form state

   378→  const resetForm = () => {
   379→    setFeatureKeys({ chat: '', invoice: '', parse: '', memory: '' });
   380→    setShowKeys({ chat: false, invoice: false, parse: false, memory: false });
   381→    setTestResults({});
   382→    setAiConfig(null);
   383→  };
   384→
   385→  /**
   386→   * Update feature key in form
   387→   */
   388→  const updateFeatureKey = (feature: string, value: string) => {
   389→    setFeatureKeys(prev => ({ ...prev, [feature]: value }));
   390→    // Clear test result when key changes
   391→    setTestResults(prev => ({ ...prev, [feature]: null }));
   392→  };
   393→
   394→  /**
   395→   * Filter companies
   396→   */

   398→    if (!searchQuery) return true;
   399→    return (
   400→      company.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   401→      company.nameAr?.includes(searchQuery) ||
   402→      company.slug.includes(searchQuery)
   403→    );
   404→  });
   405→
   406→  // Stats
   407→  const totalCompanies = companies.length;
   408→  const companiesWithAnyKey = companies.filter(c => c.hasAIConfig).length;
   409→

   411→

   424→        />
   425→
   426→        {/* ══ Alert ══ */}
   427→        {alert && (
   428→          <FadeUp delay={100}>
   429→            <GarfixAlert
   430→              type={alert.type}
   431→              message={alert.message}
   432→              onClose={() => setAlert(null)}
   433→              className="mb-6"
   434→            />
   435→          </FadeUp>
   436→        )}
   437→
   438→        {/* ══ Architecture Diagram ══ */}
   439→        <GarfixCard className="mb-8 p-6 bg-gradient-to-r from-emerald-50 to-blue-50 dark:from-emerald-950 dark:to-blue-950">
   440→          <div className="text-center">
   441→            <h3 className="text-lg font-bold text-gray-800 dark:text-white mb-4">
   442→              🏗️ بنية النظام: Per-Feature Isolation
   443→            </h3>
   444→            
   445→            <div className="flex flex-wrap justify-center gap-4 text-sm">
   446→              <div className="bg-blue-100 dark:bg-blue-900 px-4 py-2 rounded-lg">
   447→                <span className="text-xl">💬</span> Chat Token
   448→              </div>
   449→              <span className="text-gray-400">+</span>
   450→              <div className="bg-emerald-100 dark:bg-emerald-900 px-4 py-2 rounded-lg">
   451→                <span className="text-xl">📄</span> Invoice Token
   452→              </div>
   453→              <span className="text-gray-400">+</span>
   454→              <div className="bg-purple-100 dark:bg-purple-900 px-4 py-2 rounded-lg">
   455→                <span className="text-xl">🔍</span> Parse Token
   456→              </div>
   457→              <span className="text-gray-400">+</span>
   458→              <div className="bg-amber-100 dark:bg-amber-900 px-4 py-2 rounded-lg">
   459→                <span className="text-xl">🧠</span> Memory Token
   460→              </div>
   461→              <span className="text-gray-400">=</span>
   462→              <div className="bg-gray-800 text-white px-4 py-2 rounded-lg font-bold">
   463→                ✅ No Bottleneck!

   465→            </div>
   466→            
   467→            <p className="text-xs text-gray-600 dark:text-gray-400 mt-3">
   468→              كل شركة → 4 توكنات منعزلة → 1000 شركة × 4 = 4000 اتصال موزع بدون ضغط
   469→            </p>
   470→          </div>

   472→
   473→        {/* ══ Stats Cards ══ */}
   474→        <GarfixGrid cols={3} className="mb-8">
   475→          <MotionCard>
   476→            <div className="text-center p-4">
   477→              <div className="text-3xl font-bold text-emerald-600">{totalCompanies}</div>
   478→              <div className="text-sm text-gray-500 mt-1">إجمالي الشركات</div>
   479→            </div>
   480→          </MotionCard>
   481→
   482→          <MotionCard>
   483→            <div className="text-center p-4">
   484→              <div className="text-3xl font-bold text-blue-600">{companiesWithAnyKey}</div>
   485→              <div className="text-sm text-gray-500 mt-1">شركات بها AI ⚡</div>
   486→            </div>
   487→          </MotionCard>
   488→
   489→          <MotionCard>
   490→            <div className="text-center p-4">
   491→              <div className="text-3xl font-bold text-purple-600">4</div>

   494→          </MotionCard>

   496→
   497→        {/* ══ Search Bar ══ */}

   518→            <div className="text-6xl mb-4">🏢</div>

   520→            <p className="text-gray-500">الشركات التي تسجل ستظهر هنا تلقائياً</p>
   521→          </GarfixCard>
   522→        ) : (
   523→          <div className="space-y-4">
   524→            {filteredCompanies.map((company, index) => (
   525→              <FadeUp key={company.id} delay={index * 50}>
   526→                <GarfixCard 
   527→                  className={cn(
   528→                    "hover:shadow-lg transition-all duration-300",
   529→                    company.hasAIConfig && "border-l-4 border-l-emerald-500"
   530→                  )}
   531→                >
   532→                  <div className="p-6">
   533→                    <div className="flex flex-col lg:flex-row lg:items-center lg:justify-between gap-4">
   534→                      
   535→                      {/* Company Info */}
   536→                      <div className="flex items-center gap-4 flex-1">
   537→                        <div 
   538→                          className="w-14 h-14 rounded-xl flex items-center justify-center text-2xl"
   539→                          style={{ backgroundColor: company.color || '#f0fdf4' }}
   540→                        >
   541→                          {company.emoji || '🏢'}

   544→                        <div className="flex-1">

   550→                          {/* Feature Keys Status */}
