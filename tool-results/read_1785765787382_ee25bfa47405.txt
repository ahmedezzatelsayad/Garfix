   770→function InvoiceForm({

   774→  editing: Invoice | null;
   775→  onClose: () => void;
   776→  onSaved: (reviewQueueWarnings: string[], inventoryWarnings?: string[]) => void;
   777→}) {
   778→  const createInvoiceMutation = useCreateInvoice();
   779→  const updateInvoiceMutation = useUpdateInvoice();
   780→  const updateInvoiceStatusMutation = useUpdateInvoiceStatus();
   781→
   782→  const [invoiceNumber, setInvoiceNumber] = useState(editing?.invoiceNumber || `INV-${Date.now().toString().slice(-6)}`);

   785→  const [clientPhone, setClientPhone] = useState(editing?.clientPhone || "");
   786→  const [clientAddress, setClientAddress] = useState(editing?.clientAddress || "");
   787→  const [issueDate, setIssueDate] = useState(editing?.issueDate || todayStr());
   788→  const [dueDate, setDueDate] = useState(editing?.dueDate || addDaysStr(30));
   789→  const [status, setStatus] = useState(editing?.status || "draft");
   790→  const [lineItems, setLineItems] = useState<LineItem[]>(
   791→    editing?.lineItems?.length ? editing.lineItems : [{ description: "", qty: 1, price: 0 }]
   792→  );
   793→  const [taxRate, setTaxRate] = useState(editing?.taxRate ?? parseFloat(company.defaultTaxRate || "0"));
   794→  const [shipping, setShipping] = useState(editing?.shipping ?? 0);
   795→  const [discount, setDiscount] = useState(editing?.discount ?? 0);
   796→  const [notes, setNotes] = useState(editing?.notes || "");
   797→  const [saving, setSaving] = useState(false);

   805→    setLineItems((items) => items.map((it, idx) => {

   809→        next.total = Number(next.qty) * Number(next.price);
   810→      }
   811→      return next;
   812→    }));
   813→  };
   814→  const addItem = () => setLineItems((items) => [...items, { description: "", qty: 1, price: 0 }]);
   815→  const removeItem = (i: number) => setLineItems((items) => items.filter((_, idx) => idx !== i));

   820→      setSelectedProducts(prev => ({ ...prev, [index]: product }));
   821→      updateItem(index, "description", product.name);
   822→      if (product.sellingPrice != null) {
   823→        updateItem(index, "price", product.sellingPrice);
   824→      }
   825→    } else {
   826→      setSelectedProducts(prev => {
   827→        const next = { ...prev };
   828→        delete next[index];
   829→        return next;
   830→      });

   843→    }
   844→  };
   845→
   846→  const subtotal = lineItems.reduce((s, it) => s + (Number(it.qty) * Number(it.price)), 0);
   847→  const discounted = Math.max(0, subtotal - Number(discount));
   848→  const taxAmount = (discounted * Number(taxRate)) / 100;
   849→  const total = discounted + taxAmount + Number(shipping);
   850→
   851→  const handleSubmit = async () => {
   852→    if (!invoiceNumber || !clientName) {
   853→      toast.error("رقم الفاتورة واسم العميل مطلوبان");
   854→      return;
   855→    }
   856→    setSaving(true);
   857→    try {
   858→      // SECURITY: `status` is NOT sent to the general edit endpoint. For edits,
   859→      // status changes are routed through PATCH /api/invoices/[id]/status
   860→      // (operational statuses only) after the general update succeeds. The
   861→      // `paid`/`partial` statuses are never set here — they result from a real
   862→      // payment via the "تسجيل دفعة" action (PATCH /api/invoices/[id]/payment,
   863→      // which requires finance_access + updates `paid` + audits). For new
   864→      // invoices (POST) the initial status is allowed.
   865→      const isEdit = !!editing;

   883→      } else {
   884→        const result = await createInvoiceMutation.mutateAsync(payload as CreateInvoicePayload);
   885→        // Task 14: capture review-queue / oversell warnings from POST /api/invoices.
   886→        if (Array.isArray(result.reviewQueueWarnings)) {
   887→          createdWarnings = result.reviewQueueWarnings;

   889→        if (Array.isArray(result.warnings)) {

   895→      // route it through the dedicated /status endpoint (writes an audit

   897→      if (isEdit && editing!.status !== status) {

   899→          await updateInvoiceStatusMutation.mutateAsync({ id: editing!.id, status });
   900→        } catch (statusErr) {

   911→        toast.warning(`⚠️ ${createdWarnings.length} صنف يحتاج مراجعة — انظر البانر أدناه`);
   912→      }
   913→      if (createdInventoryWarnings.length > 0) {
   914→        toast.warning(`⚠️ ${createdInventoryWarnings.length} تحذير من المختزن — انظر البانر أدناه`);
   915→      }
   916→      onSaved(createdWarnings, createdInventoryWarnings);
