   103→
   104→// ── Main Component ──────────────────────────────────────────
   105→
   106→export default function FounderApiKeyPoolPage() {
   107→  // ── State ───────────────────────────────────────────────
   108→  
   109→  const [keys, setKeys] = useState<ApiKeyData[]>([]);
   110→  const [stats, setStats] = useState<PoolStats | null>(null);
   111→  const [isLoading, setIsLoading] = useState(true);
   112→  
   113→  // Modal state
   114→  const [showAddModal, setShowAddModal] = useState(false);
   115→  const [isAdding, setIsAdding] = useState(false);
   116→  
   117→  // Form state
   118→  const [newKeysText, setNewKeysText] = useState('');
   119→  const [selectedProvider, setSelectedProvider] = useState('openrouter');
   120→  const [selectedModel, setSelectedModel] = useState(DEFAULT_MODEL);
   121→  const [notes, setNotes] = useState('');
   122→  
   123→  // Alert state
   124→  const [alert, setAlert] = useState<{ type: 'success' | 'error' | 'warning'; message: string } | null>(null);
   125→
   126→  // Pagination & Filter state
   127→  const [currentPage, setCurrentPage] = useState(1);
   128→  const [pageSize, setPageSize] = useState(10);
   129→  const [searchQuery, setSearchQuery] = useState('');
   130→  const [statusFilter, setStatusFilter] = useState<string>('all');
   131→  
   132→  // Select All state
   133→  const [selectedKeyIds, setSelectedKeyIds] = useState<Set<string>>(new Set());
   134→
   135→  // ── Effects ─────────────────────────────────────────────
   136→  
   137→  useEffect(() => {
   138→    fetchPoolData();
   139→  }, []);
   140→
   141→  // ── API Calls ────────────────────────────────────────────
   142→
   143→  /**
   144→   * Fetch pool data (keys + stats)
   145→   */
   146→  const fetchPoolData = async () => {
   147→    try {
   148→      setIsLoading(true);
   149→      
   150→      const response = await fetch('/api/founder-panel/api-key-pool');
   151→      const data = await response.json();
   152→      
   153→      if (data.success) {
   154→        setKeys(data.data.keys || []);
   155→        setStats(data.data.stats || null);
   156→        
   157→        // Check for low key alert
   158→        if (data.data.stats?.keysRunningLow) {
   159→          setAlert({
   160→            type: 'warning',
   161→            message: `⚠️ تنبيه: باقي ${data.data.stats.availableKeys} مفتاح فقط! حتضيف مفاتيح جديدة.`,
   162→          });
   163→        }
   164→      } else {
   165→        setAlert({ type: 'error', message: data.error || 'فشل تحميل البيانات' });
   166→      }
   167→    } catch (error) {
   168→      console.error('Fetch pool error:', error);
   169→      setAlert({ type: 'error', message: 'خطأ في الاتصال بالخادم' });
   170→    } finally {
   171→      setIsLoading(false);
   172→    }
   173→  };
   174→
   175→  /**
   176→   * Add new keys to the pool
   177→   */
   178→  const addKeysToPool = async () => {
   179→    if (!newKeysText.trim()) {
   180→      setAlert({ type: 'error', message: 'يرجى إدخال مفاتيح API واحدة على الأقل' });
   181→      return;
   182→    }
   183→
   184→    // Parse keys from text (one per line or comma-separated)
   185→    const parsedKeys = newKeysText
   186→      .split(/[\n,]+/)
   187→      .map(k => k.trim())
   188→      .filter(k => k.length > 10); // Basic validation
   189→    
   190→    if (parsedKeys.length === 0) {
   191→      setAlert({ type: 'error', message: 'لم يتم العثور على مفاتيح صالحة' });
   192→      return;
   193→    }
   194→
   195→    try {
   196→      setIsAdding(true);
   197→      
   198→      const response = await fetch('/api/founder-panel/api-key-pool', {
   199→        method: 'POST',
   200→        headers: { 'Content-Type': 'application/json' },
   201→        body: JSON.stringify({
   202→          keys: parsedKeys,
   203→          provider: selectedProvider,
   204→          model: selectedModel,
   205→          notes: notes || undefined,
   206→        }),
   207→      });
   208→
   209→      const data = await response.json();
   210→      
   211→      if (data.success) {
   212→        setAlert({ 
   213→          type: 'success', 
   214→          message: `✅ تم إضافة ${data.data.addedCount} مفتاح بنجاح!` 
   215→        });
   216→        setShowAddModal(false);
   217→        setNewKeysText('');
   218→        setNotes('');
   219→        fetchPoolData(); // Refresh data
   220→      } else {
   221→        setAlert({ type: 'error', message: data.error || 'فشل إضافة المفاتيح' });
   222→      }
   223→    } catch (error) {
   224→      console.error('Add keys error:', error);
   225→      setAlert({ type: 'error', message: 'خطأ في الاتصال' });
   226→    } finally {
   227→      setIsAdding(false);
   228→    }
   229→  };
   230→
   231→  /**
   232→   * Revoke a key
   233→   */
   234→  const revokeKey = async (keyId: string) => {
   235→    if (!confirm('هل أنت متأكد من إلغاء هذا المفتاح؟')) return;
   236→
   237→    try {
   238→      const response = await fetch(`/api/founder-panel/api-key-pool/${keyId}`, {
   239→        method: 'DELETE',
   240→      });
   241→
   242→      const data = await response.json();
   243→      
   244→      if (data.success) {
   245→        setAlert({ type: 'success', message: '✅ تم إلغاء المفتاح' });
   246→        fetchPoolData();
   247→      } else {
   248→        setAlert({ type: 'error', message: data.error || 'فشل الإلغاء' });
   249→      }
   250→    } catch (error) {
   251→      setAlert({ type: 'error', message: 'خطأ في الاتصال' });
   252→    }
   253→  };
   254→
   255→  // ── Filtering & Pagination Logic ──────────────────────────────
   256→  
   257→  // Reset page when filters change
   258→  useEffect(() => {
   259→    setCurrentPage(1);
   260→  }, [searchQuery, statusFilter]);
   261→
   262→  // Filter keys
   263→  const filteredKeys = useMemo(() => {
   264→    return keys.filter(key => {
   265→      // Status filter
   266→      if (statusFilter !== 'all' && key.status !== statusFilter) return false;
   267→      
   268→      // Search filter
   269→      if (searchQuery) {
   270→        const query = searchQuery.toLowerCase();
   271→        return (
   272→          key.keyValue.toLowerCase().includes(query) ||
   273→          key.model.toLowerCase().includes(query) ||
   274→          key.provider.toLowerCase().includes(query) ||
   275→          key.assignedToUserName?.toLowerCase().includes(query) ||
   276→          key.assignedToCompanyName?.toLowerCase().includes(query)
   277→        );
   278→      }
   279→      
   280→      return true;
   281→    });
   282→  }, [keys, searchQuery, statusFilter]);
   283→
   284→  // Pagination calculations
   285→  const totalPages = Math.ceil(filteredKeys.length / pageSize);
   286→  const paginatedKeys = useMemo(() => {
   287→    const startIndex = (currentPage - 1) * pageSize;
   288→    return filteredKeys.slice(startIndex, startIndex + pageSize);
   289→  }, [filteredKeys, currentPage, pageSize]);
   290→
   291→  // Select All handlers
   292→  const isAllSelected = paginatedKeys.length > 0 && 
   293→    paginatedKeys.every(k => selectedKeyIds.has(k.id));
   294→  const isSomeSelected = paginatedKeys.some(k => selectedKeyIds.has(k.id));
   295→
   296→  const toggleSelectAll = () => {
   297→    if (isAllSelected) {
   298→      const newSet = new Set(selectedKeyIds);
   299→      paginatedKeys.forEach(k => newSet.delete(k.id));
   300→      setSelectedKeyIds(newSet);
   301→    } else {
   302→      setSelectedKeyIds(new Set([...selectedKeyIds, ...paginatedKeys.map(k => k.id)]));
   303→    }
   304→  };
   305→
   306→  const toggleSelectKey = (keyId: string) => {
   307→    const newSet = new Set(selectedKeyIds);
   308→    if (newSet.has(keyId)) {
   309→      newSet.delete(keyId);
   310→    } else {
   311→      newSet.add(keyId);
   312→    }
   313→    setSelectedKeyIds(newSet);
   314→  };
   315→
   316→  const handleBulkRevoke = async () => {
   317→    if (selectedKeyIds.size === 0) return;
   318→    // TODO: Implement bulk revoke API call
   319→    setAlert({ type: 'success', message: `تم إلغاء ${selectedKeyIds.size} مفتاح` });
   320→    setSelectedKeyIds(new Set());
   321→  };
   322→
   323→  // ── Render ───────────────────────────────────────────────
   324→
   325→  return (
   326→    <GarfixPageTransition>
   327→      <GarfixContainer maxWidth="7xl">
   328→        
   329→        {/* ══ Page Header ══ */}
   330→        <GarfixPageHeader
   331→          title="مجمع مفاتيح API"
   332→          subtitle="إدارة وتوزيع مفاتيح AI تلقائياً على المستخدمين الجدد"
   333→          badge={stats ? {
   334→            text: `${stats.availableKeys} متاح`,
   335→            variant: stats.availableKeys > 5 ? 'default' : 'destructive',
   336→          } : undefined}
   337→          action={
   338→            <GarfixButton
   339→              onClick={() => setShowAddModal(true)}
   340→              className="gap-2"
   341→            >
   342→              ➕ إضافة مفاتيح جديدة
   343→            </GarfixButton>
   344→          }
   345→        />
   346→
   347→        {/* ══ Alert ══ */}
   348→        {alert && (
   349→          <FadeUp delay={100}>
   350→            <GarfixAlert
   351→              type={alert.type}
   352→              message={alert.message}
   353→              onClose={() => setAlert(null)}
   354→              className="mb-6"
   355→            />
   356→          </FadeUp>
   357→        )}
   358→
   359→        {/* ══ Stats Cards ══ */}
   360→        {!isLoading && stats && (
   361→          <GarfixGrid cols={1} colsSm={2} colsMd={3} colsLg={4} className="mb-8">
   362→            <MotionCard>
   363→              <div className="text-center p-4">
   364→                <div className="text-3xl font-bold text-gray-800">{stats.totalKeys}</div>
   365→                <div className="text-sm text-gray-500 mt-1">إجمالي المفاتيح</div>
   366→              </div>
   367→            </MotionCard>
   368→
   369→            <MotionCard delay={50}>
   370→              <div className="text-center p-4">
   371→                <div className="text-3xl font-bold text-emerald-600">{stats.availableKeys}</div>
   372→                <div className="text-sm text-gray-500 mt-1">متاحة للتوزيع ✅</div>
   373→              </div>
   374→            </MotionCard>
   375→
   376→            <MotionCard delay={100}>
   377→              <div className="text-center p-4">
   378→                <div className="text-3xl font-bold text-blue-600">{stats.assignedKeys}</div>
   379→                <div className="text-sm text-gray-500 mt-1">مستخدمة حالياً 👥</div>
   380→              </div>
   381→            </MotionCard>
   382→
   383→            <MotionCard delay={150}>
   384→              <div className="text-center p-4">
   385→                <div className="text-3xl font-bold text-red-600">{stats.exhaustedKeys}</div>
   386→                <div className="text-sm text-gray-500 mt-1">منتهية ⚠️</div>
   387→              </div>
   388→            </MotionCard>
   389→          </GarfixGrid>
   390→        )}
   391→
   392→        {/* ══ Usage Today Bar ══ */}
   393→        {!isLoading && stats && (
   394→          <GarfixCard className="mb-6 p-4">
   395→            <div className="flex items-center justify-between mb-2">
   396→              <span className="text-sm font-semibold text-gray-700">استهلاك اليوم</span>
   397→              <span className="text-xs text-gray-500">{stats.totalUsageToday} طلب</span>
   398→            </div>
   399→            <div className="w-full bg-gray-200 rounded-full h-3">
   400→              <div 
   401→                className={cn(
   402→                  "h-3 rounded-full transition-all",
   403→                  stats.totalUsageToday > 800 ? "bg-red-500" :
   404→                  stats.totalUsageToday > 500 ? "bg-yellow-500" : "bg-emerald-500"
   405→                )}
   406→                style={{ width: `${Math.min((stats.totalUsageToday / 1000) * 100, 100)}%` }}
   407→              />
   408→            </div>
   409→            <p className="text-xs text-gray-500 mt-1">
   410→              {stats.totalUsageToday}/1000 طلب يومي (معدل لكل مفتاح)
   411→            </p>
   412→          </GarfixCard>
   413→        )}
   414→
   415→        {/* ══ Keys List ══ */}
   416→        <GarfixCard className="mb-6">
   417→          <div className="p-4 border-b border-gray-200">
   418→            <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   419→              <h3 className="font-bold text-gray-800">🔑 قائمة المفاتيح</h3>
   420→              
   421→              {/* Search & Filter Bar */}
   422→              <div className="flex flex-wrap items-center gap-3 w-full sm:w-auto">
   423→                {/* Search Input */}
   424→                <div className="relative flex-1 sm:w-64">
   425→                  <Search className="absolute start-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   426→                  <input
   427→                    type="text"
   428→                    placeholder="بحث في المفاتيح..."
   429→                    value={searchQuery}
   430→                    onChange={(e) => setSearchQuery(e.target.value)}
   431→                    className="w-full ps-10 pe-3 py-2 text-sm rounded-lg border border-gray-300 dark:border-gray-600 bg-white dark:bg-gray-700 focus:ring-2 focus:ring-emerald-500 focus:border-transparent"
   432→                  />
   433→                </div>
   434→                
   435→                {/* Status Filter */}
   436→                <select
   437→                  value={statusFilter}
   438→                  onChange={(e) => setStatusFilter(e.target.value)}
   439→                  className="px-3 py-2 text-sm rounded-lg border border-gray-300 dark:border-gray-600 bg-white dark:bg-gray-700 focus:ring-2 focus:ring-emerald-500"
   440→                >
   441→                  <option value="all">كل الحالات</option>
   442→                  <option value="available">✅ متاح</option>
   443→                  <option value="assigned">📤 مستخدم</option>
   444→                  <option value="exhausted">🔴 منتهي</option>
   445→                  <option value="revoked">❌ ملغي</option>
   446→                </select>
   447→              </div>
   448→            </div>
   449→            
   450→            {/* Results Count & Select All */}
   451→            {!isLoading && filteredKeys.length > 0 && (
   452→              <div className="mt-3 flex items-center justify-between text-sm">
   453→                <span className="text-gray-600">
   454→                  {filteredKeys.length} مفتاح (من {keys.length} إجمالي)
   455→                </span>
   456→                <button
   457→                  onClick={toggleSelectAll}
   458→                  className="flex items-center gap-2 px-3 py-1.5 rounded-lg hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors"
   459→                >
   460→                  {isAllSelected ? (
   461→                    <CheckSquare className="w-4 h-4 text-emerald-600" />
   462→                  ) : isSomeSelected ? (
   463→                    <CheckSquare className="w-4 h-4 text-emerald-400" />
   464→                  ) : (
   465→                    <Square className="w-4 h-4 text-muted-foreground" />
   466→                  )}
   467→                  <span>{isAllSelected ? 'إلغاء الكل' : 'تحديد الكل'}</span>
   468→                </button>
   469→              </div>
   470→            )}
   471→          </div>
   472→          
   473→          {/* Bulk Actions Bar */}
   474→          {selectedKeyIds.size > 0 && (
   475→            <div className="px-4 py-3 bg-emerald-50 dark:bg-emerald-900/20 border-b border-emerald-200 dark:border-emerald-800">
   476→              <div className="flex items-center justify-between">
   477→                <span className="text-sm font-medium text-emerald-700 dark:text-emerald-300">
   478→                  ✓ محدد: {selectedKeyIds.size} مفاتيح
   479→                </span>
   480→                <GarfixButton
   481→                  size="sm"
   482→                  variant="destructive"
   483→                  onClick={handleBulkRevoke}
   484→                >
   485→                  🗑️ إلغاء المحددة
   486→                </GarfixButton>
   487→              </div>
   488→            </div>
   489→          )}
   490→          
   491→          {isLoading ? (
   492→            <div className="p-6 space-y-4">
   493→              {[1, 2, 3].map(i => (
   494→                <GarfixSkeleton key={i} className="h-16 w-full" />
   495→              ))}
   496→            </div>
   497→          ) : filteredKeys.length === 0 ? (
   498→            <div className="p-12 text-center">
   499→              <div className="text-5xl mb-4">🗝️</div>
   500→              <h3 className="text-lg font-semibold text-gray-700 mb-2">لا توجد مفاتيح</h3>
   501→              <p className="text-sm text-gray-500 mb-4">
   502→                {searchQuery || statusFilter !== 'all' 
   503→                  ? 'لا توجد نتائج مطابقة للبحث' 
   504→                  : 'ابدأ بإضافة مفاتيح API لتوزيعها تلقائياً على المستخدمين الجدد'
   505→                }
   506→              </p>
   507→              {(!searchQuery && statusFilter === 'all') && (
   508→                <GarfixButton onClick={() => setShowAddModal(true)}>
   509→                  ➕ إضافة أول مفتاح
   510→                </GarfixButton>
   511→              )}
   512→            </div>
   513→          ) : (
   514→            <>
   515→              <div className="divide-y divide-gray-100">
   516→                {paginatedKeys.map((key, index) => (
   517→                  <FadeUp key={key.id} delay={index * 30}>
   518→                    <div className={cn(
   519→                      "p-4 hover:bg-gray-50 transition-colors",
   520→                      selectedKeyIds.has(key.id) && "bg-emerald-50/50 dark:bg-emerald-900/20"
   521→                    )}>
   522→                      <div className="flex items-start justify-between gap-4">
   523→                        {/* Checkbox + Key Info */}
   524→                        <div className="flex items-start gap-3 flex-1 min-w-0">
   525→                          {/* Checkbox */}
   526→                          <button
   527→                            onClick={() => toggleSelectKey(key.id)}
   528→                            className="flex-shrink-0 mt-1 p-1 rounded hover:bg-gray-200 dark:hover:bg-gray-700 transition-colors"
   529→                          >
   530→                            {selectedKeyIds.has(key.id) ? (
   531→                              <CheckSquare className="w-4 h-4 text-emerald-600" />
   532→                            ) : (
   533→                              <Square className="w-4 h-4 text-muted-foreground" />
   534→                            )}
   535→                          </button>
   536→                          
   537→                          {/* Key Info */}
