   561→            >
   562→              {loading ? <Loader2 size={14} className="animate-spin" /> : <Sparkles size={14} />}

   570→      {tab === "file" && (

   610→                onClick={() => { setFileName(null); setFileBase64(null); }}

   617→          <div className="flex justify-end mt-2.5">
   618→            <button
   619→              onClick={handleParseFile}
   620→              disabled={loading || !fileBase64}
   621→              className="inline-flex items-center justify-center gap-1.5 py-2.5 px-6 rounded-[10px] bg-primary text-primary-foreground border-none text-[13px] font-extrabold cursor-pointer disabled:cursor-not-allowed disabled:opacity-60 w-full sm:w-auto"
   622→            >
   623→              {loading ? <Loader2 size={14} className="animate-spin" /> : <Sparkles size={14} />}

   626→          </div>
   627→        </div>
   628→      )}
   629→
   630→      {/* Meta info */}
   631→      {meta && (
   632→        <div className="py-3 px-4 rounded-[10px] bg-accent border border-border text-[12px] text-accent-foreground flex gap-4 flex-wrap">
   633→          <span>⏱️ {meta.processingMs}ms</span>
   634→          <span>📋 {meta.ordersCount} طلب</span>
   635→          <span>📦 {meta.itemsCount} عنصر</span>
   636→          {meta.totalTokens ? <span>🧮 {meta.totalTokens} tokens</span> : null}
   637→          <span>🤖 {meta.model}</span>
   638→          {meta.retried ? <span className="text-destructive">⚠️ تطلب إعادة محاولة</span> : null}
   639→        </div>
   640→      )}
   641→
   642→      {/* Parsed orders */}
   643→      {orders.length > 0 && (
   644→        <div className="flex flex-col gap-3">
   645→          <div className="flex flex-col sm:flex-row sm:justify-between sm:items-center gap-2.5">
   646→            <h2 className="text-[16px] font-bold">
   647→              الطلبات المستخرجة ({orders.length})
   648→            </h2>
   649→            <div className="flex flex-wrap gap-2.5 items-center">
   650→              <label className="flex items-center gap-1 text-[11px] cursor-pointer">
   651→                <input
   652→                  type="checkbox"
   653→                  checked={createJournalEntries}
   654→                  onChange={(e) => setCreateJournalEntries(e.target.checked)}

   656→                إنشاء قيود محاسبية

   659→                onClick={handleSave}
   660→                disabled={saving}

   662→              >
   663→                {saving ? <Loader2 size={14} className="animate-spin" /> : <Save size={14} />}

   667→          </div>
   668→
   669→          {orders.map((order, idx) => {

   672→            const subtotal = order.items.reduce((s, it) => s + it.qty * it.unitPrice, 0);
   673→            const taxAmount = (subtotal * order.taxRate) / 100;
   674→            const total = subtotal + taxAmount + order.shipping - order.discount;
   675→            return (

   677→                key={idx}
   678→                className="bg-card rounded-lg border border-border overflow-hidden"

   680→                {/* Order header */}
   681→                <div
   682→                  className="py-3 px-4 border-b border-border flex flex-wrap justify-between items-center gap-2 bg-muted"

   684→                  <div className="flex items-center gap-2.5">

   692→                      <div className="text-[13px] font-bold">

   694→                      </div>
   695→                      <div className="text-[11px] text-muted-foreground">

   697→                      </div>
   698→                    </div>
   699→                  </div>
   700→                  <div className="flex items-center gap-2.5">
   701→                    <div className="text-[16px] font-extrabold text-primary">
   702→                      {total.toFixed(3)} {activeCompany.currency}
   703→                    </div>
   704→                    <button
   705→                      onClick={() => setEditingIdx(isEditing ? null : idx)}
   706→                      className="bg-transparent border border-border text-muted-foreground rounded-[6px] p-1 cursor-pointer flex"
   707→                      title="تعديل"
   708→                    >
   709→                      <Edit2 size={12} />
   710→                    </button>
   711→                    <button
   712→                      onClick={() => removeOrder(idx)}
   713→                      className="bg-transparent border border-border text-destructive rounded-[6px] p-1 cursor-pointer flex"
   714→                      title="حذف"
   715→                    >
   716→                      <Trash2 size={12} />
   717→                    </button>

   721→                {/* Order body (when expanded) */}
   722→                {isExpanded && (
   723→                  <div className="p-4">
   724→                    {isEditing ? (
   725→                      <div className="flex flex-col gap-3">
   726→                        <div className="grid grid-cols-[repeat(auto-fit,minmax(180px,1fr))] gap-2.5">

   758→                              className={inputStyle}

   760→                          </div>

   851→                              <th className="text-center py-1.5 px-2 font-semibold text-muted-foreground">كمية</th>

   854→                            </tr>
   855→                          </thead>
   856→                          <tbody>
   857→                            {order.items.map((item, itemIdx) => {
   858→                              // 🆕 Determine match status based on item data
   859→                              const matchStatus = getMatchStatusFromResult(item.matchResult);
   860→                              
   861→                              return (
   862→                                <tr key={itemIdx} className="border-b border-border">

   877→                                    {(item.qty * item.unitPrice).toFixed(3)}

   882→                          </tbody>

   884→                        </div>
   885→                        <div className="flex flex-wrap justify-end mt-2.5 gap-x-4 gap-y-1 text-[12px]">
   886→                          <span>مجموع: <strong>{subtotal.toFixed(3)}</strong></span>
   887→                          {order.taxRate > 0 && <span>ضريبة ({order.taxRate}%): <strong>{taxAmount.toFixed(3)}</strong></span>}
   888→                          {order.shipping > 0 && <span>شحن: <strong>{order.shipping.toFixed(3)}</strong></span>}
   889→                          {order.discount > 0 && <span>خصم: <strong>-{order.discount.toFixed(3)}</strong></span>}
   890→                          <span className="font-extrabold text-primary">الإجمالي: {total.toFixed(3)}</span>
   891→                        </div>
   892→                        {(order.clientAddress || order.notes) && (
   893→                          <div className="mt-2.5 pt-2.5 border-t border-border text-[11px] text-muted-foreground">
   894→                            {order.clientAddress && <div>📍 {order.clientAddress}</div>}
   895→                            {order.notes && <div>📝 {order.notes}</div>}
   896→                          </div>
   897→                        )}
   898→                      </div>
   899→                    )}
   900→                  </div>
   901→                )}
   902→              </div>
   903→            );
   904→          })}
   905→        </div>
   906→      )}
   907→
   908→      {/* Empty state when no orders yet */}
   909→      {!loading && orders.length === 0 && !meta && (
   910→        <div className="p-10 text-center text-muted-foreground bg-card rounded-[14px] border border-border">
   911→          <Sparkles size={36} className="opacity-30 mb-2.5" />
   912→          <div className="text-[14px] font-bold mb-1">
   913→            ابدأ بإدخال نص أو رفع صورة
   914→          </div>
   915→          <div className="text-[12px]">
   916→            سيقوم الذكاء الاصطناعي باستخراج بيانات الطلبات تلقائياً
   917→          </div>
   918→        </div>
   919→      )}
   920→
   921→      {showReviewQueue && activeCompany && (
   922→        <ReviewQueueModal
   923→          companySlug={activeCompany.slug}
   924→          onClose={() => setShowReviewQueue(false)}
   925→        />
   926→      )}
   927→    </div>
   928→  );
   929→}
