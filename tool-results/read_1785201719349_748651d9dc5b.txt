    35→// ── Helper functions ──
    36→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    37→  const charWidth = (pt) => pt * 20;
    38→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    39→  let titlePt = preferredPt;
    40→  let lines = null;
    41→  while (titlePt >= minPt) {
    42→    const cpl = charsPerLine(titlePt);
    43→    if (cpl < 2) { titlePt -= 2; continue; }
    44→    lines = splitTitleLines(title, cpl);
    45→    if (lines.length <= 3) break;
    46→    titlePt -= 2;
    47→  }
    48→  if (!lines || lines.length > 3) {
    49→    lines = splitTitleLines(title, charsPerLine(minPt));
    50→    titlePt = minPt;
    51→  }
    52→  return { titlePt, titleLines: lines };
    53→}
    54→
    55→function splitTitleLines(title, charsPerLine) {
    56→  if (title.length <= charsPerLine) return [title];
    57→  const safeBreakAfter = " \u3001\u3002\uFF0C\uFF1B\uFF1A\u2027\u2014\u2013\u2019ofandorintowithatbyforinonupout";
    58→  const lines = [];
    59→  let remaining = title;
    60→  while (remaining.length > charsPerLine) {
    61→    let breakAt = charsPerLine;
    62→    for (let i = charsPerLine; i > Math.floor(charsPerLine * 0.6); i--) {
    63→      if (safeBreakAfter.includes(remaining[i - 1])) { breakAt = i; break; }
    64→    }
    65→    if (breakAt === charsPerLine && remaining[charsPerLine]) {
    66→      for (let i = charsPerLine; i > Math.floor(charsPerLine * 0.6); i--) {
    67→        if (safeBreakAfter.includes(remaining[i])) { breakAt = i + 1; break; }
    68→      }
    69→    }
    70→    lines.push(remaining.substring(0, breakAt));
    71→    remaining = remaining.substring(breakAt);
    72→  }
    73→  if (remaining) lines.push(remaining);
    74→  // Merge orphan last line (1-2 chars) into previous
    75→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
    76→    lines[lines.length - 2] += lines[lines.length - 1];
    77→    lines.pop();
    78→  }
    79→  return lines;
    80→}

    82→function calcCoverSpacing(params) {
    83→  const {
    84→    titleLineCount = 1, titlePt = 36, hasSubtitle = false,
    85→    hasEnglishLabel = false, metaLineCount = 0, fixedHeight = 400, pageHeight = 16838,
    86→  } = params;
    87→  const titleBlock = titleLineCount * Math.ceil(titlePt * 23) + (titleLineCount > 1 ? (titleLineCount - 1) * 200 : 0);
    88→  const subtitleBlock = hasSubtitle ? 600 : 0;
    89→  const labelBlock = hasEnglishLabel ? 700 : 0;
    90→  const metaBlock = metaLineCount * 320;
    91→  const contentHeight = titleBlock + subtitleBlock + labelBlock + metaBlock + fixedHeight + 1000;
    92→  const available = pageHeight - contentHeight;
    93→  const topSpacing = Math.min(4500, Math.max(1800, Math.floor(available * 0.35)));
    94→  const bottomSpacing = Math.min(3000, Math.max(400, available - topSpacing));
    95→  return { topSpacing, midSpacing: 800, bottomSpacing };
    96→}

   122→    children.push(new Paragraph({

   140→      children: [new TextRun({ text: line, size: 24, color: P.metaColor,
   141→        font: { eastAsia: "Microsoft YaHei", ascii: "Arial" } })],
   142→    }));
   143→  }
   144→  children.push(new Paragraph({ spacing: { before: spacing.bottomSpacing } }));
   145→  children.push(new Paragraph({
   146→    indent: { left: padL, right: padR },
   147→    border: { top: { style: BorderStyle.SINGLE, size: 2, color: P.accent, space: 8 } },
   148→    spacing: { before: 200 },
   149→    children: [
   150→      new TextRun({ text: config.footerLeft || "", size: 16, color: P.footerColor, font: { ascii: "Arial" } }),
   151→      new TextRun({ text: "                                        " }),
   152→      new TextRun({ text: config.footerRight || "", size: 16, color: P.footerColor, font: { ascii: "Arial" } }),
   153→    ],
   154→  }));

   156→    width: { size: 100, type: WidthType.PERCENTAGE },
   157→    layout: "FIXED",
   158→    borders: allNoBorders,

   242→    })),
   243→  });

   245→    cantSplit: true,

   247→      width: { size: colWidths[ci], type: WidthType.PERCENTAGE },
   248→      shading: { type: ShadingType.CLEAR, fill: ri % 2 === 0 ? t.surface : "FFFFFF" },

   251→        font: { eastAsia: "Microsoft YaHei", ascii: "Arial" } })] })],

   253→  }));
   254→  return new Table({

   261→      insideVertical: { style: BorderStyle.NONE },
   262→    },
   263→    rows: [headerRow, ...dataRows],
   264→  });
   265→}
